# state_snapshot.txt — 仪表盘快照生成器（L9 · 公理 2）
# 用法: python3 state_snapshot.txt   （读库计数 → 更新 state/.library-state.json + 重写 DASHBOARD.md）
import os, re, json, sys
from datetime import date

sys.stdout.reconfigure(encoding='utf-8') if hasattr(sys.stdout, 'reconfigure') else None
ROOT = os.getcwd()
STATE = os.path.join(ROOT, 'state', '.library-state.json')
DASH = os.path.join(ROOT, 'state', 'DASHBOARD.md')

def count_lib():
    c = {'pitfalls': 0, 'patterns': 0, 'decisions': 0, 'reflections': 0}
    for lib, key in [('knowledge/pitfalls.md', 'pitfalls'), ('knowledge/patterns.md', 'patterns'),
                     ('knowledge/decisions.md', 'decisions'), ('knowledge/reflections.md', 'reflections')]:
        p = os.path.join(ROOT, lib)
        if not os.path.exists(p): continue
        text = re.sub(r'<!--.*?-->', '', open(p, encoding='utf-8').read(), flags=re.S)
        c[key] = sum(1 for b in re.split(r'\n### ', text)[1:] if re.match(r'^[A-Z]{1,2}-\d+', b))
    return c

def links_total():
    n = 0
    for root, _, files in os.walk(ROOT):
        if any(x in root for x in ['.git', 'node_modules', 'experimental', '_sandbox']): continue
        for f in files:
            if f.endswith('.md'):
                n += len(re.findall(r'\]\(([^)#]+)', open(os.path.join(root, f), encoding='utf-8').read()))
    return n

def main():
    state = {'schema': 'library-state-v1', 'updated_at': str(date.today()),
             'counts': count_lib(), 'links_total': links_total()}
    if os.path.exists(STATE):
        old = json.load(open(STATE, encoding='utf-8'))
        for k in ['emotion', 'goal', 'ablation', 'sleep']:
            if k in old: state[k] = old[k]
    state.setdefault('emotion', {'valence': 0.0, 'arousal': 0.0, 'certainty': 0.5, 'stakes': 0.0})
    os.makedirs(os.path.dirname(STATE), exist_ok=True)
    json.dump(state, open(STATE, 'w', encoding='utf-8'), ensure_ascii=False, indent=2)
    e = state['emotion']
    lines = ['# 仪表盘（%s 快照）' % date.today(), '',
             '## 概览',
             '- 条目：pitfalls %d / patterns %d / decisions %d / reflections %d' % tuple(state['counts'][k] for k in ['pitfalls','patterns','decisions','reflections']),
             '- 互链：%d 条 · 情绪向量：valence %.1f / arousal %.1f / certainty %.1f / stakes %.1f' % (state['links_total'], e['valence'], e['arousal'], e['certainty'], e['stakes']),
             '- 当前目标：%s · 睡眠固化：上次 %s（闸门拦截 %d 条）' % (state['goal'].get('current',''), state['sleep'].get('last_run','从未'), state['sleep'].get('gated_count', 0))]
    open(DASH, 'w', encoding='utf-8').write('\n'.join(lines) + '\n')
    print('快照已更新: 条目', state['counts'])

if __name__ == '__main__':
    main()
