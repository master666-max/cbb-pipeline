# run_tests.txt — library-bootstrap 安装模拟器（测试 T1-T7）
# 用法: py -X utf8 run_tests.txt
# 模拟 agent 按 install.md 逐模块安装到 sandbox（模板复制+填槽的最小实现），
# 然后跑 bootstrap_verify / lint / 幂等 / 依赖拒绝 / 双入口 断言。
import os, re, shutil, subprocess, sys, json

sys.stdout.reconfigure(encoding='utf-8')
SKILL = os.path.dirname(os.path.abspath(__file__))
SB = os.path.join(SKILL, '_sandbox')
PASS, FAIL = [], []

def slot_fill(text):
    return (text.replace('{{SLOT:WORKSPACE_NAME}}', '测试工作区').replace('{{SLOT:HARNESS_NAME}}', 'ZCode')
                .replace('{{SLOT:DATE}}', '2026-09-04').replace('{{SLOT:OWNER_NAME}}', 'miza')
                .replace('{{SLOT:PY_RUNNER}}', 'py -X utf8').replace('{{SLOT:CRON_MODE}}', 'manual')
                .replace('{{SLOT:ENTRY_FILE}}', 'AGENTS.md').replace('{{SLOT:EXTRA_LINES}}', '')
                .replace('{{SLOT:...}}', ''))

def install_file(src_rel, dst_rel):
    src = os.path.join(SKILL, src_rel)
    dst = os.path.join(SB, dst_rel)
    os.makedirs(os.path.dirname(dst), exist_ok=True)
    txt = slot_fill(open(src, encoding='utf-8').read())
    existed = os.path.exists(dst)
    open(dst, 'w', encoding='utf-8').write(txt)
    return existed

def pyrun(*args):
    r = subprocess.run([sys.executable, '-X', 'utf8'] + list(args), capture_output=True, text=True, cwd=SB, encoding='utf-8')
    return r.returncode, (r.stdout or '') + (r.stderr or '')

def fresh():
    shutil.rmtree(SB, ignore_errors=True)
    os.makedirs(SB)

def install_core():
    install_file('modules/m-core/template/ROOT-README.md', 'README.md')
    install_file('modules/m-core/template/pitfalls.md', 'knowledge/pitfalls.md')
    install_file('modules/m-core/template/patterns.md', 'knowledge/patterns.md')
    install_file('modules/m-core/template/_trajectories.md', 'knowledge/trajectories/.keep')
    for d in ['knowledge/skills', 'knowledge/trajectories', 'archive/_cold', 'archive/2026', 'workspace', 'env']:
        os.makedirs(os.path.join(SB, d), exist_ok=True)

def install_index():
    install_file('modules/m-index/template/knowledge/_index.md', 'knowledge/_index.md')
    install_file('adapters/entries/AGENTS.md', 'AGENTS.md')
    install_file('adapters/entries/CLAUDE.md', 'CLAUDE.md')

def install_loop():
    install_file('modules/m-loop/template/knowledge/skills/task-start/SKILL.md', 'knowledge/skills/task-start/SKILL.md')
    install_file('modules/m-loop/template/knowledge/skills/wrap-up/SKILL.md', 'knowledge/skills/wrap-up/SKILL.md')

def install_tools():
    install_file('modules/m-tools/template/tools/lint_library.txt', 'tools/lint_library.txt')
    install_file('modules/m-tools/template/tools/search_knowledge.txt', 'tools/search_knowledge.txt')

def verify(*args):
    return pyrun(os.path.join(SKILL, 'scripts', 'bootstrap_verify.txt'), *args)

def check(name, cond, detail=''):
    (PASS if cond else FAIL).append(name)
    print(('PASS' if cond else 'FAIL') + ' | ' + name + ((' | ' + detail) if detail and not cond else ''))
