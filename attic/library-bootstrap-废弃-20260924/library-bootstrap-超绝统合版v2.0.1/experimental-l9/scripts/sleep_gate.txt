# sleep_gate.txt — 睡眠固化闸门（L9 · 三道检查防污染）
# 用法: python3 sleep_gate.txt <草稿文件路径>   （草稿=拟写入 patterns/pitfalls 的新条目块）
import os, re, sys

sys.stdout.reconfigure(encoding='utf-8') if hasattr(sys.stdout, 'reconfigure') else None
ROOT = os.getcwd()
problems = []

if len(sys.argv) < 2:
    print('用法: sleep_gate.txt <草稿文件>'); sys.exit(2)
draft = open(sys.argv[1], encoding='utf-8').read()

# 闸门①证据校验：实例链接必须 exists
for m in re.finditer(r'^- 实例：(.+)$', draft, re.M):
    p = m.group(1).strip().rstrip('`')
    if not os.path.exists(os.path.join(ROOT, p)) and not os.path.exists(p):
        problems.append('闸门①证据缺失: %s' % p)
if not re.search(r'^- 实例：', draft, re.M):
    problems.append('闸门①草稿缺「实例」行（无证据不入库）')

# 闸门②措辞纪律：禁第一人称情绪断言（公理 1/7）
for bad in ['我高兴', '我感到', '让我开心', '真棒', '令人兴奋', '我自豪']:
    if bad in draft:
        problems.append('闸门②措辞越界: 含第一人称情绪断言「%s」' % bad)

# 闸门③去重冲突：关键词与既有条目撞车
kw = re.search(r'- 关键词：(.*)', draft)
if kw:
    for lib in ['knowledge/pitfalls.md', 'knowledge/patterns.md']:
        p = os.path.join(ROOT, lib)
        if not os.path.exists(p): continue
        existing = open(p, encoding='utf-8').read()
        for term in kw.group(1).split()[:3]:
            if re.search(r'- 关键词：[^\n]*' + re.escape(term), existing):
                problems.append('闸门③疑似与既有条目撞关键词「%s」（去重后再提交）' % term)

for p in problems:
    print('✗', p)
print('sleep_gate: %d 问题 → %s' % (len(problems), '拦截' if problems else '放行（可双写皮层+DISTILL）'))
sys.exit(1 if problems else 0)
