def read(p):
    MONITOR["files_read"] += 1
    return io.open(os.path.join(ROOT, p), encoding="utf-8").read()

def parse_entries(text, prefix):
    """按 ### {prefix}-NNN 切条目；模板占位（{日期}/{模式名}）跳过并计数"""
    out, cur = [], None
    for line in text.splitlines():
        m = re.match(r"^### (" + prefix + r"-\d+) / (.+?) / (.+)$", line.strip())
        if m:
            if cur: out.append(cur)
            eid, date, title = m.group(1), m.group(2).strip(), m.group(3).strip()
            if "{" in date or "{" in title:
                MONITOR["skipped_template"] += 1
                cur = None; continue
            if not re.match(r"^\d{4}-\d{2}-\d{2}$", date):
                MONITOR["no_date"].append(eid); date = "2026-09-06"
            cur = {"id": eid, "date": date, "title": title, "fields": {}}
            continue
        if cur is not None:
            fm = re.match(r"^-( .+?)：(.*)$", line.rstrip())
            if fm: cur["fields"].setdefault(fm.group(1).strip(), []).append(fm.group(2).strip())
            else: cur["fields"].setdefault("_body", []).append(line.rstrip())
    if cur: out.append(cur)
    return out

def core_content(e):
    """标题 + 首个核心字段句（教训/决策/洞察/核心步骤），截 220 字"""
    f = e["fields"]
    for k in ("教训", "决策", "洞察", "核心步骤"):
        if k in f:
            return (e["title"] + "：" + f[k][0])[:220]
    body = " ".join(x.strip() for x in f.get("_body", []) if x.strip())
    return (e["title"] + "：" + body)[:220] if body else e["title"]

def to_entry(e, default_imp):
    f = e["fields"]
    content = core_content(e)
    if len(content) <= len(e["title"]) + 1:
        MONITOR["field_gaps"].append(e["id"] + "(content 空)")
    kw = []
    for sep in ("、", " ", "，"):
        raw = f.get("关键词", [])
        if raw: kw = [x for x in re.split(r"[、，]", raw[0]) if x]; break
    links = re.findall(r"\[\[(.+?)\]\]", " ".join(sum(f.values(), [])))
    state = (f.get("状态", [""])[0])
    imp = 3 if "superseded" in state else (7 if state == "active" or "决策" in f else default_imp)
    conf = "高" if f.get("置信度", [""])[0].startswith("高") else ("中" if f.get("置信度") else "中")
    return {"id": e["id"], "created_at": e["date"], "updated_at": "", "content": content,
            "keywords": kw[:12], "links": list(dict.fromkeys(links))[:8],
            "source_event_id": "genesis-v4", "importance": imp, "confidence": conf,
            "validity": {"t_valid": e["date"]}}

entries = []
for name, path in SRC.items():
    text = read(path)
    prefix = {"pitfalls": "P", "patterns": "PT", "reflections": "R", "decisions": "D"}[name]
    es = parse_entries(text, prefix)
    MONITOR["parsed"] += len(es)
    for e in es:
        entries.append(to_entry(e, default_imp=6))
