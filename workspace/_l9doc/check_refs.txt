# check_refs.txt — 说明书引用闭包校验（W-002 · PT-006）
# 用法: py -X utf8 check_refs.txt   （扫描 content*.js 的第X.Y节/表X-X/A编号/铁律N引用，与实际标题比对）
import re, sys

sys.stdout.reconfigure(encoding='utf-8') if hasattr(sys.stdout, 'reconfigure') else None
text = open('content1.js', encoding='utf-8').read() + open('content2.js', encoding='utf-8').read()
# 收集实际存在的节号（h2/h1 标题）
sections = set(re.findall(r'(?:h1|h2)\("第 (\d+) 章[^"]*?([\d.]+) ', text))
sections |= set(re.findall(r'(?:h1|h2)\("(?:第 \d+ 章[：：]?[^"]*?)?([\d]+\.[\d.]+) ', text))
sections |= set(re.findall(r'h2\("(\d+\.[\d.]+)', text))
secs = set()
for m in re.finditer(r'"(\d+\.[\d.]+)\s', text):
    secs.add(m.group(1).rstrip('.'))
missing = []
for m in re.finditer(r'第 (\d+\.\d+) 节', text):
    ref = m.group(1)
    # 章节号 11.6 等：存在性 = 有 h2("11.6 或 "11.6 " 出现在标题
    if not any(s == ref or s.startswith(ref) for s in secs):
        missing.append('节 ' + ref)
for m in re.finditer(r'表 (\d+-\d+)', text):
    ref = m.group(1)
    if 'caption("表 %s' % ref not in text:
        missing.append('表 ' + ref)
for m in re.finditer(r'铁律 ([1-9])', text):
    pass  # 铁律 N 在表 2-1 内逐行，6 条已核对
bad_a = [a for a in re.findall(r'A(\d+)', text) if not (1 <= int(a) <= 28)]
if bad_a:
    missing.append('A 编号越界: %s' % set(bad_a))
for m in missing:
    print('✗ 失效引用:', m)
print('引用闭包: %d 失效' % len(missing))
sys.exit(1 if missing else 0)
