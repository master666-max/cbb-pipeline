# X11 · E1 受限演化试点 —— 库调试工作区（真实库）首个受控参数演化
# 运行：py -X utf8 proto/x11_e1_pilot.txt
# 治理框架（两轴模型 E1 档，见 库调试工作区/references/两轴模型与演化授权-v1.md）：
#   - 只演化法层 law_params；filter_zero 宪章冻结在 0（W-1 真实库否决）
#   - 选择压力：train 23 条金标（recall@5 + 0.3×MRR 合成效用）
#   - 晋升门：train 效用配对提升 > margin（停摆监控：连续 5 代零采纳 margin 减半一次）
#   - 验收门（真引擎）：held 10 条 recall@5 不劣化 且 全量 33 条 recall@5 不劣化（W-1 教训）
#   - 预算账本：记评估单位；停摆触发记录
#   - 演化在内存中进行（库只读）；胜者经人侧裁决形态写入 state.law_params + 审计链 + 刷根
# 输出：data/x11_e1_pilot.json、data/x11_e1_winner.json（如有）
import sys, re, json, time, random, importlib.util, importlib.machinery
from pathlib import Path
from contextlib import redirect_stdout
from io import StringIO

BASE = Path(__file__).resolve().parent.parent
VENDOR = BASE / "proto" / "vendor" / "bootstrap_v3.9.py"   # 现行发布引擎（74e5d44d 谱系）
LIB = Path(r"D:\zcode专用！！！！危险！！！！！！！！！\库调试工作区")
DATA = BASE / "data"
META = json.loads((DATA / "meta_layer.json").read_text(encoding="utf-8"))

def load_mod(name, path):
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_file_location(name, str(path), loader=loader)
    mod = importlib.util.module_from_spec(spec)
    sys.modules[name] = mod
    spec.loader.exec_module(mod)
    return mod

evocore = load_mod("evocore", BASE / "proto" / "evocore.txt")
m = load_mod("bs39", VENDOR)
# 元层基线（现行 v3.9 修订默认）
BASE_CFG = {"w_kw": 3.0, "w_content": 1.0, "w_imp": 0.5, "w_age": 0.1, "filter_zero": 0}
MARGIN = META["margin"]
GENS, KIDS = 25, 3
SEEDS = [1, 2, 3, 4, 5]
PARAMS = ["w_kw", "w_content", "w_imp", "w_age"]     # filter_zero 宪章冻结

def recall5(lib, tasks, cfg):
    hits = 0
    for t in tasks:
        top = [fid for fid, _ in evocore.rank_lib(lib, t["query"], cfg)[:5]]
        if t["expected"][0] in top:
            hits += 1
    return hits / max(1, len(tasks))

def mrr_of(lib, tasks, cfg):
    rr = 0.0
    for t in tasks:
        top = [fid for fid, _ in evocore.rank_lib(lib, t["query"], cfg)[:5]]
        if t["expected"][0] in top:
            rr += 1.0 / (top.index(t["expected"][0]) + 1)
    return rr / max(1, len(tasks))

def U(lib, tasks, cfg):
    return recall5(lib, tasks, cfg) + 0.3 * mrr_of(lib, tasks, cfg)

def mutate(cfg, rnd):
    d = dict(cfg)
    p = rnd.choice(PARAMS)
    d[p] = round(max(0.0, d[p] * rnd.choice([0.7, 0.85, 1.2, 1.5])), 4)
    return d

allrows = []
t0 = time.time()
for seed in SEEDS:
    entries = {}
    for f in LIB.joinpath("memory").rglob("*.json"):
        e = json.loads(f.read_text(encoding="utf-8"))
        entries[f.stem] = e
    gq = json.loads((LIB / "evals" / "golden_queries.json").read_text(encoding="utf-8"))
    rnd = random.Random(seed * 911 + 17)
    idx = list(range(len(gq)))
    rnd.shuffle(idx)
    train = [gq[i] for i in idx[:23]]
    held = [gq[i] for i in idx[23:]]
    cache = {}
    def sc(cfg, tasks):
        key = (tuple(sorted(cfg.items())), id(tasks))
        if key not in cache:
            cache[key] = U(LIB, tasks, cfg)
        return cache[key]
    mainline = dict(BASE_CFG)
    margin_eff = MARGIN
    stall = 0
    evals = 0
    hist = []
    for gen in range(1, GENS + 1):
        promoted = 0
        for _ in range(KIDS):
            child = mutate(mainline, rnd)
            u_main = sc(mainline, train)
            u_child = sc(child, train)
            d = u_child - u_main
            if d > margin_eff:
                mainline = dict(child)
                promoted += 1
        if promoted == 0:
            stall += 1
            if stall >= 5 and margin_eff > MARGIN / 2:
                margin_eff = MARGIN / 2
                stall = 0
        else:
            stall = 0
        hist.append({"gen": gen, "promoted": promoted, "train_U": round(sc(mainline, train), 4)})
    # 验收门（真实全量 + held）
    held_r_base = recall5(LIB, held, BASE_CFG)
    held_r_new = recall5(LIB, held, mainline)
    full_r_base = recall5(LIB, train + held, BASE_CFG)
    full_r_new = recall5(LIB, train + held, mainline)
    r = {"seed": seed, "mainline": mainline, "margin_eff": margin_eff,
         "held_recall_base": round(held_r_base, 4), "held_recall_new": round(held_r_new, 4),
         "full_recall_base": round(full_r_base, 4), "full_recall_new": round(full_r_new, 4),
         "accept": bool(held_r_new >= held_r_base - 1e-9 and full_r_new >= full_r_base - 1e-9),
         "evals": len(cache), "hist_tail": hist[-3:]}
    allrows.append(r)
    print(f"[s{seed}] held {r['held_recall_base']}→{r['held_recall_new']} 全量 {r['full_recall_base']}→{r['full_recall_new']} "
          f"{'✓接受' if r['accept'] else '✗拒绝'} cfg={json.dumps(mainline, ensure_ascii=False)}")

# 汇总与晋升裁决
n_accept = sum(1 for r in allrows if r["accept"])
major = {}
for r in allrows:
    if r["accept"]:
        key = tuple(sorted(r["mainline"].items()))
        major[key] = major.get(key, 0) + 1
best = max(major.items(), key=lambda x: x[1]) if major else (None, 0)
winner = dict(best[0]) if best[0] else None
print(f"\n===== E1 试点汇总：{n_accept}/5 种子过验收门 =====")
if winner:
    print(f"胜者配置（{best[1]}/5 种子）：{json.dumps(winner, ensure_ascii=False)}")
    print(f"vs 现行：{json.dumps(BASE_CFG, ensure_ascii=False)}")
    (DATA / "x11_e1_winner.json").write_text(json.dumps(
        {"winner": winner, "n_accept": n_accept, "rows": allrows}, ensure_ascii=False, indent=1), encoding="utf-8")
else:
    print("无种子过验收门——现行配置保持")
(DATA / "x11_e1_pilot.json").write_text(json.dumps(
    {"baseline": BASE_CFG, "rows": allrows, "n_accept": n_accept, "winner": winner,
     "wall_s": round(time.time() - t0, 1)}, ensure_ascii=False, indent=1), encoding="utf-8")
print(f"E1 试点完成 {time.time()-t0:.1f}s → data/x11_e1_pilot.json")
