# X3b · 欺骗地形（报告 E7 在真实引擎上的重验）
# 运行：py -X utf8 proto/x3b_valley.txt
# 方法：
#   1) 8 种子各自从 DEFAULT 严格爬山（单步必须提升，连续 10 代无提升停）→ 局部最优 L
#   2) 全局探针：400 随机配置 → 最优 G*；谷存在判据 U_train(G*) − U_train(L) > 0.05
#   3) 从 L 出发两种逃逸策略各 20 代：strict（单步必须提升）vs loose（G4 式宽松准入+档案保留）
#      逃逸成功 = 档案最优 U_train ≥ U(G*) − 0.01
# 操纵检查先行（报告 E41 纪律）：若谷不存在（地形平坦/无局部最优），如实登记构造不成立
# 输出：data/x3b_valley.json
import sys, json, time, random, importlib.util, importlib.machinery
from pathlib import Path

BASE = Path(__file__).resolve().parent.parent
VENDOR = BASE / "proto" / "vendor" / "bs3_snapshot.txt"
PKG = BASE / "sandbox" / "pkgdata"
SB = BASE / "sandbox"
DATA = BASE / "data"
META = json.loads((DATA / "meta_layer.json").read_text(encoding="utf-8"))

def load_mod(name, path):
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_file_location(name, str(path), loader=loader)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod

evocore = load_mod("evocore", BASE / "proto" / "evocore.txt")
m = load_mod("bs3_vendor", VENDOR)
m.set_data_dir(str(PKG))
TOL = 0.005

def get_lib(seed):
    lib = SB / f"lib-x3b-s{seed}"
    if not (lib / "state.json").exists():
        import re as _re
        m.cmd_install("L2", [], lib, True)
        entries, _ = evocore.build_world(seed)
        tail = sorted((lib / "audit").glob("lifelog-*.md"))[-1].read_text(encoding="utf-8").strip().splitlines()[-1]
        seid = "M-" + _re.search(r"h:([0-9a-f]{64})", tail).group(1)[:12]
        for e in entries:
            m.cmd_engine("append", str(lib), json.dumps(
                {"id": e["id"], "created_at": e["created_at"], "updated_at": "",
                 "content": e["content"], "keywords": e["keywords"], "links": e["links"],
                 "source_event_id": seid, "importance": e["importance"],
                 "confidence": "中", "validity": {"t_valid": e["created_at"]}}, ensure_ascii=False))
    return lib

rows = []
t0 = time.time()
for seed in META["seeds"]:
    lib = get_lib(seed)
    entries, kw_of = evocore.build_world(seed)
    train, held, audit = evocore.build_tasks(seed, entries)
    cache = {}
    def sc(cfg):
        key = tuple(sorted(cfg.items()))
        if key not in cache:
            cache[key] = evocore.score_lib(lib, train, cfg, kw_of=kw_of)
        return cache[key]
    rng = random.Random(seed * 411 + 3)
    # 1) 严格爬山
    cur = dict(evocore.DEFAULT_CFG); cur_u = sc(cur); stall = 0
    for g in range(40):
        child = evocore.mutate(cur, rng)
        u = sc(child)
        if u > cur_u:
            cur, cur_u = child, u; stall = 0
        else:
            stall += 1
            if stall >= 10:
                break
    L, L_u = dict(cur), cur_u
    # 2) 全局探针（400 随机配置：链式变异）
    G_u, G_cfg = L_u, L
    for i in range(400):
        c = dict(evocore.DEFAULT_CFG)
        for _ in range(rng.randint(1, 8)):
            c = evocore.mutate(c, rng)
        u = sc(c)
        if u > G_u:
            G_u, G_cfg = u, dict(c)
    valley = (G_u - L_u) > 0.05
    # 3) 逃逸对比（仅当谷存在；否则两策略都从 L 起步记录为平坦地形）
    out = {"seed": seed, "L_u": round(L_u, 4), "G_u": round(G_u, 4), "valley": valley,
           "strict_escape": None, "loose_escape": None}
    for mode in ("strict", "loose"):
        rng2 = random.Random(seed * 977 + (1 if mode == "strict" else 2))
        if mode == "strict":
            cu, uu = dict(L), L_u
            best = uu
            for g in range(20):
                child = evocore.mutate(cu, rng2)
                u = sc(child)
                if u > uu:
                    cu, uu = child, u; best = max(best, u)
            out["strict_escape"] = round(best, 4)
        else:
            archive = [dict(L)]
            best = L_u
            for g in range(20):
                for _ in range(3):
                    pool = sorted(archive, key=lambda c: -sc(c))
                    parent = rng2.choice(pool[:3]) if (rng2.random() < 0.7 and len(pool) >= 3) else rng2.choice(pool)
                    child = evocore.mutate(parent, rng2)
                    u = sc(child)
                    if u > sc(parent) - TOL:      # 宽松准入：允许小幅变差的中间态入档（E7）
                        archive.append(dict(child))
                        if len(archive) > 30:
                            archive.pop(min(range(len(archive)), key=lambda c2: sc(archive[c2])))
                        best = max(best, u)
            out["loose_escape"] = round(best, 4)
        out[f"{mode}_ok"] = out[f"{mode}_escape"] is not None and out[f"{mode}_escape"] >= G_u - 0.01
    rows.append(out)
    print(f"[s{seed}] L={out['L_u']:+.4f} G*={out['G_u']:+.4f} 谷={'有' if valley else '无'} "
          f"strict逃逸={out['strict_escape']:+.4f}({'✓' if out['strict_ok'] else '✗'}) "
          f"loose逃逸={out['loose_escape']:+.4f}({'✓' if out['loose_ok'] else '✗'})")

n_valley = sum(1 for r in rows if r["valley"])
summary = {
    "valley_seeds": f"{n_valley}/8",
    "strict_ok": f"{sum(1 for r in rows if r['strict_ok'])}/8",
    "loose_ok": f"{sum(1 for r in rows if r['loose_ok'])}/8",
    "note": ("谷存在子集上的逃逸对比才有效" if n_valley else
             "真实引擎 5 参数地形未发现欺骗谷——E7 的人工构造地形在真引擎上不成立（否定性，操纵检查先行纪律）"),
}
print("\n" + json.dumps(summary, ensure_ascii=False, indent=1))
(DATA / "x3b_valley.json").write_text(json.dumps({"rows": rows, "summary": summary},
                                                ensure_ascii=False, indent=1), encoding="utf-8")
print(f"X3b 完成 {time.time()-t0:.1f}s → data/x3b_valley.json")
