def load_mod(name, path):
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_file_location(name, str(path), loader=loader)
    mod = importlib.util.module_from_spec(spec)
    sys.modules[name] = mod
    spec.loader.exec_module(mod)
    return mod

gq = json.loads((LIB / "evals" / "golden_queries.json").read_text(encoding="utf-8"))
queries = []
for i, x in enumerate(gq):
    queries.append((x["query"], x["expected"][0], "golden"))
    if i in PARAPHRASE:
        queries.append((PARAPHRASE[i], x["expected"][0], "para"))
print(f"影子查询集：{len(queries)} 条（原始 33 + 自然改写 {len(queries)-33}）")

# 条目特征预载
entries = {}
for f in LIB.joinpath("memory").rglob("*.json"):
    e = json.loads(f.read_text(encoding="utf-8"))
    entries[f.stem] = e
import time as _t
def feats_of(fid, q):
    e = entries[fid]
    kws = " ".join(e.get("keywords") or [])
    qws = q.split()
    kw = sum(1 for w in qws if w in kws)
    ct = sum(1 for w in qws if w in e["content"])
    imp = float(e.get("importance", 0))
    try:
        age = (_t.time() - _t.mktime(_t.strptime(e["created_at"][:10], "%Y-%m-%d"))) / 86400
    except Exception:
        age = 0.0
    return {"kw": float(kw), "cont": float(ct), "imp": imp, "age": round(age, 2)}

(LIB / "evals" / "s0_traces.jsonl").touch(exist_ok=True)   # 绕过 pilot 首跑 bug：getsize 对不存在文件抛错被静默吞
P = pilot.Pilot(str(LIB / "evals" / "s0_traces.jsonl"), anonymize=False)
n_shown, n_adopt = 0, 0
for q, golden, src in queries:
    ranked = evocore.rank_lib(LIB, q, V39)[:5]
    cands = []
    for fid, (score, e, c) in ranked:
        cands.append({"id": fid, "feat": feats_of(fid, q)})   # pilot.log 读的是 feat 键（落盘后转为 f）
    adopted = golden if golden in [c["id"] for c in cands] else None
    P.log(q, cands, adopted_id=adopted, extra={"src": src})
    n_shown += 1
    if adopted:
        n_adopt += 1
print(f"影子记录完成：{n_shown} 条轨迹，其中金标命中（采纳）{n_adopt} 条")
