# 解包交接 zip（修正 CP437 乱码文件名）→ 大审查/混元/自演化离线实验/
import zipfile
from pathlib import Path

SRC = Path(r"D:\zcode专用！！！！危险！！！！！！！！！\大审查\混元\自演化离线实验_完整包-交接.zip")
DST = Path(r"D:\zcode专用！！！！危险！！！！！！！！！\大审查\混元")

zf = zipfile.ZipFile(SRC)
fixed, raw = 0, 0
for info in zf.infolist():
    name = info.filename
    try:
        fixed_name = name.encode("cp437").decode("utf-8")
        raw += 1
    except (UnicodeEncodeError, UnicodeDecodeError):
        fixed_name = name
    if fixed_name != name:
        fixed += 1
    target = DST / fixed_name
    if info.is_dir():
        target.mkdir(parents=True, exist_ok=True)
        continue
    target.parent.mkdir(parents=True, exist_ok=True)
    target.write_bytes(zf.read(info))
print(f"解包完成：{len(zf.infolist())} 条目，修正乱码名 {fixed} 个 → {DST / '自演化离线实验'}")
