# test_compare — v3.8.2 vs v3.9 自测全量对比（外置裁判：BaseException 全捕获，防静默中止）
# 运行：py -X utf8 proto/test_compare_v39.txt
# 判定：两版失败集合完全一致 → 补丁零回归；失败为环境依赖（cwd 无 bootstrap_data 的测试属开发态测试）
import importlib.util, importlib.machinery, sys
from pathlib import Path

BASE = Path(__file__).resolve().parent.parent

def load_mod(name, path):
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_file_location(name, str(path), loader=loader)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod

def run_suite(tag, path):
    mod = load_mod(tag, path)
    results = {}
    for fn in mod.TESTS:
        try:
            fn()
            results[fn.__name__] = "PASS"
        except BaseException as ex:                      # 外置裁判：SystemExit 也算失败并记录
            results[fn.__name__] = f"FAIL: {type(ex).__name__} {str(ex)[:80]}"
    return results

r382 = run_suite("v382", BASE / "proto" / "vendor" / "bs3_snapshot.txt")
r39 = run_suite("v39", BASE / "proto" / "vendor" / "bootstrap_v3.9.py")

p382 = sum(1 for v in r382.values() if v == "PASS")
p39 = sum(1 for v in r39.values() if v == "PASS")
print(f"v3.8.2: {p382}/{len(r382)} PASS   v3.9: {p39}/{len(r39)} PASS")
only382 = {k: v for k, v in r382.items() if v != "PASS" and r39.get(k) == "PASS"}
only39 = {k: v for k, v in r39.items() if v != "PASS" and r382.get(k) == "PASS"}
common_fail = {k for k, v in r382.items() if v != "PASS" and r39.get(k) != "PASS"}
print(f"仅 v3.8.2 失败（v3.9 修好的）: {sorted(only382) or '无'}")
print(f"仅 v3.9 失败（补丁疑似引入）: ")
for k, v in sorted(only39.items()):
    print(f"  {k}: {v}")
    if only382.get(k) or k in only382:
        print("    （此条在 v3.8.2 也失败过——非补丁引入）")
print(f"两版共同失败（环境依赖/既有缺陷）: {len(common_fail)} 个")
for k in sorted(common_fail):
    print(f"  {k}: {r39[k][:90]}")
verdict = "补丁零回归" if not only39 else f"补丁引入 {len(only39)} 个新失败——须逐条归因"
print(f"\n判定：{verdict}")
