# migrate_sha256.txt — SHA-1 manifest 一次性迁移到 SHA-256（附录 E 步骤 3-4）
# 用法: python3 migrate_sha256.txt   （在已装库根目录运行；幂等，重跑零变更）
import os, sys, json, hashlib

sys.stdout.reconfigure(encoding='utf-8') if hasattr(sys.stdout, 'reconfigure') else None
MF = '.library-manifest.json'
BAK = '.library-manifest.sha1.bak'

def h256(p):
    return hashlib.sha256(open(p, 'rb').read()).hexdigest()[:16]

if not os.path.exists(MF):
    print('无 manifest，退出'); sys.exit(0)
mf = json.load(open(MF, encoding='utf-8'))
if mf.get('hash_algo') == 'sha256-16':
    print('已是 SHA-256，幂等退出（零变更）'); sys.exit(0)
# 旧 manifest 备份（铁律 1：不覆盖、不删除）
if not os.path.exists(BAK):
    os.rename(MF, BAK)
old = json.load(open(BAK, encoding='utf-8'))
new = dict(old)
new['hash_algo'] = 'sha256-16'
new['files'] = [{'path': f['path'], 'sha256_16': h256(f['path'])} for f in old['files'] if os.path.exists(f['path'])]
json.dump(new, open(MF, 'w', encoding='utf-8'), ensure_ascii=False, indent=2)
print('迁移完成: %d 文件重算 SHA-256；旧 manifest 保留为 %s' % (len(new['files']), BAK))
