# exp_vector_search.txt — TF-IDF 语义检索兜底（实验模块，需 scikit-learn）
# 用法: python3 exp_vector_search.txt --build | python3 exp_vector_search.txt 关键词
# 无 sklearn 时打印提示并退出 2（由 search 层决定降级）。
import sys, os, re, json
sys.stdout.reconfigure(encoding='utf-8') if hasattr(sys.stdout, 'reconfigure') else None
try:
    from sklearn.feature_extraction.text import TfidfVectorizer
    from sklearn.metrics.pairwise import cosine_similarity
except ImportError:
    print('[实验模块] 缺 scikit-learn，语义兜底不可用（pip install scikit-learn）'); sys.exit(2)

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
IDX = os.path.join(os.path.dirname(os.path.abspath(__file__)), '.vec_index.pkl')

def entries():
    out = []
    for lib in ['knowledge/pitfalls.md', 'knowledge/patterns.md', 'knowledge/reflections.md', 'knowledge/decisions.md']:
        p = os.path.join(ROOT, lib)
        if not os.path.exists(p): continue
        text = re.sub(r'<!--.*?-->', '', open(p, encoding='utf-8').read(), flags=re.S)
        for block in re.split(r'\n### ', text)[1:]:
            pid = block.partition('\n')[0].split('/')[0].strip()
            if re.match(r'^[A-Z]{1,2}-\d+$', pid):
                out.append((pid, block))
    return out

if '--build' in sys.argv:
    es = entries()
    docs = [b for _, b in es]
    if not docs:
        print('空库：暂无条目，向量索引留待首条知识写入后再构建'); sys.exit(0)
    vec = TfidfVectorizer(analyzer='char_wb', ngram_range=(2, 4)).fit(docs)
    import pickle
    pickle.dump({'ids': [i for i, _ in es], 'matrix': vec.transform(docs), 'vocab': vec}, open(IDX, 'wb'))
    print('向量索引构建完成: %d 条' % len(es)); sys.exit(0)

q = ' '.join(sys.argv[1:])
if not q: print('用法: exp_vector_search.txt 关键词'); sys.exit(2)
if not os.path.exists(IDX): print('先 --build 构建索引'); sys.exit(2)
import pickle
idx = pickle.load(open(IDD := IDX, 'rb'))
qv = idx['vocab'].transform([q])
sims = cosine_similarity(qv, idx['matrix'])[0]
ranked = sorted(zip(idx['ids'], sims), key=lambda x: -x[1])[:5]
for pid, s in ranked:
    if s > 0.01: print('☆语义 [%s] %.2f' % (pid, s))
