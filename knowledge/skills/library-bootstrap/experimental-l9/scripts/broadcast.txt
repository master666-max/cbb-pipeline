# broadcast.txt — GWT 广播仲裁器（L9 · 谁有权写入上下文窗口，显式竞争）
# 用法: python3 broadcast.txt --task "任务描述"
# 候选源 = {关键词直击条目, reflections 摘要}；打分 = 相关性 × 情绪调制；落选→PENDING(当前只写不读；回读功能属机器变异候选，需走 MUTATION-SOP)
import os, re, sys, json

sys.stdout.reconfigure(encoding='utf-8') if hasattr(sys.stdout, 'reconfigure') else None
ROOT = os.getcwd()
STATE = os.path.join(ROOT, 'state', '.library-state.json')
PENDING = os.path.join(ROOT, 'state', 'PENDING.md')

def load_state():
    if os.path.exists(STATE):
        return json.load(open(STATE, encoding='utf-8'))
    return {'emotion': {'valence': 0.0, 'arousal': 0.0, 'certainty': 0.5, 'stakes': 0.0}, 'ablation': {}, 'goal': {}}

def parse_entries():
    out = {}
    for lib in ['knowledge/pitfalls.md', 'knowledge/patterns.md', 'knowledge/reflections.md', 'knowledge/decisions.md']:
        p = os.path.join(ROOT, lib)
        if not os.path.exists(p): continue
        text = re.sub(r'<!--.*?-->', '', open(p, encoding='utf-8').read(), flags=re.S)
        for block in re.split(r'\n### ', text)[1:]:
            head = block.partition('\n')[0]
            pid = head.split('/')[0].strip()
            if re.match(r'^[A-Z]{1,2}-\d+$', pid):
                body = block[block.index('\n') + 1:]
                out[pid] = body
    return out

# W-020 打分公式（权重单点定义，登记待 W-022 校准）：
#   score = 0.6*词面重叠度 + 0.3*图距离衰减项 + 0.1*情绪调制项
#   词面重叠度与 search_knowledge 的 relevance 共用实现（0/0.33/0.67/1.0 四档归一）
#   图距离衰减项 = 邻居一跳命中记 0.5（与 L2 扩散激活同源）
#   情绪调制项 = 0.5 + 0.25*arousal*探索类(1/0) - 0.25*(1-certainty)*猜测类(1/0)，正负对称
WEIGHT_LEX, WEIGHT_GRAPH, WEIGHT_EMO = 0.6, 0.3, 0.1
def emo_mod(s, is_exploratory, is_speculative):
    m = 0.5 + 0.25 * s['emotion']['arousal'] * (1 if is_exploratory else 0)
    m -= 0.25 * (1 - s['emotion']['certainty']) * (1 if is_speculative else 0)
    return max(0.0, m)

def score_entry(eid, body, terms, s, graph_hits):
    lex = 0
    kw = re.search(r'- 关键词：(.*)', body)
    kw_text = kw.group(1) if kw else ''
    for t in terms:
        if t in kw_text: lex += 3
        elif t in eid: lex += 2
        elif t in body: lex += 1
    lex_norm = min(lex / 9.0, 1.0)
    graph = 0.5 if eid in graph_hits else 0.0
    emo = emo_mod(s, bool(kw_text), '【待确认】' in body)
    return WEIGHT_LEX * lex_norm + WEIGHT_GRAPH * graph + WEIGHT_EMO * emo

def main():
    if '--task' not in sys.argv:
        print('用法: broadcast.txt --task "任务描述"'); sys.exit(2)
    task = sys.argv[sys.argv.index('--task') + 1]
    s = load_state()
    if s.get('ablation', {}).get('gwt'):
        print('⚠ GWT 已消融（state_intervene --ablate gwt）——跳过仲裁，task-start 退回自由检索')
        return
    terms = [t for t in re.split(r'[\s，。、/]+', task) if t.strip()][:4]
    entries = parse_entries()
    direct = [eid for eid, b in entries.items() if score_entry(eid, b, terms, s, set()) > 0]
    graph_hits = set()
    for eid in direct:
        for nb in re.findall(r'[[]([A-Z]{1,2}-\d+)[]]', entries[eid]):
            graph_hits.add(nb)
    scored = [(eid, score_entry(eid, b, terms, s, graph_hits)) for eid, b in entries.items()]
    scored = [(eid, round(sc, 3)) for eid, sc in scored if sc > 0]
    scored.sort(key=lambda x: -x[1])
    top = scored[:3]
    # W-019 arousal 四象限仲裁：复杂度中介 = 直查命中条目数（≤2 简单 / ≥3 复杂）
    n_direct = len(direct)
    mode = ''
    if s['emotion']['arousal'] > 0.5:
        if n_direct <= 2:
            mode = '收敛模式（简单任务×高唤醒：响应短、检索 k 减半）'
            top = top[:max(1, len(top) // 2)]
        else:
            mode = '探索模式（复杂任务×高唤醒：检索激进度 +0.3、响应长度不变）'
    else:
        mode = '常规模式（低唤醒：两模式均不激活）'
    print('广播清单（task=%s, arousal=%.1f, certainty=%.1f, 直查=%d, 模式=%s）:' % (task, s['emotion']['arousal'], s['emotion']['certainty'], n_direct, mode))
    for eid, sc in top:
        print('  注入 %s（分 %.3f）→ knowledge 对应库' % (eid, sc))
    if '探索模式' in mode:
        explored = [nb for eid, _ in top for nb in re.findall(r'[[]([A-Z]{1,2}-\d+)[]]', entries.get(eid, '')) if nb not in [e for e, _ in top]]
        for nb in explored[:3]:
            print('  探索扩召 %s（邻居 +0.3 激进度）→ knowledge 对应库' % nb)
    rest = scored[3:]
    if rest:
        os.makedirs(os.path.dirname(PENDING), exist_ok=True)
        with open(PENDING, 'a', encoding='utf-8') as f:
            for eid, sc in rest:
                f.write('- %s（分 %.0f，%s 落选降级）\n' % (eid, sc, __import__('datetime').date.today()))
        print('  落选 %d 条 → state/PENDING.md（不丢弃）' % len(rest))
    if not scored:
        print('  无候选（索引空或全不相关）→ 建议直接开工')

if __name__ == '__main__':
    main()
