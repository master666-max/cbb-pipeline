# -*- coding: utf-8 -*-
import os, re
cdir = os.path.join(r"D:\zcode专用！！！！危险！！！！！！！！！\迷深清洗工作", "异世界迷宫最深部_知识库", "文字")
lines = ["# 迷深语料地图（阶段0自动生成：每个文件的完整标题列表与话数范围）", ""]
for root, dirs, files in os.walk(cdir):
    for f in sorted(files):
        if not f.endswith(".md"):
            continue
        p = os.path.join(root, f)
        rel = os.path.relpath(p, cdir).replace(os.sep, "/")
        txt = open(p, encoding="utf-8", errors="replace").read()
        heads = re.findall(r"^#{1,3}\s*(.+)$", txt, re.M)
        nums = sorted(set(int(m) for m in re.findall(r"第(\d+)话", txt)))
        rng = f"（含话数标记 {nums[0]}-{nums[-1]}）" if nums else ""
        lines.append(f"## {rel}  〔{len(txt)}字〕{rng}")
        for h in heads:
            h = h.strip()
            if len(h) < 80:
                lines.append(f"- {h}")
        lines.append("")
out = os.path.join(os.path.dirname(cdir), "..", "reports", "_语料地图.md")
out = os.path.normpath(out)
open(out, "w", encoding="utf-8").write("\n".join(lines))
print("written:", out, "lines:", len(lines))
