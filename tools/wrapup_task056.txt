import io, re

p = base + r"\knowledge\_index.md"
s = io.open(p, encoding="utf-8").read()

n = int(m.group(1)) + 1

s = s.replace(anchor_line, anchor_line + "\n" + t56, 1)
io.open(p, "w", encoding="utf-8", newline="\n").write(s)
print("index updated: count ->", n, "; task-056 line appended")
