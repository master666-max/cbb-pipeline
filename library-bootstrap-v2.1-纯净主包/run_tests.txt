PASS, FAIL = [], []

def slot_fill(text):
    return (text.replace('{{SLOT:WORKSPACE_NAME}}', '测试工作区').replace('{{SLOT:HARNESS_NAME}}', 'ZCode')
                .replace('{{SLOT:DATE}}', '2026-09-04').replace('{{SLOT:OWNER_NAME}}', 'miza')
                .replace('{{SLOT:PY_RUNNER}}', 'py -X utf8').replace('{{SLOT:CRON_MODE}}', 'manual')
                .replace('{{SLOT:ENTRY_FILE}}', 'AGENTS.md').replace('{{SLOT:EXTRA_LINES}}', '')
                .replace('{{SLOT:...}}', ''))

def install_file(src_rel, dst_rel):
    src = os.path.join(SKILL, src_rel)
    dst = os.path.join(SB, dst_rel)
    os.makedirs(os.path.dirname(dst), exist_ok=True)
    txt = slot_fill(open(src, encoding='utf-8').read())
    existed = os.path.exists(dst)
    open(dst, 'w', encoding='utf-8').write(txt)
    return existed

def pyrun(*args):
    r = subprocess.run([sys.executable, '-X', 'utf8'] + list(args), capture_output=True, text=True, cwd=SB, encoding='utf-8')
    return r.returncode, (r.stdout or '') + (r.stderr or '')

def fresh():
    shutil.rmtree(SB, ignore_errors=True)

def verify(*args):

def check(name, cond, detail=''):
    (PASS if cond else FAIL).append(name)
    print(('PASS' if cond else 'FAIL') + ' | ' + name + ((' | ' + detail) if detail and not cond else ''))

# T30: SHA-256 迁移幂等（沙盒：造 sha1 旧 manifest → 迁移两次零变更）
fresh(); install_core()
import hashlib as hl
def sha1_8(p): return hl.sha1(open(p, 'rb').read()).hexdigest()[:8]
tracked = ['README.md', 'knowledge/pitfalls.md', 'knowledge/patterns.md']
json.dump({'skill': 'library-bootstrap', 'files': [{'path': p, 'sha1_8': sha1_8(os.path.join(SB, p))} for p in tracked]},
          open(os.path.join(SB, '.library-manifest.json'), 'w', encoding='utf-8'))
install_file('experimental-l8/scripts/migrate_sha256.txt', 'tools/migrate_sha256.txt')
rc1, o1 = pyrun('tools/migrate_sha256.txt')
rc2, o2 = pyrun('tools/migrate_sha256.txt')
mf = json.load(open(os.path.join(SB, '.library-manifest.json'), encoding='utf-8'))
check('T30a 迁移至 sha256-16', rc1 == 0 and mf.get('hash_algo') == 'sha256-16' and all('sha256_16' in f for f in mf['files']))
check('T30b 迁移幂等（重跑零变更）', rc2 == 0 and '幂等' in o2, o2)
check('T30c 旧 manifest 备份保留', os.path.exists(os.path.join(SB, '.library-manifest.sha1.bak')))
