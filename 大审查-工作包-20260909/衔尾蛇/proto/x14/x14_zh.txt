# X14 · 中文歧义词世界（真实式语义子串歧义；判据冻结于 reports/X14_中文歧义词世界_预注册.md）
# 运行：py -X utf8 proto/x14/x14_zh.txt
# 隔离：加载 proto/x14/evocore_zh.txt（仅 build_world 换为中文歧义词世界，脚本断言）；vendor sha 断言；库前缀 sandbox/x14-*；输出 data/x14-*。
# 判据：≥1 种子受伤 → 语义歧义通道迁移成立（失败金标词须全为族头词）；0/24 → 本 prevalence 下不足以致伤。
import sys, re, json, time, random, importlib.util, importlib.machinery, hashlib
from pathlib import Path
from contextlib import redirect_stdout
from io import StringIO

BASE = Path(__file__).resolve().parent.parent.parent
VENDOR = BASE / "proto" / "vendor" / "bs3_snapshot.txt"
PKG = BASE / "sandbox" / "pkgdata"
SB = BASE / "sandbox"
DATA = BASE / "data"
META = json.loads((DATA / "meta_layer.json").read_text(encoding="utf-8"))

h = hashlib.sha256(VENDOR.read_bytes()).hexdigest()[:16]
assert h == "ef2c645aea64fb9e", f"vendor sha 变动：{h}"
a = (BASE / "proto" / "evocore.txt").read_text(encoding="utf-8").splitlines()
b = (BASE / "proto" / "x14" / "evocore_zh.txt").read_text(encoding="utf-8").splitlines()
d = [i + 1 for i, (x, y) in enumerate(zip(a, b)) if x != y]
assert b[:25] == a[:25] and (not d or min(d) >= 26), f"反事实核心越界：diff 首行 {min(d) if d else 0}"
print(f"隔离断言 OK：vendor={h}，核心 diff 自第 {min(d)} 行起（build_world 区域）", flush=True)

HEADS = {"测试", "学习", "网络", "安全", "数据", "系统", "模型", "索引"}

def load_mod(name, path):
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_file_location(name, str(path), loader=loader)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod

evo = load_mod("evocore_zh", BASE / "proto" / "x14" / "evocore_zh.txt")
m = load_mod("bs3_vendor", VENDOR)
m.set_data_dir(str(PKG))
TOL, GENS, KIDS = 0.005, META["gens"], META["kids"]
SEEDS = list(range(1, 25))

# 普查断言
entries, _ = evo.build_world(1)
hit = sum(1 for e in entries if "测试" in " ".join(e["keywords"]))
assert hit == 12, f"测试 命中 {hit} ≠ 12"
print(f"普查断言 OK：测试 子串命中 = {hit}（预注册 12）", flush=True)

def build_lib(seed):
    lib = SB / f"x14-ng-s{seed}"
    if not (lib / "state.json").exists():
        m.cmd_install("L2", [], lib, True)
        entries, _ = evo.build_world(seed)
        tail = sorted((lib / "audit").glob("lifelog-*.md"))[-1].read_text(encoding="utf-8").strip().splitlines()[-1]
        seid = "M-" + re.search(r"h:([0-9a-f]{64})", tail).group(1)[:12]
        for e in entries:
            m.cmd_engine("append", str(lib), json.dumps(
                {"id": e["id"], "created_at": e["created_at"], "updated_at": "",
                 "content": e["content"], "keywords": e["keywords"], "links": e["links"],
                 "source_event_id": seid, "importance": e["importance"],
                 "confidence": "中", "validity": {"t_valid": e["created_at"]}}, ensure_ascii=False))
    return lib

def evolve_ng(seed):
    lib = build_lib(seed)
    entries, kw_of = evo.build_world(seed)
    train, held, audit = evo.build_tasks(seed, entries)
    rng = random.Random(seed * 31 + sum(map(ord, "PL")))    # 同流键——与 X8–X13 配对
    mainline = dict(evo.DEFAULT_CFG)
    archive = [dict(mainline)]
    cache = {}
    def sc(cfg):
        key = tuple(sorted(cfg.items()))
        if key not in cache:
            cache[key] = evo.score_lib(lib, train, cfg, kw_of=kw_of)
        return cache[key]
    for gen in range(1, GENS + 1):
        for _ in range(KIDS):
            pool = sorted(archive, key=lambda c: -sc(c))
            parent = rng.choice(pool[:3]) if (rng.random() < 0.7 and len(pool) >= 3) else rng.choice(pool)
            child = evo.mutate(parent, rng)
            if sc(child) - sc(parent) > -TOL:
                archive.append(dict(child))
                if len(archive) > 30:
                    archive.pop(min(range(len(archive)), key=lambda i: sc(archive[i])))
            if sc(child) > sc(mainline):
                mainline = dict(child)
    # 失败任务归因（测量）
    failed_words = []
    for t in audit:
        rid = [f for f, _ in evo.rank_lib(lib, t["q"], mainline)]
        if t["golden"] not in rid[:5]:
            failed_words.append(t["kw"])
    res = {"seed": seed, "final_mainline": mainline,
           "audit_recall5": round(evo.recall_at_k(lib, audit, mainline, kw_of=kw_of), 4),
           "failed_words": sorted(set(failed_words))}
    buf = StringIO()
    with redirect_stdout(buf):
        m.cmd_engine("doctor", str(lib))
    res["doctor_ok"] = "全一致" in buf.getvalue()
    (DATA / f"x14-ng-s{seed}.json").write_text(json.dumps(res, ensure_ascii=False, indent=1), encoding="utf-8")
    return res

t0 = time.time()
print("== 中文歧义词世界演化 seeds 1–24 ==", flush=True)
rows = []
for seed in SEEDS:
    r = evolve_ng(seed)
    rows.append(r)
    print(f"[s{seed}] task={r['audit_recall5']:.2f} 失败词={r['failed_words'] or '无'} "
          f"{r['final_mainline']} doctor={'OK' if r['doctor_ok'] else 'BAD'}", flush=True)

injured = [r for r in rows if r["audit_recall5"] < 0.95]
injured_seeds = [r["seed"] for r in injured]
print(f"受伤种子: {injured_seeds or '无'}", flush=True)
if injured:
    all_heads = all(w in HEADS for r in injured for w in r["failed_words"]) and all(r["failed_words"] for r in injured)
    verdict = "迁移成立：语义歧义通道在真实式词表下独立致伤" + ("" if all_heads else "（存在非头词失败——另有通道，如实登记）")
else:
    verdict = "语义歧义在 15% prevalence 下不足以致伤（W-7 降级观察项）"
print(f"判定：{verdict}")
(DATA / "x14_summary.json").write_text(json.dumps(
    {"summary": {"injured_seeds": injured_seeds, "判定": verdict}, "rows": rows},
    ensure_ascii=False, indent=1), encoding="utf-8")
print(f"X14 完成 {time.time()-t0:.1f}s → data/x14_summary.json")
