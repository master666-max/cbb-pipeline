# X19 · 门位分离检验（混元 E7 的 X 世界版；判据冻结于 reports/X19_门位分离_预注册.md）
# 运行：py -X utf8 proto/x19_gate_position.txt
# 设计：AG 臂——archive 准入加 held 非回退门（对 parent），主线加冕保持 train 规则（与 NG 全同）。
#       对照 = x8-NG 存量（晋升位门与无门均已测）。同流键 "PL"。
# 输出：data/x19-ag-s{seed}.json、data/x19_summary.json
import sys, re, json, time, random, importlib.util, importlib.machinery
from pathlib import Path
from contextlib import redirect_stdout
from io import StringIO

BASE = Path(__file__).resolve().parent.parent
VENDOR = BASE / "proto" / "vendor" / "bs3_snapshot.txt"
PKG = BASE / "sandbox" / "pkgdata"
SB = BASE / "sandbox"
DATA = BASE / "data"
META = json.loads((DATA / "meta_layer.json").read_text(encoding="utf-8"))

def load_mod(name, path):
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_file_location(name, str(path), loader=loader)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod

evo = load_mod("evocore_named", BASE / "proto" / "x12" / "evocore_named.txt")
m = load_mod("bs3_vendor", VENDOR)
m.set_data_dir(str(PKG))
TOL, GENS, KIDS = 0.005, META["gens"], META["kids"]
SEEDS = list(range(1, 25))

def build_lib(seed):
    lib = SB / f"x19-ag-s{seed}"
    if not (lib / "state.json").exists():
        m.cmd_install("L2", [], lib, True)
        entries, _ = evo.build_world(seed)
        tail = sorted((lib / "audit").glob("lifelog-*.md"))[-1].read_text(encoding="utf-8").strip().splitlines()[-1]
        seid = "M-" + re.search(r"h:([0-9a-f]{64})", tail).group(1)[:12]
        for e in entries:
            m.cmd_engine("append", str(lib), json.dumps(
                {"id": e["id"], "created_at": e["created_at"], "updated_at": "",
                 "content": e["content"], "keywords": e["keywords"], "links": e["links"],
                 "source_event_id": seid, "importance": e["importance"],
                 "confidence": "中", "validity": {"t_valid": e["created_at"]}}, ensure_ascii=False))
    return lib

def evolve_ag(seed):
    lib = build_lib(seed)
    entries, kw_of = evo.build_world(seed)
    train, held, audit = evo.build_tasks(seed, entries)
    rng = random.Random(seed * 31 + sum(map(ord, "PL")))
    mainline = dict(evo.DEFAULT_CFG)
    archive = [dict(mainline)]
    cache = {}
    def sc(cfg, tasks, tag):
        key = (tag, tuple(sorted(cfg.items())))
        if key not in cache:
            cache[key] = evo.score_lib(lib, tasks, cfg, kw_of=kw_of)
        return cache[key]
    for gen in range(1, GENS + 1):
        for _ in range(KIDS):
            pool = sorted(archive, key=lambda c: -sc(c, train, "tr"))
            parent = rng.choice(pool[:3]) if (rng.random() < 0.7 and len(pool) >= 3) else rng.choice(pool)
            child = evo.mutate(parent, rng)
            train_ok = sc(child, train, "tr") - sc(parent, train, "tr") > -TOL
            held_ok = sc(child, held, "hd") >= sc(parent, held, "hd") - TOL
            if train_ok and held_ok:                            # 准入门：train ∧ held 双条件（预注册 §一）
                archive.append(dict(child))
                if len(archive) > 30:
                    archive.pop(min(range(len(archive)), key=lambda i: sc(archive[i], train, "tr")))
            if sc(child, train, "tr") > sc(mainline, train, "tr"):   # 主线加冕与 NG 全同（train 单面）
                mainline = dict(child)
    res = {"seed": seed, "final_mainline": mainline,
           "audit_recall5": round(evo.recall_at_k(lib, audit, mainline, kw_of=kw_of), 4)}
    buf = StringIO()
    with redirect_stdout(buf):
        m.cmd_engine("doctor", str(lib))
    res["doctor_ok"] = "全一致" in buf.getvalue()
    (DATA / f"x19-ag-s{seed}.json").write_text(json.dumps(res, ensure_ascii=False, indent=1), encoding="utf-8")
    return res

t0 = time.time()
rows = []
for seed in SEEDS:
    r = evolve_ag(seed)
    rows.append(r)
    print(f"[s{seed}] task={r['audit_recall5']:.2f} {r['final_mainline']} doctor={'OK' if r['doctor_ok'] else 'BAD'}", flush=True)

injured = sorted(r["seed"] for r in rows if r["audit_recall5"] < 0.95)
ng_injured = [5, 8, 11, 13]
print(f"AG 受伤种子: {injured or '无'}（NG 基线受伤 = {ng_injured}）")
if len(injured) <= 1:
    verdict = "SUPPORTED（门位是 X8 失败的关键变量：准入位拦截有效）"
elif len(injured) <= 3:
    verdict = "PARTIAL（准入位部分有效）"
else:
    verdict = "NOT SUPPORTED（门位无关，X8 失败另有原因）"
print(f"判定：{verdict}")
(DATA / "x19_summary.json").write_text(json.dumps({"summary": {"injured": injured, "ng_injured": ng_injured,
                                                               "判定": verdict}, "rows": rows},
                                                  ensure_ascii=False, indent=1), encoding="utf-8")
print(f"X19 完成 {time.time()-t0:.1f}s")
