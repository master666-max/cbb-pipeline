# F1 · 通道②真实库 prevalence 普查 + 通道③基线扫描（只读，W-8/W-9 前置输入）
# 运行：py -X utf8 proto/f1_real_prevalence.txt
# 扫描对象：sandbox/lib-real（真实工作区库的迁移副本，X5 建库）。零新数据收集，纯只读。
# 输出：data/f1_real_prevalence.json
import json, re, time
from pathlib import Path
from collections import Counter

BASE = Path(__file__).resolve().parent.parent
LIB = BASE / "sandbox" / "lib-real"
OUT = BASE / "data" / "f1_real_prevalence.json"
NOW = time.time()

def load_entries(lib):
    out = []
    for p in sorted(lib.rglob("*.json")):
        if p.name == "state.json":
            continue
        try:
            e = json.loads(p.read_text(encoding="utf-8"))
        except Exception:
            continue
        if not isinstance(e, dict) or "content" not in e:
            continue
        out.append(e)
    return out

entries = load_entries(LIB)
n = len(entries)

# ── 通道②：关键词子串碰撞暴露面 ──
kw_lists = [(" ".join(e.get("keywords") or [])).strip() for e in entries]
kw_lists = [k for k in kw_lists if k]
uniq_kws = sorted({k for k in kw_lists})
collide = {}                                               # 关键词 -> 作为子串命中的其他条目数
for k in uniq_kws:
    hit = sum(1 for e in entries
              if k in " ".join(e.get("keywords") or []) and k not in (e.get("keywords") or []))
    hit_direct = sum(1 for e in entries if k in " ".join(e.get("keywords") or []))
    collide[k] = {"entries_with_kw": hit_direct, "cross_family_hits": hit - hit_direct if False else hit_direct - 1 if hit_direct else 0,
                  "substring_total": hit_direct}
    collide[k] = {"entries_with_kw": hit_direct}
# 头词式碰撞：kw 是其他 kw 的真子串
head = {k: [x for x in uniq_kws if x != k and k in x] for k in uniq_kws}
head = {k: v for k, v in head.items() if v}
kw_hit_counts = {k: sum(1 for e in entries if k in " ".join(e.get("keywords") or [])) for k in uniq_kws}

# ── 通道③：年龄结构 + importance ──
ages, imps = [], []
for e in entries:
    try:
        c = time.strptime((e.get("created_at") or "")[:10], "%Y-%m-%d")
        ages.append(round((NOW - time.mktime(c)) / 86400, 1))
    except Exception:
        pass
    if isinstance(e.get("importance"), (int, float)):
        imps.append(e["importance"])

# ── 淹没面：以每条目的首个关键词为查询，内容命中条目数（c_hit≥1 的竞争者）──
flood = []
for e in entries:
    kws = e.get("keywords") or []
    if not kws:
        continue
    q = kws[0]
    c = sum(1 for x in entries if any(w and w in (x.get("content") or "") for w in [q]) and x.get("id") != e.get("id"))
    flood.append(c)

summary = {
    "n_entries": n,
    "n_unique_keywords": len(uniq_kws),
    "channel2_head_collisions": head,
    "channel2_kw_hit_counts": kw_hit_counts,
    "channel3_age_days": {"n": len(ages), "min": min(ages) if ages else None,
                          "median": sorted(ages)[len(ages)//2] if ages else None,
                          "max": max(ages) if ages else None},
    "importance": {"mean": round(sum(imps)/len(imps), 2) if imps else None,
                   "max": max(imps) if imps else None},
    "flood_surface_per_kw": {"min": min(flood) if flood else None,
                             "median": sorted(flood)[len(flood)//2] if flood else None,
                             "max": max(flood) if flood else None},
}
print(f"条目数={n} 唯一关键词={len(uniq_kws)}")
print(f"通道②头词式碰撞（kw⊂其他kw）: {head or '无'}")
print(f"通道③年龄: n={len(ages)} 中位={summary['channel3_age_days']['median']} 天 最老={summary['channel3_age_days']['max']} 天")
print(f"淹没面（关键词查询的内容命中竞争者）: 中位={summary['flood_surface_per_kw']['median']} 最大={summary['flood_surface_per_kw']['max']}")
(OUT).write_text(json.dumps({"summary": summary,
                             "kw_hit_counts": kw_hit_counts,
                             "ages": ages}, ensure_ascii=False, indent=1), encoding="utf-8")
print(f"完成 → {OUT.name}")
