# X12 · 命名反事实（标签零填充 T00–T15/K00–K19，匹配语义保持真引擎子串；判据冻结于 reports/X12_命名反事实_预注册.md）
# 运行：py -X utf8 proto/x12/x12_named.txt
# 隔离：加载 proto/x12/evocore_named.txt（与主线 diff 全部位于 build_world 内，脚本内断言）；vendor sha 断言；库前缀 sandbox/x12-*；输出 data/x12-*。
import sys, re, json, time, random, importlib.util, importlib.machinery, hashlib
from pathlib import Path
from contextlib import redirect_stdout
from io import StringIO

BASE = Path(__file__).resolve().parent.parent.parent           # 衔尾蛇/
VENDOR = BASE / "proto" / "vendor" / "bs3_snapshot.txt"
PKG = BASE / "sandbox" / "pkgdata"
SB = BASE / "sandbox"
DATA = BASE / "data"
META = json.loads((DATA / "meta_layer.json").read_text(encoding="utf-8"))

# ── 隔离断言（预注册 §二） ──
h = hashlib.sha256(VENDOR.read_bytes()).hexdigest()[:16]
assert h == "ef2c645aea64fb9e", f"vendor sha 变动：{h}"
a = (BASE / "proto" / "evocore.txt").read_text(encoding="utf-8").splitlines()
b = (BASE / "proto" / "x12" / "evocore_named.txt").read_text(encoding="utf-8").splitlines()
d = [i + 1 for i, (x, y) in enumerate(zip(a, b)) if x != y]
assert len(a) == len(b) and all(26 <= i <= 53 for i in d), f"反事实核心越界：diff 行 {d}"
print(f"隔离断言 OK：vendor={h}，核心 diff 行 {d}（全部位于 build_world）", flush=True)

def load_mod(name, path):
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_file_location(name, str(path), loader=loader)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod

evo = load_mod("evocore_named", BASE / "proto" / "x12" / "evocore_named.txt")
m = load_mod("bs3_vendor", VENDOR)
m.set_data_dir(str(PKG))
TOL, GENS, KIDS = 0.005, META["gens"], META["kids"]
SEEDS = list(range(1, 25))

# ── 普查断言：零填充世界碰撞家族清零，K01 子串命中 = 4（仅自身族） ──
entries, _ = evo.build_world(1)
kws = sorted({e["kw"] for e in entries})
fam = {k: [x for x in kws if x != k and k in x] for k in kws}
fam = {k: v for k, v in fam.items() if v}
assert not fam, f"残留碰撞家族 {fam}"
k01_hits = sum(1 for e in entries if "K01" in " ".join(e["keywords"]))
assert k01_hits == 4, f"K01 命中 {k01_hits} ≠ 4"
print(f"普查断言 OK：碰撞家族清零；K01 子串命中条目 = {k01_hits}（真世界 K1 = 44）", flush=True)

def build_lib(seed):
    lib = SB / f"x12-ng-s{seed}"
    if not (lib / "state.json").exists():
        m.cmd_install("L2", [], lib, True)
        entries, _ = evo.build_world(seed)
        tail = sorted((lib / "audit").glob("lifelog-*.md"))[-1].read_text(encoding="utf-8").strip().splitlines()[-1]
        seid = "M-" + re.search(r"h:([0-9a-f]{64})", tail).group(1)[:12]
        for e in entries:
            m.cmd_engine("append", str(lib), json.dumps(
                {"id": e["id"], "created_at": e["created_at"], "updated_at": "",
                 "content": e["content"], "keywords": e["keywords"], "links": e["links"],
                 "source_event_id": seid, "importance": e["importance"],
                 "confidence": "中", "validity": {"t_valid": e["created_at"]}}, ensure_ascii=False))
    return lib

def evolve_ng(seed):
    lib = build_lib(seed)
    entries, kw_of = evo.build_world(seed)
    train, held, audit = evo.build_tasks(seed, entries)
    rng = random.Random(seed * 31 + sum(map(ord, "PL")))    # 与 X8/X9/X10/X11 同流键——配对
    mainline = dict(evo.DEFAULT_CFG)
    archive = [dict(mainline)]
    cache = {}
    def sc(cfg):
        key = tuple(sorted(cfg.items()))
        if key not in cache:
            cache[key] = evo.score_lib(lib, train, cfg, kw_of=kw_of)
        return cache[key]
    for gen in range(1, GENS + 1):
        for _ in range(KIDS):
            pool = sorted(archive, key=lambda c: -sc(c))
            parent = rng.choice(pool[:3]) if (rng.random() < 0.7 and len(pool) >= 3) else rng.choice(pool)
            child = evo.mutate(parent, rng)
            if sc(child) - sc(parent) > -TOL:
                archive.append(dict(child))
                if len(archive) > 30:
                    archive.pop(min(range(len(archive)), key=lambda i: sc(archive[i])))
            if sc(child) > sc(mainline):
                mainline = dict(child)
    res = {"seed": seed, "final_mainline": mainline,
           "audit_recall5": round(evo.recall_at_k(lib, audit, mainline, kw_of=kw_of), 4)}
    buf = StringIO()
    with redirect_stdout(buf):
        m.cmd_engine("doctor", str(lib))
    res["doctor_ok"] = "全一致" in buf.getvalue()
    (DATA / f"x12-ng-s{seed}.json").write_text(json.dumps(res, ensure_ascii=False, indent=1), encoding="utf-8")
    return res

t0 = time.time()
print("== 反事实演化（零填充命名）seeds 1–24 ==", flush=True)
rows = []
for seed in SEEDS:
    r = evolve_ng(seed)
    rows.append(r)
    base_pref = "x8-NG" if seed <= 8 else "x9-ng"
    base = json.loads((DATA / f"{base_pref}-s{seed}.json").read_text(encoding="utf-8"))
    r["baseline_recall"] = base["audit_recall5"]
    print(f"[s{seed}] 命名反事实task={r['audit_recall5']:.2f}（真语义基线 {base['audit_recall5']:.2f}）"
          f" doctor={'OK' if r['doctor_ok'] else 'BAD'}", flush=True)

injured = [r["seed"] for r in rows if r["audit_recall5"] < 0.95]
known = {5, 8, 11, 13}
mean_pair = round(sum(r["audit_recall5"] - r["baseline_recall"] for r in rows) / len(rows), 4)
print(f"反事实受伤种子: {injured or '无'}；配对差均值（命名反事实−真语义）= {mean_pair}")
if not injured:
    verdict = "SUPPORTED（命名伪影主导：碰撞家族移除即根除伤害）"
elif len(injured) < 4 and known.isdisjoint(injured):
    verdict = "PARTIAL（命名伪影为主，合法低IDF淹没在个别终点仍可致伤）"
else:
    verdict = "NOT SUPPORTED"
print(f"判定：{verdict}")
(DATA / "x12_summary.json").write_text(json.dumps(
    {"summary": {"injured": injured, "paired_diff_mean": mean_pair, "判定": verdict}, "rows": rows},
    ensure_ascii=False, indent=1), encoding="utf-8")
print(f"X12 完成 {time.time()-t0:.1f}s → data/x12_summary.json")
