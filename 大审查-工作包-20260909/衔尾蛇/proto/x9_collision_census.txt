# X9 · 碰撞面普查与危险切分预报器（测量线，零新机制；判据冻结于 reports/X9_碰撞面普查_预注册.md）
# 运行：py -X utf8 proto/x9_collision_census.txt
# 三段：①前瞻组演化（seeds 9–24，NG 臂=X8 无门基线逐位同构）②碰撞面普查 ③预报器（L1∧L2 规则，判据先冻结）对回溯+前瞻两组出混淆矩阵。
# 输出：data/x9-ng-s{seed}.json、data/x9_census.json、data/x9_summary.json
import sys, re, json, time, random, importlib.util, importlib.machinery
from pathlib import Path
from contextlib import redirect_stdout
from io import StringIO

BASE = Path(__file__).resolve().parent.parent
VENDOR = BASE / "proto" / "vendor" / "bs3_snapshot.txt"
PKG = BASE / "sandbox" / "pkgdata"
SB = BASE / "sandbox"
DATA = BASE / "data"
META = json.loads((DATA / "meta_layer.json").read_text(encoding="utf-8"))

def load_mod(name, path):
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_file_location(name, str(path), loader=loader)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod

evocore = load_mod("evocore", BASE / "proto" / "evocore.txt")
m = load_mod("bs3_vendor", VENDOR)
m.set_data_dir(str(PKG))
TOL, GENS, KIDS = 0.005, META["gens"], META["kids"]
PROSPECT = list(range(9, 25))                         # 预注册前瞻组

def build_lib(seed):
    lib = SB / f"lib-x9-ng-s{seed}"
    if not (lib / "state.json").exists():
        m.cmd_install("L2", [], lib, True)
        entries, _ = evocore.build_world(seed)
        tail = sorted((lib / "audit").glob("lifelog-*.md"))[-1].read_text(encoding="utf-8").strip().splitlines()[-1]
        seid = "M-" + re.search(r"h:([0-9a-f]{64})", tail).group(1)[:12]
        for e in entries:
            m.cmd_engine("append", str(lib), json.dumps(
                {"id": e["id"], "created_at": e["created_at"], "updated_at": "",
                 "content": e["content"], "keywords": e["keywords"], "links": e["links"],
                 "source_event_id": seid, "importance": e["importance"],
                 "confidence": "中", "validity": {"t_valid": e["created_at"]}}, ensure_ascii=False))
    return lib

def evolve_ng(seed):
    """与 x8 NG 臂逐位同构（同 rng 键、同验收、无链接）。"""
    lib = build_lib(seed)
    entries, kw_of = evocore.build_world(seed)
    train, held, audit = evocore.build_tasks(seed, entries)
    rng = random.Random(seed * 31 + sum(map(ord, "PL")))
    mainline = dict(evocore.DEFAULT_CFG)
    archive = [dict(mainline)]
    cache = {}
    def sc(cfg):
        key = tuple(sorted(cfg.items()))
        if key not in cache:
            cache[key] = evocore.score_lib(lib, train, cfg, kw_of=kw_of)
        return cache[key]
    for gen in range(1, GENS + 1):
        for _ in range(KIDS):
            pool = sorted(archive, key=lambda c: -sc(c))
            parent = rng.choice(pool[:3]) if (rng.random() < 0.7 and len(pool) >= 3) else rng.choice(pool)
            child = evocore.mutate(parent, rng)
            if sc(child) - sc(parent) > -TOL:
                archive.append(dict(child))
                if len(archive) > 30:
                    archive.pop(min(range(len(archive)), key=lambda i: sc(archive[i])))
            if sc(child) > sc(mainline):
                mainline = dict(child)
    res = {"seed": seed, "final_mainline": mainline,
           "audit_recall5": round(evocore.recall_at_k(lib, audit, mainline, kw_of=kw_of), 4)}
    buf = StringIO()
    with redirect_stdout(buf):
        m.cmd_engine("doctor", str(lib))
    res["doctor_ok"] = "全一致" in buf.getvalue()
    (DATA / f"x9-ng-s{seed}.json").write_text(json.dumps(res, ensure_ascii=False, indent=1), encoding="utf-8")
    return res

t0 = time.time()
print("== ① 前瞻组演化 seeds 9–24 ==", flush=True)
for seed in PROSPECT:
    r = evolve_ng(seed)
    print(f"[s{seed}] task={r['audit_recall5']:.2f} {r['final_mainline']} doctor={'OK' if r['doctor_ok'] else 'BAD'}", flush=True)

print("== ② 碰撞面普查 ==", flush=True)
census = {"kw_family": {}, "per_seed": {}}
es1, _ = evocore.build_world(1)
kws = sorted({e["keywords"][0] for e in es1})
for k in kws:
    fam = [x for x in kws if x != k and k in x]
    census["kw_family"][k] = {"children": fam, "collision_entries": (len(fam) + 1) * 4 if fam else 4}
for seed in range(1, 25):
    es, _ = evocore.build_world(seed)
    tr, hd, au = evocore.build_tasks(seed, es)
    k1 = sum(1 for t in au if t["kw"] == "K1")
    t1 = sum(1 for t in au if t["q"].split()[0] == "T1")
    t1_content = sum(1 for e in es if "T1" in e["content"])
    census["per_seed"][str(seed)] = {"audit_K1": k1, "audit_T1_topic": t1, "T1_content_entries": t1_content}
print(f"K1 kw碰撞面={census['kw_family']['K1']['collision_entries']}（组合论 11×4=44） T1内容碰撞条目(seed1)={census['per_seed']['1']['T1_content_entries']}", flush=True)
(DATA / "x9_census.json").write_text(json.dumps(census, ensure_ascii=False, indent=1), encoding="utf-8")

print("== ③ 预报器（L1∧L2，规则冻结于预注册） ==", flush=True)
def L1(seed):
    c = census["per_seed"][str(seed)]
    return (c["audit_K1"] + c["audit_T1_topic"]) > 0
def L2(cfg):
    return cfg["w_kw"] > 6 or cfg["w_content"] > 3 or cfg["w_imp"] > 0.05 or cfg["w_age"] > 0.02

runs = []
for src, tag in ((("x8-NG", "x8-TG"), "回溯x8"), (("x9-ng",), "前瞻x9")):
    for pref in src:
        for p in sorted(DATA.glob(f"{pref}-s*.json")):
            r = json.loads(p.read_text(encoding="utf-8"))
            seed = r["seed"]
            if tag == "前瞻x9" and seed not in PROSPECT:
                continue
            runs.append({"tag": tag, "pref": pref, "seed": seed, "cfg": r["final_mainline"],
                         "actual": r["audit_recall5"] < 0.95, "recall": r["audit_recall5"]})
conf = {"回溯x8": {"TP": 0, "FP": 0, "FN": 0, "TN": 0}, "前瞻x9": {"TP": 0, "FP": 0, "FN": 0, "TN": 0}}
for r in runs:
    pred = L1(r["seed"]) and L2(r["cfg"])
    c = conf[r["tag"]]
    if pred and r["actual"]: c["TP"] += 1
    elif pred and not r["actual"]: c["FP"] += 1
    elif not pred and r["actual"]: c["FN"] += 1
    else: c["TN"] += 1

for tag, c in conf.items():
    print(f"{tag}: TP={c['TP']} FP={c['FP']} FN={c['FN']} TN={c['TN']}")
    for r in runs:
        if r["tag"] == tag and (r["actual"] or (L1(r["seed"]) and L2(r["cfg"]))):
            print(f"   {'✓' if (L1(r['seed']) and L2(r['cfg']))==r['actual'] else '✗'} {r['pref']}-s{r['seed']}: "
                  f"实际受伤={r['actual']}({r['recall']:.2f}) 预报={L1(r['seed']) and L2(r['cfg'])} "
                  f"L1={L1(r['seed'])} L2={L2(r['cfg'])} {r['cfg']}")

# 剂量关系（描述性）：L2 阳性 run 中 audit 毒任务数 vs 损失量
l2p = [r for r in runs if L2(r["cfg"])]
if len(l2p) >= 2:
    xs = [census["per_seed"][str(r["seed"])]["audit_K1"] + census["per_seed"][str(r["seed"])]["audit_T1_topic"] for r in l2p]
    ys = [1 - r["recall"] for r in l2p]
    mx, my = sum(xs)/len(xs), sum(ys)/len(ys)
    cov = sum((x-mx)*(y-my) for x, y in zip(xs, ys))
    sx = (sum((x-mx)**2 for x in xs) * sum((y-my)**2 for y in ys)) ** 0.5
    dose = cov/sx if sx else None
    print(f"剂量相关（L2阳性 n={len(l2p)}，毒任务数 vs 损失量）: r={dose:.3f}" if dose is not None else "剂量相关: 无法计算")

# 预注册判定
ret, pro = conf["回溯x8"], conf["前瞻x9"]
if ret == {"TP": 2, "FP": 0, "FN": 0, "TN": 14} and pro["FP"] == 0 and pro["FN"] == 0:
    verdict = "SUPPORTED（碰撞面×切分定位预报器在留出种子上全对）"
elif pro["FP"] + pro["FN"] <= 1 and ret["FP"] == 0 and ret["FN"] == 0:
    verdict = "PARTIAL（预报器不彻底，见混淆矩阵）"
else:
    verdict = "NOT SUPPORTED（规则与实际不符，见混淆矩阵）"
census_ok = census["kw_family"]["K1"]["collision_entries"] == 44
print(f"普查对账（K1=44）: {census_ok}")
print(f"判定：{verdict}")
summary = {"census_K1_44": census_ok, "confusion": conf, "判定": verdict, "前瞻组受伤实际":
           [r["seed"] for r in runs if r["tag"] == "前瞻x9" and r["actual"]]}
(DATA / "x9_summary.json").write_text(json.dumps({"summary": summary, "runs": runs},
                                                 ensure_ascii=False, indent=1), encoding="utf-8")
print(f"X9 完成 {time.time()-t0:.1f}s → data/x9_summary.json")
