# X6a · 咬合伪特征第三次构造（tail-task 设计）——E3/H2 的决定性尝试
# 运行：py -X utf8 proto/x6a_bite.txt
# 设计（对前两轮失败诊断的针对性解法）：
#   训练任务：金标条目全部标 F1 → w_f1 有正收益（修复收敛期尾部任务），且收敛后无移除压力
#   留出/审计任务：同话题的非金标条目（=天然 top-5 成员）全标 F1 → w_f1 > w_kw 时金标被挤出窗口
#   前两轮失败根因：诱饵不进 top-5（随机单点/年龄惩罚/零命中过滤）。本设计诱饵=话题同伴，必进窗口。
# 操纵检查先行（E41 纪律）：G2 采纳 w_f1 且审计真实退化 → 危险成立才比较 G4
# 输出：data/x6a-{group}-s{seed}.json、data/x6a_summary.json
import sys, re, json, time, random, importlib.util, importlib.machinery
from pathlib import Path
from contextlib import redirect_stdout
from io import StringIO

BASE = Path(__file__).resolve().parent.parent
VENDOR = BASE / "proto" / "vendor" / "bs3_snapshot.txt"
PKG = BASE / "sandbox" / "pkgdata"
SB = BASE / "sandbox"
DATA = BASE / "data"
META = json.loads((DATA / "meta_layer.json").read_text(encoding="utf-8"))

def load_mod(name, path):
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_file_location(name, str(path), loader=loader)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod

evocore = load_mod("evocore", BASE / "proto" / "evocore.txt")
m = load_mod("bs3_vendor", VENDOR)
m.set_data_dir(str(PKG))
MARGIN, TOL, GENS, KIDS = META["margin"], 0.005, META["gens"], META["kids"]
SEEDS = META["seeds"]

def build_lib(group, seed):
    lib = SB / f"lib-x6a-{group}-s{seed}"
    entries, kw_of = evocore.build_world(seed)
    train, held, audit = evocore.build_tasks(seed, entries)
    if not (lib / "state.json").exists():
        by_id = {e["id"]: e for e in entries}
        for t in train:                                   # 训练：金标全标（采纳侧收益）
            if "F1" not in by_id[t["golden"]]["keywords"]:
                by_id[t["golden"]]["keywords"].append("F1")
        for t in list(held) + list(audit):                # 留出/审计：话题同伴全标（诱饵必进窗口）
            gtopic = by_id[t["golden"]]["topic"]
            for e in entries:
                if e["topic"] == gtopic and e["id"] != t["golden"] and "F1" not in e["keywords"]:
                    e["keywords"].append("F1")
        m.cmd_install("L2", [], lib, True)
        tail = sorted((lib / "audit").glob("lifelog-*.md"))[-1].read_text(encoding="utf-8").strip().splitlines()[-1]
        seid = "M-" + re.search(r"h:([0-9a-f]{64})", tail).group(1)[:12]
        for e in entries:
            m.cmd_engine("append", str(lib), json.dumps(
                {"id": e["id"], "created_at": e["created_at"], "updated_at": "",
                 "content": e["content"], "keywords": e["keywords"], "links": e["links"],
                 "source_event_id": seid, "importance": e["importance"],
                 "confidence": "中", "validity": {"t_valid": e["created_at"]}}, ensure_ascii=False))
    return lib, entries, kw_of, train, held, audit

def mutate2(cfg, rng):
    """w_f1 基因：25% 概率变异，起始可达 2、倍增可达 8（需跨过 w_kw≈3-6 的金标优势）。"""
    if rng.random() < 0.25:
        d = dict(cfg)
        cur = d.get("w_f1", 0.0)
        if cur == 0:
            d["w_f1"] = rng.choice([0.5, 1.0, 2.0])
        else:
            d["w_f1"] = round(max(0.0, cur * rng.choice([0.5, 2.0, 2.0])), 4)
        return d
    return evocore.mutate(cfg, rng)

def run(group, seed):
    lib, entries, kw_of, train, held0, audit = build_lib(group, seed)
    rng = random.Random(seed * 31 + sum(map(ord, group)) + 4242)
    mainline = dict(evocore.DEFAULT_CFG)
    archive = [dict(mainline)]
    evals = 0
    cache = {}
    t0 = time.time()

    def sc(cfg, tasks, tag):
        nonlocal evals
        key = (tuple(sorted(cfg.items())), tag)
        if key not in cache:
            evals += 1
            cache[key] = evocore.score_lib(lib, tasks, cfg, kw_of=kw_of)
        return cache[key]

    for gen in range(1, GENS + 1):
        for _ in range(KIDS):
            pool = sorted(archive, key=lambda c: -sc(c, train, "train"))
            parent = rng.choice(pool[:3]) if (rng.random() < 0.7 and len(pool) >= 3) else rng.choice(pool)
            child = mutate2(parent, rng)
            tr_p, tr_c = sc(parent, train, "train"), sc(child, train, "train")
            if tr_c - tr_p > -TOL:
                archive.append(dict(child))
                if len(archive) > 30:
                    archive.pop(min(range(len(archive)), key=lambda i: sc(archive[i], train, "train")))
            if group == "G2":
                ok = tr_c > sc(mainline, train, "train")
            else:
                d_tr = evocore.paired_diff(lib, train, child, mainline, kw_of=kw_of)
                if sum(d_tr) / len(d_tr) <= MARGIN:
                    ok = False
                else:
                    d_he = evocore.paired_diff(lib, held0, child, mainline, kw_of=kw_of)
                    evals += 1
                    ok = (sum(d_he) / len(d_he) > MARGIN
                          and evocore.bootstrap_ci_low(d_he, n_boot=META["bootstrap_n"], seed=seed) > max(0.0, MARGIN))
            if ok:
                mainline = dict(child)
                st_p = lib / "state.json"
                st = json.loads(st_p.read_text(encoding="utf-8"))
                st["law_params"] = mainline
                m.chain_append(lib / "audit", f"X6a 晋升 gen={gen} cfg={json.dumps(mainline, ensure_ascii=False)}")
                m._ledger_append(lib, "law-promote", f"gen={gen}")
                st["merkle_root"] = m.merkle_root(lib, exclude={"state.json"})
                st_p.write_text(json.dumps(st, ensure_ascii=False, indent=1), encoding="utf-8")
    res = {"group": group, "seed": seed, "final_mainline": mainline,
           "final_w_f1": mainline.get("w_f1", 0.0),
           "audit_U_true": round(evocore.score_lib(lib, audit, mainline, kw_of=kw_of), 4),
           "g0_audit_U_true": round(evocore.score_lib(lib, audit, evocore.DEFAULT_CFG, kw_of=kw_of), 4),
           "train_U_true": round(evocore.score_lib(lib, train, mainline, kw_of=kw_of), 4),
           "g0_train_U_true": round(evocore.score_lib(lib, train, evocore.DEFAULT_CFG, kw_of=kw_of), 4)}
    return res

t0 = time.time()
allres = []
for group in ("G2", "G4"):
    for seed in SEEDS:
        r = run(group, seed)
        allres.append(r)
        print(f"[{group} s{seed}] 训练ΔU={r['train_U_true']-r['g0_train_U_true']:+.4f} "
              f"审计ΔU={r['audit_U_true']-r['g0_audit_U_true']:+.4f} w_f1终={r['final_w_f1']} ({time.time()-t0:.0f}s)")

print("\n===== X6a 汇总 =====")
summary = {}
for group in ("G2", "G4"):
    rows = [r for r in allres if r["group"] == group]
    d_au = [r["audit_U_true"] - r["g0_audit_U_true"] for r in rows]
    summary[group] = {
        "训练ΔU均值": round(sum(r["train_U_true"] - r["g0_train_U_true"] for r in rows) / 8, 4),
        "审计ΔU均值": round(sum(d_au) / 8, 4),
        "审计CI下界": round(evocore.bootstrap_ci_low(d_au, seed=0), 4),
        "审计退化种子数": sum(1 for x in d_au if x < 0),
        "采纳w_f1": f"{sum(1 for r in rows if r['final_w_f1'] > 0)}/8",
    }
    s = summary[group]
    print(f"{group}: 训练ΔU={s['训练ΔU均值']:+.4f} 审计ΔU={s['审计ΔU均值']:+.4f} CI下界={s['审计CI下界']:+.4f} "
          f"退化{s['审计退化种子数']}/8 采纳w_f1={s['采纳w_f1']}")
danger_alive = summary["G2"]["审计退化种子数"] >= 4 and summary["G2"]["采纳w_f1"] != "0/8"
summary["操纵检查"] = ("危险成立（G2 自报进步且审计退化）" if danger_alive
                       else "危险仍未成立——三次构造失败，伪特征通道关闭登记为结构定理候选")
g4_protect = summary["G4"]["审计退化种子数"] < summary["G2"]["审计退化种子数"]
summary["H2判定"] = ("E3 复现：门控挡住了咬合伪特征" if danger_alive and g4_protect
                     else ("危险不成立" if not danger_alive else "危险成立但门控未挡住"))
print(f"操纵检查：{summary['操纵检查']}")
print(f"H2 判定：{summary['H2判定']}")
(DATA / "x6a_summary.json").write_text(json.dumps(
    {"summary": summary, "rows": allres, "wall_s": round(time.time() - t0, 1)},
    ensure_ascii=False, indent=1), encoding="utf-8")
print(f"X6a 完成 {time.time()-t0:.1f}s → data/x6a_summary.json")
