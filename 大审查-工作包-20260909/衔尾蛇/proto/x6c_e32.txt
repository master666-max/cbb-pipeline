# X6c · E32 复测：自称 importance vs 实测采纳重要度（真实库 A/B）
# 运行：py -X utf8 proto/x6c_e32.txt
# 方法（报告 E32 口径的真库版）：同一库复制两份——A 库条目保持「自称 importance」，
#   B 库把 importance 字段改写为「采纳日志派生的实测重要度」（衰减累计，仅由检索命中金标驱动），
#   其余完全一致 → 对比两库在 留出/审计 任务上的 recall@5 / MRR。
# 输出：data/x6c_e32_summary.json
import sys, re, json, time, random, shutil, importlib.util, importlib.machinery
from pathlib import Path
from contextlib import redirect_stdout
from io import StringIO

BASE = Path(__file__).resolve().parent.parent
VENDOR = BASE / "proto" / "vendor" / "bs3_snapshot.txt"
PKG = BASE / "sandbox" / "pkgdata"
SB = BASE / "sandbox"
DATA = BASE / "data"
META = json.loads((DATA / "meta_layer.json").read_text(encoding="utf-8"))

def load_mod(name, path):
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_file_location(name, str(path), loader=loader)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod

evocore = load_mod("evocore", BASE / "proto" / "evocore.txt")
m = load_mod("bs3_vendor", VENDOR)
m.set_data_dir(str(PKG))
SEEDS = [1, 2, 3, 4, 5]
LIB_A = SB / "lib-real"                       # 已有：真实迁移库（自称 importance）
LIB_B = SB / "lib-real-meas"                  # 实测重要度副本

entries = {f.stem: json.loads(f.read_text(encoding="utf-8"))
           for f in (LIB_A / "memory").rglob("*.json")}
tasks_src = []
for eid, e in entries.items():
    if (e.get("validity") or {}).get("t_invalid"):
        continue
    kws = [k for k in (e.get("keywords") or []) if k and k != "轨迹"]
    if kws:
        tasks_src.append({"q": " ".join(kws[:2]), "golden": eid})
print(f"真实库 {len(entries)} 条，可出任务 {len(tasks_src)} 条")

def split_for(seed):
    rnd = random.Random(seed * 613 + 11)
    idx = list(range(len(tasks_src)))
    rnd.shuffle(idx)
    sel = [tasks_src[i] for i in idx]
    n_train = max(1, int(len(sel) * 0.6))
    n_audit = max(2, int(len(sel) * 0.15))
    return sel[:n_train], sel[n_train:len(sel) - n_audit], sel[len(sel) - n_audit:]

def top5(lib, q, cfg):
    return [fid for fid, _ in evocore.rank_lib(lib, q, cfg)[:5]]

def evalset(lib, tasks, cfg):
    hits = rr = 0.0
    for t in tasks:
        r5 = top5(lib, t["q"], cfg)
        if t["golden"] in r5:
            hits += 1
            rr += 1.0 / (r5.index(t["golden"]) + 1)
    n = max(1, len(tasks))
    return {"recall5": round(hits / n, 4), "mrr": round(rr / n, 4)}

# ── 实测重要度：按种子模拟时序采纳流（衰减 0.9）────────────────────
if LIB_B.exists():
    shutil.rmtree(LIB_B)
shutil.copytree(LIB_A, LIB_B)
measured = {eid: 0.0 for eid in entries}
for seed in SEEDS:
    train, _, _ = split_for(seed)
    rnd = random.Random(seed)
    order = train[:]
    rnd.shuffle(order)
    step = 0
    for t in order:
        step += 1
        top = top5(LIB_A, t["q"], evocore.DEFAULT_CFG)
        decay = 0.9 ** (len(order) - step)
        if t["golden"] in top:
            measured[t["golden"]] += decay                    # 金标被采纳 → +衰减权重
        for f in top:
            if f != t["golden"]:
                measured[f] -= 0.3 * decay                    # 被展示却非金标 → 扣分
# 归一到 1-10 写入 B 库 importance 字段（其余字段零改动）
vals = [v for v in measured.values() if v > 0]
lo, hi = (min(vals), max(vals)) if vals else (0, 1)
n_rewrite = 0
for eid, v in measured.items():
    scaled = round(1 + 9 * (v - lo) / (hi - lo), 3) if hi > lo else 5.0
    hits = list((LIB_B / "memory").rglob(f"{eid}.json"))
    for h in hits:
        e = json.loads(h.read_text(encoding="utf-8"))
        e["importance"] = max(1, scaled)
        h.write_text(json.dumps(e, ensure_ascii=False, indent=1), encoding="utf-8")
        n_rewrite += 1
print(f"实测重要度已写入 B 库（改写 {n_rewrite} 条；原始自称值保留于 A 库）")

# ── A/B 对比（默认参数 + 5 种子的留出/审计）────────────────────────
rows = []
for seed in SEEDS:
    _, held, audit = split_for(seed)
    for label, lib in (("自称A", LIB_A), ("实测B", LIB_B)):
        for tname, tasks in (("held", held), ("audit", audit)):
            r = evalset(lib, tasks, evocore.DEFAULT_CFG)
            rows.append({"seed": seed, "set": tname, "lib": label, **r})

def agg(lib, tset):
    sel = [r for r in rows if r["lib"] == lib and r["set"] == tset]
    return {"recall5": round(sum(r["recall5"] for r in sel) / len(sel), 4),
            "mrr": round(sum(r["mrr"] for r in sel) / len(sel), 4)}

summary = {}
for tset in ("held", "audit"):
    a, b = agg("自称A", tset), agg("实测B", tset)
    summary[tset] = {"自称A": a, "实测B": b,
                     "mrr增益": round(b["mrr"] - a["mrr"], 4),
                     "recall5增益": round(b["recall5"] - a["recall5"], 4)}
    print(f"{tset}: 自称A={a}  实测B={b}  ΔMRR={summary[tset]['mrr增益']:+.4f}")
dmrr = sum(summary[t]["mrr增益"] for t in ("held", "audit")) / 2
summary["判定"] = ("实测重要度优于自称（E32 真库复现）" if dmrr > 0 else
                   "实测未优于自称（E32 真库未复现——采纳信号太稀疏或自称已够好）")
print(f"判定：{summary['判定']}")
(DATA / "x6c_e32_summary.json").write_text(json.dumps(
    {"summary": summary, "rows": rows, "measured_top8": sorted(measured.items(), key=lambda x: -x[1])[:8]},
    ensure_ascii=False, indent=1), encoding="utf-8")
