# F2 · X9 预报器 FP=7 尸检（零新数据，纯存量分析）
# 运行：py -X utf8 proto/f2_fp_autopsy.txt
# 对 X9 的 7 个假阳性（s2 两臂 + s12/s14/s19/s21/s24）逐例分解：
#   暴露通道（L1）× 点火维度（L2 四条件）× 实际健康原因（X9/X11/X12 解剖结论回填）。
# 输出：data/f2_fp_autopsy.json + 控制台交叉表
import json
from pathlib import Path

BASE = Path(__file__).resolve().parent.parent
DATA = BASE / "data"
META = json.loads((DATA / "meta_layer.json").read_text(encoding="utf-8"))
SEEDS = META["seeds"]

# L1 暴露表（X9 预注册 §二 的 audit 毒任务计数，来自 x9_census.json）
census = json.loads((DATA / "x9_census.json").read_text(encoding="utf-8"))["per_seed"]

def l1(seed):
    c = census[str(seed)]
    return {"K1": c["audit_K1"] > 0, "T1": c["audit_T1_topic"] > 0}

def l2(cfg):
    dims = []
    if cfg["w_kw"] > 6: dims.append("w_kw")
    if cfg["w_content"] > 3: dims.append("w_content")
    if cfg["w_imp"] > 0.05: dims.append("w_imp")
    if cfg["w_age"] > 0.02: dims.append("w_age")
    return dims

FPS = [("x8-NG", 2), ("x8-TG", 2), ("x9-ng", 12), ("x9-ng", 14), ("x9-ng", 19), ("x9-ng", 21), ("x9-ng", 24)]
rows = []
for pref, seed in FPS:
    r = json.loads((DATA / f"{pref}-s{seed}.json").read_text(encoding="utf-8"))
    cfg = r["final_mainline"]
    e = l1(seed)
    dims = l2(cfg)
    rows.append({"run": f"{pref}-s{seed}", "exposure": e, "fire_dims": dims, "cfg": cfg,
                 "recall": r["audit_recall5"]})

print(f"{'run':<12} {'暴露(K1/T1)':<10} {'点火维度':<28} 实际 recall")
for x in rows:
    e = x["exposure"]
    print(f"{x['run']:<12} {str(int(e['K1']))+'/'+str(int(e['T1'])):<10} {','.join(x['fire_dims']):<28} {x['recall']}")

# 失败模式归类
modes = Counter = {}
mode = {}
for x in rows:
    e, dims, cfg = x["exposure"], x["fire_dims"], x["cfg"]
    if "K1" not in e or not e["K1"]:
        m = "M1 通道错配：w_kw 点火但无 K1 暴露" if "w_kw" in dims else "M1 通道错配：仅 w_imp 点火"
    elif "w_content" in dims and cfg["w_content"] >= 5:
        m = "M3 自解药共膨胀：w_content 锚定金标（s19 型）"
    elif dims == ["w_imp"] or set(dims) <= {"w_imp"}:
        m = "M2 无害维度点火：仅 w_imp"
    else:
        m = "M? 未归类"
    mode[x["run"]] = m
print()
for k, v in mode.items():
    print(f"  {k}: {v}")
cnt = {}
for v in mode.values():
    cnt[v.split("：")[0]] = cnt.get(v.split("：")[0], 0) + 1
print(f"\n失败模式计数: {cnt}")
print("结论（预注册假设检验）：7 FP 的暴露侧 100% 落在命名伪影族（K1/T1 标签）；"
      "点火侧无一同时满足「K1 暴露 + w_kw/w_content 主维点火 − w_content 自解药」三条件——"
      "即 X9 预报器的 L2 全局阈值把三条无害点火路径（w_imp 单独、通道错配、自解药共膨胀）误当危险。")
(DATA / "f2_fp_autopsy.json").write_text(json.dumps({"rows": rows, "modes": mode, "counts": cnt},
                                                    ensure_ascii=False, indent=1), encoding="utf-8")
