# X2b · 防作弊（贪婪变异不可表达，报告 E2）+ X2c-v2 · 灾难性遗忘（目标切换，报告 E16）
# 运行：py -X utf8 proto/x2bc.txt
#
# X2b 设计（v2 修正）：v1 曾把 w_imp=100 与合法修复（w_age=0/filter_zero=1）捆在一起，
#   得出"贪婪过门"的假阳性——增益来自合法部分。v2 分离：
#   ①纯贪婪（仅堆 w_imp）从默认出发 → 应进不了任何门；
#   ②合法改良配置下扫 w_imp ∈ {0,0.5,2,10,100} → 应单调不升（堆料非优化方向）。
#
# X2c 设计（v2 修正）：v1 的 B 目标"省无关"被诊断出结构性平坦（话题=5 条目/主题，
#   top-5 无关数恒 ≈4，参数不可动）→ v1 构造不成立，如实登记为否定性结果。
#   v2 给法层加 k_return（返回窗口 1..5，真实系统的检索 cap 基因，E72 同族）：
#   目标 B = A − 0.08×k_return（省返回条目=省注意力/ token 成本）。
#   遗忘通道：A-optimal 下 golden 排 ~1-5 名，B 把 k 压到 1 会砍掉"golden 非 #1"的尾部任务。
#
# 输出：data/x2bc_results_{TAG}.json
import sys, re, json, time, random, importlib.util, importlib.machinery
from pathlib import Path

BASE = Path(__file__).resolve().parent.parent
VENDOR = BASE / "proto" / "vendor" / "bs3_snapshot.txt"
PKG = BASE / "sandbox" / "pkgdata"
SB = BASE / "sandbox"
DATA = BASE / "data"
META = json.loads((DATA / "meta_layer.json").read_text(encoding="utf-8"))

def load_mod(name, path):
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_file_location(name, str(path), loader=loader)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod

evocore = load_mod("evocore", BASE / "proto" / "evocore.txt")
m = load_mod("bs3_vendor", VENDOR)
m.set_data_dir(str(PKG))
MARGIN = META["margin"]
GENS, KIDS = 20, 3
SEEDS = META["seeds"]
COEF = float(sys.argv[1]) if len(sys.argv) > 1 else 0.08   # B 目标的 k_return 成本系数（敏感性检查用）
TAG = f"coef{COEF}"

def build_lib(prefix, seed):
    lib = SB / f"{prefix}-s{seed}"
    if not (lib / "state.json").exists():
        m.cmd_install("L2", [], lib, True)
        entries, _ = evocore.build_world(seed)
        tail = sorted((lib / "audit").glob("lifelog-*.md"))[-1].read_text(encoding="utf-8").strip().splitlines()[-1]
        seid = "M-" + re.search(r"h:([0-9a-f]{64})", tail).group(1)[:12]
        for e in entries:
            m.cmd_engine("append", str(lib), json.dumps(
                {"id": e["id"], "created_at": e["created_at"], "updated_at": "",
                 "content": e["content"], "keywords": e["keywords"], "links": e["links"],
                 "source_event_id": seid, "importance": e["importance"],
                 "confidence": "中", "validity": {"t_valid": e["created_at"]}}, ensure_ascii=False))
    return lib

# ══ X2b · 防作弊 ═══════════════════════════════════════════════════
print("== X2b 防作弊（报告 E2）==")
entries, kw_of = evocore.build_world(1)
train, held, audit = evocore.build_tasks(1, entries)
u_def = evocore.score_lib(LIB := (SB / "lib-x0"), train, evocore.DEFAULT_CFG, kw_of=kw_of)

pure_greedy = [
    {"w_kw": 3.0, "w_content": 1.0, "w_imp": 10.0, "w_age": 0.1, "filter_zero": 0},
    {"w_kw": 3.0, "w_content": 1.0, "w_imp": 100.0, "w_age": 0.1, "filter_zero": 0},
    {"w_kw": 0.0, "w_content": 0.0, "w_imp": 10.0, "w_age": 0.0, "filter_zero": 0},
]
rows_greedy = []
for c in pure_greedy:
    tr = evocore.score_lib(LIB, train, c, kw_of=kw_of)
    rows_greedy.append({"cfg": c, "train": round(tr, 4), "vs_default": round(tr - u_def, 4),
                        "pass_G2": bool(tr - u_def > MARGIN)})
    print(f"  纯贪婪 w_imp={c['w_imp']:<5} train={tr:+.4f}（vs 默认 {tr-u_def:+.4f}）过G2门={tr-u_def>MARGIN}")

base_good = {"w_kw": 3.0, "w_content": 1.0, "w_imp": 0.5, "w_age": 0.0, "filter_zero": 1}
rows_scan = []
print(f"  合法改良底座 {base_good} 上扫 w_imp：")
for w in (0.0, 0.5, 2.0, 10.0, 100.0):
    c = dict(base_good); c["w_imp"] = w
    tr = evocore.score_lib(LIB, train, c, kw_of=kw_of)
    rows_scan.append({"w_imp": w, "train": round(tr, 4)})
    print(f"    w_imp={w:<6} train={tr:+.4f}")
mono = all(rows_scan[i]["train"] >= rows_scan[i + 1]["train"] - 1e-9 for i in range(len(rows_scan) - 1))
x2b_verdict = (not any(r["pass_G2"] for r in rows_greedy)) and mono
print(f"  → 判定：纯贪婪不可过门 ✓；w_imp 扫描{'单调不升 ✓' if mono else '存在上升 ✗'}"
      f" → {'作弊结构性不可表达（堆料非优化方向）' if x2b_verdict else '需人工审查'}")

# ══ X2c-v2 · 灾难性遗忘（k_return 成本轴）══════════════════════════
print("\n== X2c-v2 灾难性遗忘（A→B 切换，B=A−0.08×k_return；单一 vs Pareto）==")

def mutate2(cfg, rnd):
    if rnd.random() < 0.30:
        d = dict(cfg)
        d["k_return"] = max(1, min(5, d.get("k_return", 5) + rnd.choice([-1, 1])))
        return d
    return evocore.mutate(cfg, rnd)

def UA_task(lib, t, cfg, kw_of):
    k = cfg.get("k_return", 5)
    top = [fid for fid, _ in evocore.rank_lib(lib, t["q"], cfg)[:k]]
    hit = 1.0 if t["golden"] in top else 0.0
    n_irrel = sum(1 for f in top if kw_of.get(f) != t["kw"])
    return hit - evocore.LAMBDA * n_irrel

def sc_A(lib, tasks, cfg, kw_of):
    return sum(UA_task(lib, t, cfg, kw_of) for t in tasks) / len(tasks)

def sc_B(lib, tasks, cfg, kw_of):
    return sc_A(lib, tasks, cfg, kw_of) - 0.08 * cfg.get("k_return", 5)

def UB_task(lib, t, cfg, kw_of):
    return UA_task(lib, t, cfg, kw_of) - COEF * cfg.get("k_return", 5)

def paired_obj(lib, tasks, a, b, obj, kw_of):
    f = UA_task if obj == "A" else UB_task
    return [f(lib, t, a, kw_of) - f(lib, t, b, kw_of) for t in tasks]

def nondominated(pairs):
    keep = []
    for i, (a, b, c) in enumerate(pairs):
        if not any((a2 >= a and b2 >= b) and (a2 > a or b2 > b)
                   for j, (a2, b2, c2) in enumerate(pairs) if j != i):
            keep.append((a, b, c))
    return keep

def run_x2c(seed, policy):
    lib = build_lib(f"lib-x2c2-{policy}-c{COEF}", seed)
    entries, kw_of = evocore.build_world(seed)
    train, held, audit = evocore.build_tasks(seed, entries)
    rng = random.Random(seed * 977 + (1 if policy == "single" else 2))
    cache = {}
    def sc(cfg, obj):
        key = (tuple(sorted(cfg.items())), obj)
        if key not in cache:
            f = sc_A if obj == "A" else sc_B
            cache[key] = f(lib, train, cfg, kw_of)
        return cache[key]
    mainline = dict(evocore.DEFAULT_CFG)
    pairs = [(sc(mainline, "A"), sc(mainline, "B"), dict(mainline))]
    snap = None
    b_promos = 0
    for obj in ("A", "B"):
        for gen in range(1, GENS + 1):
            for _ in range(KIDS):
                if policy == "single":
                    parent = mainline
                else:
                    pool = sorted(pairs, key=lambda x: -(x[0] if obj == "A" else x[1]))
                    parent = (rng.choice(pool[:3]) if (rng.random() < 0.7 and len(pool) >= 3) else rng.choice(pool))[2]
                child = mutate2(parent, rng)
                d = paired_obj(lib, train, child, parent, obj, kw_of)
                if sum(d) / len(d) > MARGIN:
                    mainline = dict(child)
                    if obj == "B":
                        b_promos += 1
                    if policy == "single":
                        pairs = [(sc(mainline, "A"), sc(mainline, "B"), dict(mainline))]
                if policy == "pareto":
                    pairs.append((sc(child, "A"), sc(child, "B"), dict(child)))
                    pairs = nondominated(pairs)[:30]
        if obj == "A":
            snap = {"mainline_UA": round(sc(mainline, "A"), 4),
                    "bestUA_archive": round(max(p[0] for p in pairs), 4),
                    "mainline_k": mainline.get("k_return", 5)}
    A_end = (max(p[0] for p in pairs) if policy == "pareto" else sc(mainline, "A"))
    k_end = (max(pairs, key=lambda p: p[1])[2] if policy == "pareto" else mainline).get("k_return", 5)
    denom = snap["bestUA_archive"] if policy == "pareto" else snap["mainline_UA"]
    return {"seed": seed, "policy": policy, "snapshot": snap,
            "A_end": round(A_end, 4), "k_end": k_end,
            "b_promos": b_promos, "A_retention": round(A_end / denom, 4) if denom else None}

t0 = time.time()
x2c_rows = []
for policy in ("single", "pareto"):
    for seed in SEEDS:
        r = run_x2c(seed, policy)
        x2c_rows.append(r)
        print(f"  [{policy} s{seed}] 切换时 A={r['snapshot']['mainline_UA']:+.4f}(k={r['snapshot']['mainline_k']})"
              f"  终态A={r['A_end']:+.4f}(k={r['k_end']})  B晋升{r['b_promos']}次  保留率={r['A_retention']}")
ret_s = [r["A_retention"] for r in x2c_rows if r["policy"] == "single"]
ret_p = [r["A_retention"] for r in x2c_rows if r["policy"] == "pareto"]
print(f"\n  A 保留率：单一最优 均值 {sum(ret_s)/8:.3f} | Pareto 均值 {sum(ret_p)/8:.3f}"
      f"  （报告 E16 方向：Pareto ≥ 单一）")
(DATA / f"x2bc_results_{TAG}.json").write_text(json.dumps(
    {"x2b": {"pure_greedy": rows_greedy, "w_imp_scan": rows_scan,
             "verdict_作弊不可表达": x2b_verdict},
     "x2c": {"version": "v2-k_return", "rows": x2c_rows,
             "retention_single_mean": round(sum(ret_s) / 8, 4),
             "retention_pareto_mean": round(sum(ret_p) / 8, 4),
             "v1_否定性登记": "B=省无关 目标被诊断结构性平坦（话题5条目/主题，top5无关数恒≈4），构造不成立"}},
    ensure_ascii=False, indent=1), encoding="utf-8")
print(f"\nX2b/X2c-v2 完成 {time.time()-t0:.1f}s → data/x2bc_results_{TAG}.json")
