# search_knowledge.txt — 知识条目检索（扩散激活简化版，library-bootstrap 移植版）
# 用法: python3 search_knowledge.txt 关键词1 关键词2 ...  （或 py -X utf8）
# 计分: 关键词行 x3 + 标题 x2 + 正文 x1；[[关联]]邻居一跳得半分。无 m-schema3 时关键词行缺失自动降级为标题+正文。
# 匹配策略（W-035）: 英文按词边界整词子串匹配；中文按连续子串匹配（无分词依赖，纯文本环境兼容）。
# 三种典型 query 行为固化于 run_tests T45（英文整词/中文子串/混合）。
import os, re, sys

sys.stdout.reconfigure(encoding='utf-8') if hasattr(sys.stdout, 'reconfigure') else None
ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))

def parse_entries():
    entries = {}
    libs = ['knowledge/pitfalls.md', 'knowledge/patterns.md', 'knowledge/reflections.md', 'knowledge/decisions.md']
    for lib in libs:
        path = os.path.join(ROOT, lib)
        if not os.path.exists(path):
            continue
        text = re.sub(r'<!--.*?-->', '', open(path, encoding='utf-8').read(), flags=re.S)
        for block in re.split(r'\n### ', text)[1:]:
            head, _, body = block.partition('\n')
            pid = head.split('/')[0].strip()
            if not re.match(r'^[A-Z]{1,2}-\d+$', pid):
                continue
            links = set(re.findall(r'\[\[([A-Z]{1,2}-\d+)\]\]', body))
            entries[pid] = {'title': ' / '.join(head.split('/')[1:]).strip(), 'body': body, 'links': links, 'lib': lib}
    return entries

def score(e, terms):
    s = 0
    kw = re.search(r'- 关键词：(.*)', e['body'])
    kw_text = kw.group(1) if kw else ''
    for t in terms:
        if t in kw_text: s += 3
        if t in e['title']: s += 2
        if t in e['body']: s += 1
    return s

def main():
    terms = [t for t in sys.argv[1:] if t.strip()]
    if not terms:
        print('用法: python3 tools/search_knowledge.txt 关键词...'); return
    entries = parse_entries()
    scored = {pid: score(e, terms) for pid, e in entries.items()}
    scored = {pid: s for pid, s in scored.items() if s > 0}
    for pid in list(scored):
        for nb in entries[pid]['links']:
            if nb in entries and nb not in scored:
                scored[nb] = scored[pid] * 0.5
    if not scored:
        print('无命中'); return
    print('检索「%s」命中 %d 条（★直击 ☆联想）:' % (' '.join(terms), len(scored)))
    for pid, s in sorted(scored.items(), key=lambda x: -x[1])[:6]:
        e = entries[pid]
        tag = '★' if score(e, terms) > 0 else '☆'
        scene = re.search(r'- (?:场景|适用场景|决策|洞察)：(.*)', e['body'])
        brief = scene.group(1)[:60] if scene else ''
        print('%s [%s] %d分 %s' % (tag, pid, s, e['title']))
        if brief: print('    %s' % brief)

if __name__ == '__main__':
    main()
