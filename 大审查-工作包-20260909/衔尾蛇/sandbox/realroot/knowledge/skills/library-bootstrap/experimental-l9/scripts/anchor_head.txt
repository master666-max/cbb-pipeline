# anchor_head.txt — 哈希链头外部锚定（W-011；每日 wrap-up 或睡眠管线尾部运行）
# 用法: python3 anchor_head.txt <锚定目录>   （锚定目录必须在库外，如 ../_anchors/）
# 动作: 读当月 lifelog 末行链头哈希，追加「日期 | 链头哈希 | 算法版本」到锚定表
import os, sys
from datetime import date

sys.stdout.reconfigure(encoding='utf-8') if hasattr(sys.stdout, 'reconfigure') else None
if len(sys.argv) < 2:
    print('用法: anchor_head.txt <库外锚定目录>'); sys.exit(2)
anchor_dir = sys.argv[1]
log_dir = os.path.join(os.getcwd(), 'knowledge', 'lifelog')
head = ''
fn = os.path.join(log_dir, date.today().strftime('%Y-%m') + '.mdl')
if os.path.exists(fn):
    for line in reversed(open(fn, encoding='utf-8').read().splitlines()):
        if line and not line.startswith('#'):
            head = line.split(' | ')[3]
            break
if not head:
    print('✗ 当月 lifelog 无行，先运行 lifelog_append'); sys.exit(1)
os.makedirs(anchor_dir, exist_ok=True)
af = os.path.join(anchor_dir, 'anchor-table.md')
if not os.path.exists(af):
    open(af, 'w', encoding='utf-8').write('# 外部锚定表（日期 | 链头哈希 | 算法版本）\n\n')
open(af, 'a', encoding='utf-8').write('| %s | %s | sha256-16 |\n' % (date.today().isoformat(), head))
print('已锚定: %s -> %s' % (date.today().isoformat(), head))
