# X1 · 静态基线组 G0/G1 + 存储三探针（存储轴 + 遗忘轴）
# 运行：py -X utf8 proto/x1_static.txt
# 输出：data/x1_static.jsonl（每种子一行）、data/x1_probes.json、reports/X1_静态基线.md
# 纪律：结果"过于干净"先怀疑实验（教训 #8）；否定性与正面同权重记录
import sys, os, re, json, time, hashlib, shutil, subprocess, importlib.util, importlib.machinery
from pathlib import Path
from datetime import date, timedelta
from contextlib import redirect_stdout
from io import StringIO

BASE = Path(__file__).resolve().parent.parent
VENDOR = BASE / "proto" / "vendor" / "bs3_snapshot.txt"
PKG = BASE / "sandbox" / "pkgdata"
SB = BASE / "sandbox"
DATA = BASE / "data"
SEEDS = [1, 2, 3, 4, 5, 6, 7, 8]

def load_mod(name, path):
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_file_location(name, str(path), loader=loader)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod

m = load_mod("bs3_vendor", VENDOR)
m.set_data_dir(str(PKG))
evocore = load_mod("evocore", BASE / "proto" / "evocore.txt")

G1_CFG = {"w_kw": 3.0, "w_content": 1.0, "w_imp": 0.5, "w_age": 0.1, "filter_zero": 1}  # 报告 E1 人工答案
t0 = time.time()
rows = []

# ══ 1. 八种子 G0 库 + G0/G1 基线 ════════════════════════════════════
worlds = {}
for seed in SEEDS:
    lib = SB / f"lib-g0-s{seed}"
    if not (lib / "state.json").exists():
        m.cmd_install("L2", [], lib, True)
        entries, kw_of = evocore.build_world(seed)
        tail = sorted((lib / "audit").glob("lifelog-*.md"))[-1].read_text(encoding="utf-8").strip().splitlines()[-1]
        seid = "M-" + re.search(r"h:([0-9a-f]{64})", tail).group(1)[:12]
        for e in entries:
            rec = {"id": e["id"], "created_at": e["created_at"], "updated_at": "",
                   "content": e["content"], "keywords": e["keywords"], "links": e["links"],
                   "source_event_id": seid, "importance": e["importance"],
                   "confidence": "中", "validity": {"t_valid": e["created_at"]}}
            m.cmd_engine("append", str(lib), json.dumps(rec, ensure_ascii=False))
    # 库已存在则从盘上恢复世界（不重复写入）
    entries, kw_of = evocore.build_world(seed)
    train, held, audit = evocore.build_tasks(seed, entries)
    worlds[seed] = {"entries": entries, "kw_of": kw_of, "train": train, "held": held, "audit": audit}
    row = {"seed": seed, "ts": time.strftime("%F %T")}
    for gname, cfg in (("G0", evocore.DEFAULT_CFG), ("G1", G1_CFG)):
        for sname, ts in (("train", train), ("held", held), ("audit", audit)):
            row[f"{gname}_{sname}_U"] = round(evocore.score_lib(lib, ts, cfg, kw_of=kw_of), 4)
        row[f"{gname}_audit_recall5"] = round(evocore.recall_at_k(lib, audit, cfg, kw_of=kw_of), 4)
    buf = StringIO()
    with redirect_stdout(buf):
        m.cmd_engine("doctor", str(lib))
    row["doctor"] = buf.getvalue().strip().splitlines()[0]
    st = json.loads((lib / "state.json").read_text(encoding="utf-8"))
    row["merkle16"] = st["merkle_root"][:16]
    rows.append(row)
    print(f"[s{seed}] G0 audit U={row['G0_audit_U']}  G1 audit U={row['G1_audit_U']}  | {row['doctor']}")

(DATA / "x1_static.jsonl").write_text(
    "\n".join(json.dumps(r, ensure_ascii=False) for r in rows) + "\n", encoding="utf-8")

# 配对差（同种子 G1−G0）
d_audit = [r["G1_audit_U"] - r["G0_audit_U"] for r in rows]
ci = evocore.bootstrap_ci_low(d_audit, seed=1)
print(f"\nG1−G0 审计集配对差：均值 {sum(d_audit)/len(d_audit):+.4f}  正种子 {sum(1 for x in d_audit if x>0)}/8  bootstrap95下界 {ci:+.4f}")

# ══ 2. 并发探针：8 进程真引擎 CLI append（幂等：跑过即跳过）════════
lib_p = SB / "lib-probe"
if list(lib_p.rglob("c-*.json")):
    # 探针库已消费——从持久状态恢复测量（stdout 曾被输出管道截断的补救；doctor 只读可重测）
    st_now = json.loads((lib_p / "state.json").read_text(encoding="utf-8"))
    w8, _ = evocore.build_world(8)
    accum_base = sum(e["importance"] for e in w8)          # install=0，基线=世界 importance 之和
    accum_gained = st_now["importance_accum"] - accum_base
    buf = StringIO()
    with redirect_stdout(buf):
        m.cmd_engine("doctor", str(lib_p))
    doctor_p = buf.getvalue()
    chain_bad = len(re.findall(r"断链|格式异常", doctor_p))
    n_c = len(list(lib_p.rglob("c-*.json")))
    rcs = "recovered"
    print(f"\n[并发·恢复] 盘上 c 条目 {n_c}/8  accum 增量 {accum_gained}/40（丢失 {40 - accum_gained}）  "
          f"doctor<< {doctor_p.strip().splitlines()[0]} >>")
else:
    if not (lib_p / "state.json").exists():
        m.cmd_install("L2", [], lib_p, True)
        entries, _ = evocore.build_world(8)
        tail = sorted((lib_p / "audit").glob("lifelog-*.md"))[-1].read_text(encoding="utf-8").strip().splitlines()[-1]
        seid = "M-" + re.search(r"h:([0-9a-f]{64})", tail).group(1)[:12]
        for e in entries:
            m.cmd_engine("append", str(lib_p), json.dumps(
                {"id": e["id"], "created_at": e["created_at"], "updated_at": "", "content": e["content"],
                 "keywords": e["keywords"], "links": [], "source_event_id": seid,
                 "importance": e["importance"], "confidence": "中", "validity": {}}, ensure_ascii=False))
    st_before = json.loads((lib_p / "state.json").read_text(encoding="utf-8"))
    accum_before = st_before["importance_accum"]
    procs = []
    for i in range(8):
        rec = {"id": f"c-{i}", "created_at": date.today().isoformat(), "updated_at": "",
               "content": f"并发探针条目 {i}", "keywords": ["probe"], "links": [],
               "source_event_id": "probe-concurrent", "importance": 5,
               "confidence": "中", "validity": {}}
        procs.append(subprocess.Popen([sys.executable, "-X", "utf8", str(VENDOR), "engine", "append",
                                       "--lib", str(lib_p), "--text", json.dumps(rec, ensure_ascii=False)],
                                      stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL, cwd=str(BASE)))
    rcs = [p.wait() for p in procs]
    st_after = json.loads((lib_p / "state.json").read_text(encoding="utf-8"))
    accum_gained = st_after["importance_accum"] - accum_before
    buf = StringIO()
    with redirect_stdout(buf):
        m.cmd_engine("doctor", str(lib_p))
    doctor_p = buf.getvalue()
    chain_bad = len(re.findall(r"断链|格式异常", doctor_p))
    print(f"\n[并发] 8 进程 rc={rcs}  accum 增量 {accum_gained}/40（丢失 {40 - accum_gained}）  doctor<< {doctor_p.strip().splitlines()[0]} >>")

# ══ 3. 截断探针（只读副本；两层判定）═══════════════════════════════
lib_t = SB / "lib-trunc"
if lib_t.exists():
    shutil.rmtree(lib_t)
shutil.copytree(SB / "lib-g0-s1", lib_t)
af = sorted((lib_t / "audit").glob("lifelog-*.md"))[-1]
lines = af.read_text(encoding="utf-8").splitlines()
assert len(lines) >= 5
removed = lines[-2:]
lines = lines[:-2]
af.write_bytes(("\n".join(lines) + "\n").encode("utf-8"))     # LF 落盘（红线 26）
st = json.loads((lib_t / "state.json").read_text(encoding="utf-8"))
st["merkle_root"] = m.merkle_root(lib_t, exclude={"state.json"})   # 攻击者重算根
(lib_t / "state.json").write_bytes(json.dumps(st, ensure_ascii=False, indent=1).encode("utf-8"))
buf = StringIO()
with redirect_stdout(buf):
    m.cmd_engine("doctor", str(lib_t))
doctor_t = buf.getvalue()
removed_ids = sorted({mm.group(1) for mm in (re.search(r"entry:(\S+) →", x) for x in removed) if mm})
crypto_blind = ("断链" not in doctor_t) and ("根不一致" not in doctor_t)
set_layer_flagged = "memory有而audit无" in doctor_t
print(f"\n[截断] 删除 {removed_ids}（audit 尾 2 行）+ 重算根")
print(f"  密码学层（链/根） blind={crypto_blind}（报告 2.3 预测 True=盲）")
print(f"  集合层（四方对账） flagged={set_layer_flagged}（但与'漏刷根'类 bug 同型=噪声通道，报告 2.4）")
print("  doctor<<\n" + "\n".join("    " + x for x in doctor_t.strip().splitlines()) + "\n  >>")

# ══ 4. 遗忘探针（幂等：attic 已有 e0 则只复测不重做）═══════════════
lib_f = SB / "lib-g0-s1"
if (lib_f / "memory" / "attic" / "e0.json").exists():
    caught = "skipped"
    attic = sorted(p.stem for p in (lib_f / "memory" / "attic").glob("*.json"))
    moved_ok = set(attic) == {f"e{i}" for i in range(8)}
    live = {p.stem for p in (lib_f / "memory").rglob("*.json") if "attic" not in str(p)}
    stay_ok = {"e8", "e9"} <= live
    buf = StringIO()
    with redirect_stdout(buf):
        m.cmd_engine("doctor", str(lib_f))
    doc_after = buf.getvalue().strip().splitlines()[0]
    print(f"\n[遗忘] 探针已消费——复测：出仓精确={moved_ok} 留守零误伤={stay_ok} doctor<< {doc_after} >>")
else:
    inv_old = (date.today() - timedelta(days=100)).isoformat()   # 超 TTL 90 天 → 应出仓
    inv_new = (date.today() - timedelta(days=10)).isoformat()   # 未超 TTL → 应留守
    expect_out, expect_stay = [f"e{i}" for i in range(8)], ["e8", "e9"]
    for eid, inv in [(x, inv_old) for x in expect_out] + [(x, inv_new) for x in expect_stay]:
        hits = list((lib_f / "memory").rglob(f"{eid}.json"))
        assert len(hits) == 1, f"{eid} 定位失败"
        e = json.loads(hits[0].read_text(encoding="utf-8"))
        e.setdefault("validity", {})["t_invalid"] = inv
        hits[0].write_text(json.dumps(e, ensure_ascii=False, indent=1), encoding="utf-8")  # 人工失效裁决（绕过引擎）
    buf = StringIO()
    with redirect_stdout(buf):
        m.cmd_engine("doctor", str(lib_f))
    caught = "根不一致" in buf.getvalue()
    print(f"\n[遗忘] 绕过引擎直接编辑 10 条 → doctor 抓到根不一致：{caught}（写后刷根纪律的有效性证据）")
    buf = StringIO()
    with redirect_stdout(buf):
        m.cmd_engine("retire", str(lib_f))
    attic = sorted(p.stem for p in (lib_f / "memory" / "attic").glob("*.json"))
    moved_ok = set(attic) == set(expect_out)
    live = {p.stem for p in (lib_f / "memory").rglob("*.json") if "attic" not in str(p)}
    stay_ok = set(expect_stay) <= live
    buf = StringIO()
    with redirect_stdout(buf):
        m.cmd_engine("doctor", str(lib_f))
    doc_after = buf.getvalue().strip().splitlines()[0]
    print(f"[遗忘] retire 后 attic={attic}  出仓精确={moved_ok}  留守零误伤={stay_ok}  doctor<< {doc_after} >>")

# ══ 5. 收口 ════════════════════════════════════════════════════════
probes = {
    "concurrent": {"procs": 8, "rcs": rcs, "accum_gained": accum_gained, "expected": 40,
                   "lost": 40 - accum_gained, "doctor_first_line": doctor_p.strip().splitlines()[0],
                   "chain_break_mentions": chain_bad},
    "truncation": {"removed": removed_ids, "crypto_blind": crypto_blind,
                   "set_layer_flagged": set_layer_flagged, "doctor": doctor_t.strip().splitlines()},
    "forget": {"doctor_caught_manual_edit": caught, "attic": attic, "moved_exact": moved_ok,
               "stay_untouched": stay_ok, "doctor_after": doc_after},
    "paired_G1_minus_G0_audit": {"diffs": d_audit, "mean": round(sum(d_audit) / 8, 4),
                                 "n_pos": sum(1 for x in d_audit if x > 0), "ci_low": round(ci, 4)},
}
(DATA / "x1_probes.json").write_text(json.dumps(probes, ensure_ascii=False, indent=1), encoding="utf-8")
print(f"\nX1 完成，耗时 {time.time() - t0:.1f}s → data/x1_probes.json")
