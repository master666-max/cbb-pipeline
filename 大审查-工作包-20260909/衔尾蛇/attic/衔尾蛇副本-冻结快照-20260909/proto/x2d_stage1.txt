# X2d · 重组实验 第一阶段：批量写入触发反思 → 产出素材包（合成由 harness 按源码原版提示词执行）
# 运行：py -X utf8 proto/x2d_stage1.txt
# 产出：lib-x2d（80 主库 + 30 扩展条目）、素材包路径与全文、pre/post 批量写入的检索指标
import sys, re, json, time, random, importlib.util, importlib.machinery
from pathlib import Path
from datetime import date, timedelta
from contextlib import redirect_stdout
from io import StringIO

BASE = Path(__file__).resolve().parent.parent
VENDOR = BASE / "proto" / "vendor" / "bs3_snapshot.txt"
PKG = BASE / "sandbox" / "pkgdata"
SB = BASE / "sandbox"
DATA = BASE / "data"

def load_mod(name, path):
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_file_location(name, str(path), loader=loader)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod

evocore = load_mod("evocore", BASE / "proto" / "evocore.txt")
m = load_mod("bs3_vendor", VENDOR)
m.set_data_dir(str(PKG))
LIB = SB / "lib-x2d"
PARAM_SETS = {
    "default": evocore.DEFAULT_CFG,
    "evolved": {"w_kw": 6.0, "w_content": 1.8, "w_imp": 0.0, "w_age": 0.0, "filter_zero": 1},  # X2a G4-s1 终态
}

def build_ext(n=30, topics=6, kws=25, seed=99):
    """扩展批次：新域条目（T0-T5 与主库重叠制造 token 竞争；重要度 3-6 温和）。"""
    rnd = random.Random(seed)
    today = date.today()
    out = []
    for i in range(n):
        t = rnd.randrange(topics)
        while True:
            k = rnd.randrange(kws)
            if f"K{k}" != f"T{t}":
                break
        created = (today - timedelta(days=rnd.randint(0, 30))).isoformat()
        out.append({"id": f"x{i}", "topic": f"T{t}", "kw": f"K{k}",
                    "created_at": created, "updated_at": "",
                    "content": f"T{t} K{k} 扩展批次条目 {i}",
                    "keywords": [f"K{k}"], "links": [],
                    "importance": rnd.randint(3, 6), "confidence": "中",
                    "validity": {"t_valid": created}})
    return out

def measure(lib, label, rev=0):
    entries, kw_of = evocore.build_world(1)
    _, _, audit = evocore.build_tasks(1, entries)
    kw_qs = [f"K{k}" for k in range(20)]
    out = {}
    for pname, cfg in PARAM_SETS.items():
        out[pname] = {
            "audit_recall5": round(evocore.recall_at_k(lib, audit, cfg, rev=rev, kw_of=kw_of), 4),
            "audit_U": round(evocore.score_lib(lib, audit, cfg, rev=rev, kw_of=kw_of), 4),
            "kw_recall5": round(sum(
                1 for q in kw_qs
                if any(kw_of.get(fid) == q for fid in evocore.ranked_ids(lib, q, cfg, rev=rev, k=5))) / len(kw_qs), 4),
        }
    print(f"  [{label}] " + json.dumps(out, ensure_ascii=False))
    return out

if not (LIB / "state.json").exists():
    m.cmd_install("L2", [], LIB, True)
    entries, _ = evocore.build_world(1)
    tail = sorted((LIB / "audit").glob("lifelog-*.md"))[-1].read_text(encoding="utf-8").strip().splitlines()[-1]
    seid = "M-" + re.search(r"h:([0-9a-f]{64})", tail).group(1)[:12]
    for e in entries:
        m.cmd_engine("append", str(LIB), json.dumps(
            {"id": e["id"], "created_at": e["created_at"], "updated_at": "",
             "content": e["content"], "keywords": e["keywords"], "links": e["links"],
             "source_event_id": seid, "importance": e["importance"],
             "confidence": "中", "validity": {"t_valid": e["created_at"]}}, ensure_ascii=False))

pre = measure(LIB, "pre-80条", rev=0)

# 批量写入 30 条扩展条目（importance_accum 必然过反思阈值 50）
ext = build_ext()
st = json.loads((LIB / "state.json").read_text(encoding="utf-8"))
tail = sorted((LIB / "audit").glob("lifelog-*.md"))[-1].read_text(encoding="utf-8").strip().splitlines()[-1]
seid = "M-" + re.search(r"h:([0-9a-f]{64})", tail).group(1)[:12]
for e in ext:
    m.cmd_engine("append", str(LIB), json.dumps(
        {"id": e["id"], "created_at": e["created_at"], "updated_at": "",
         "content": e["content"], "keywords": e["keywords"], "links": e["links"],
         "source_event_id": seid, "importance": e["importance"],
         "confidence": "中", "validity": {"t_valid": e["created_at"]}}, ensure_ascii=False))

post = measure(LIB, "post-110条(批量后)", rev=1)
buf = StringIO()
with redirect_stdout(buf):
    m.cmd_engine("wrapup", str(LIB))
wrapup_out = buf.getvalue()
triggered = "反思触发" in wrapup_out
print(f"\nwrapup 反思触发：{triggered}  accum={json.loads((LIB/'state.json').read_text(encoding='utf-8'))['importance_accum']}")
buf = StringIO()
with redirect_stdout(buf):
    m.cmd_engine("reflect", str(LIB))
reflect_out = buf.getvalue()
materials = sorted((LIB / "memory" / "reflect").glob("materials-*.md"))[-1]
print(f"素材包：{materials.name}\n--- 素材包全文 ---")
print(materials.read_text(encoding="utf-8"))
(DATA / "x2d_stage1.json").write_text(json.dumps(
    {"pre": pre, "post": post, "reflect_triggered": triggered,
     "materials_file": str(materials), "wrapup": wrapup_out},
    ensure_ascii=False, indent=1), encoding="utf-8")
