# X2d · 重组实验 第二阶段：合成条目入库 vs 拼接摘要对照 → 检索保持率与密度
# 运行：py -X utf8 proto/x2d_stage2.txt   （前置：x2d_stage1 已跑、data/x2d_R_entries.json 已备）
# 合成方式声明：R-1/2/3 由 harness（ZCode）按源码 L670-679 REFLECT_SYNTHESIS 原版提示词逐字执行产出，
#   依据条目号全部来自素材包（无据不写）；对照库只加 1 条内容等价的拼接摘要（无 links）。
import sys, re, json, time, shutil, importlib.util, importlib.machinery
from pathlib import Path
from contextlib import redirect_stdout
from io import StringIO

BASE = Path(__file__).resolve().parent.parent
VENDOR = BASE / "proto" / "vendor" / "bs3_snapshot.txt"
PKG = BASE / "sandbox" / "pkgdata"
SB = BASE / "sandbox"
DATA = BASE / "data"

def load_mod(name, path):
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_file_location(name, str(path), loader=loader)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod

evocore = load_mod("evocore", BASE / "proto" / "evocore.txt")
m = load_mod("bs3_vendor", VENDOR)
m.set_data_dir(str(PKG))
LIB = SB / "lib-x2d"
LIB_R = SB / "lib-x2d-R"
LIB_C = SB / "lib-x2d-C"
PARAM_SETS = {
    "default": evocore.DEFAULT_CFG,
    "evolved": {"w_kw": 6.0, "w_content": 1.8, "w_imp": 0.0, "w_age": 0.0, "filter_zero": 1},
}

r_entries = json.loads((DATA / "x2d_R_entries.json").read_text(encoding="utf-8"))

# 分支两库（在批量后状态上分叉）
for dst in (LIB_R, LIB_C):
    if dst.exists():
        shutil.rmtree(dst)
    shutil.copytree(LIB, dst)

def append_entries(lib, recs):
    for rec in recs:
        m.cmd_engine("append", str(lib), json.dumps(rec, ensure_ascii=False))

append_entries(LIB_R, r_entries)
ctrl = {
    "id": "R-CTRL", "created_at": "2026-09-08", "updated_at": "",
    "content": "拼接摘要（对照：与 R-1/2/3 内容等价但无独立条目化与依据链接）："
               + " ".join(e["content"] for e in r_entries),
    "keywords": ["规律", "摘要", "反思"], "links": [],
    "source_event_id": "reflect-20260908", "importance": 7,
    "confidence": "高", "validity": {"t_valid": "2026-09-08"},
}
append_entries(LIB_C, [ctrl])

# 各库 reflect-done（重置累计器，走引擎不手改 state）
for lib in (LIB_R, LIB_C):
    buf = StringIO()
    with redirect_stdout(buf):
        m.cmd_engine("reflect-done", str(lib))

def measure(lib, label):
    entries, kw_of = evocore.build_world(1)
    _, _, audit = evocore.build_tasks(1, entries)
    kw_qs = [f"K{k}" for k in range(20)]
    n_entries = len(list((lib / "memory").rglob("*.json")))
    out = {"n_entries": n_entries}
    for pname, cfg in PARAM_SETS.items():
        out[pname] = {
            "audit_recall5": round(evocore.recall_at_k(lib, audit, cfg, kw_of=kw_of), 4),
            "audit_U": round(evocore.score_lib(lib, audit, cfg, kw_of=kw_of), 4),
            "kw_recall5": round(sum(
                1 for q in kw_qs
                if any(kw_of.get(fid) == q for fid in evocore.ranked_ids(lib, q, cfg, k=5))) / len(kw_qs), 4),
        }
    buf = StringIO()
    with redirect_stdout(buf):
        m.cmd_engine("doctor", str(lib))
    out["doctor_ok"] = "全一致" in buf.getvalue()
    print(f"  [{label}] " + json.dumps(out, ensure_ascii=False))
    return out

print("== X2d 第二阶段测量（stage1 的 pre/post 见 data/x2d_stage1.json）==")
m_R = measure(LIB_R, "合成库 R（+3 规律条目，带依据链接）")
m_C = measure(LIB_C, "对照库 C（+1 拼接摘要，无链接）")

s1 = json.loads((DATA / "x2d_stage1.json").read_text(encoding="utf-8"))
keep = {}
for pname in PARAM_SETS:
    keep[pname] = {
        "audit_recall5_保持率_R_vs_batch": round(m_R[pname]["audit_recall5"] / max(1e-9, s1["post"][pname]["audit_recall5"]), 4),
        "kw_recall5_保持率_R_vs_batch": round(m_R[pname]["kw_recall5"] / max(1e-9, s1["post"][pname]["kw_recall5"]), 4),
        "kw_recall5_恢复_R_vs_pre": round(m_R[pname]["kw_recall5"] / max(1e-9, s1["pre"][pname]["kw_recall5"]), 4),
        "R_vs_C_audit_recall5": round(m_R[pname]["audit_recall5"] - m_C[pname]["audit_recall5"], 4),
        "R_vs_C_kw_recall5": round(m_R[pname]["kw_recall5"] - m_C[pname]["kw_recall5"], 4),
    }
print("\n保持率与 R/C 对比：")
print(json.dumps(keep, ensure_ascii=False, indent=1))
(DATA / "x2d_stage2.json").write_text(json.dumps(
    {"R": m_R, "C": m_C, "compare": keep, "r_entries": r_entries, "ctrl": ctrl},
    ensure_ascii=False, indent=1), encoding="utf-8")
print(f"\nX2d 完成 → data/x2d_stage2.json")
