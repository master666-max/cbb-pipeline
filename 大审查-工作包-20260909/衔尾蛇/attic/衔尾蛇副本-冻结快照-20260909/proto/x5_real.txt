def load_mod(name, path):
    loader = importlib.machinery.SourceFileLoader(name, str(path))
    spec = importlib.util.spec_from_file_location(name, str(path), loader=loader)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod

evocore = load_mod("evocore", BASE / "proto" / "evocore.txt")
m = load_mod("bs3_vendor", VENDOR)
m.set_data_dir(str(PKG))
MARGIN, TOL, GENS, KIDS = META["margin"], 0.005, 25, 3
SEEDS = [1, 2, 3, 4, 5]
LIB = SB / "lib-real"

# 条目清单与查询构造：有关键词的条目才能出任务（真实语料的天然限制，如实统计）
all_entries = {f.stem: json.loads(f.read_text(encoding="utf-8"))
               for f in (LIB / "memory").rglob("*.json")}
tasks_src = []
for eid, e in all_entries.items():
    if (e.get("validity") or {}).get("t_invalid"):
        continue
    kws = [k for k in (e.get("keywords") or []) if k and k != "轨迹"]
    if kws:
        tasks_src.append({"q": " ".join(kws[:2]), "golden": eid, "kw_hint": kws[0]})
print(f"真实库条目 {len(all_entries)} 条；可出任务（有关键词）{len(tasks_src)} 条")
n_train = max(1, int(len(tasks_src) * 0.6))
n_audit = max(2, int(len(tasks_src) * 0.15))

def split_for(seed):
    rnd = random.Random(seed * 613 + 11)
    idx = list(range(len(tasks_src)))
    rnd.shuffle(idx)
    sel = [tasks_src[i] for i in idx]
    return (sel[:n_train], sel[n_train:len(tasks_src) - n_audit], sel[len(tasks_src) - n_audit:])

def build_lib(group, seed):
    """真实库 substrate 每组一份（演化晋升写各自 state.json/审计链，互不污染）。"""
    lib = SB / f"lib-real-{group}-s{seed}"
    if not (lib / "state.json").exists():
        import shutil
        shutil.copytree(LIB, lib)
    return lib

def runs(lib, tasks, cfg, kw_of):
    hits = rr = 0.0
    for t in tasks:
        ranked = [fid for fid, _ in evocore.rank_lib(lib, t["q"], cfg)[:5]]
        if t["golden"] in ranked:
            hits += 1
            rr += 1.0 / (ranked.index(t["golden"]) + 1)
    n = max(1, len(tasks))
    return {"recall5": round(hits / n, 4), "mrr": round(rr / n, 4)}

def utility_of(lib, t, cfg, kw_of):
    top = [fid for fid, _ in evocore.rank_lib(lib, t["q"], cfg)[:5]]
    hit = 1.0 if t["golden"] in top else 0.0
    n_irrel = sum(1 for f in top if f != t["golden"])     # 真实语料无 kw 元素可判无关，用非金标计数
    return hit - evocore.LAMBDA * n_irrel

def score_true(lib, tasks, cfg):
    return sum(utility_of(lib, t, cfg) for t in tasks) / max(1, len(tasks))

# ── G0 基线（每种子）────────────────────────────────────────────────
g0 = {}
for seed in SEEDS:
    train, held, audit = split_for(seed)
    g0[seed] = {"audit_recall5": runs(LIB, audit, evocore.DEFAULT_CFG, None)["recall5"],
                "audit_mrr": runs(LIB, audit, evocore.DEFAULT_CFG, None)["mrr"]}
print(f"G0 真实基线：{json.dumps(g0, ensure_ascii=False)}")

def promote(lib, gen, cfg):
    st_p = lib / "state.json"
    st = json.loads(st_p.read_text(encoding="utf-8"))
    st["law_params"] = cfg
    m.chain_append(lib / "audit", f"X5 真实库法层晋升 gen={gen} cfg={json.dumps(cfg, ensure_ascii=False)}")
    m._ledger_append(lib, "law-promote", f"gen={gen}")
    st["merkle_root"] = m.merkle_root(lib, exclude={"state.json"})
    st_p.write_text(json.dumps(st, ensure_ascii=False, indent=1), encoding="utf-8")

def run_evo(group, seed, scene="clean"):
    lib = build_lib(group, seed)
    train, held0, audit = split_for(seed)
    rng = random.Random(seed * 31 + sum(map(ord, group)))
    beta = 0.0
    mainline = dict(evocore.DEFAULT_CFG)
    archive = [dict(mainline)]
    cache = {}
    evals = 0

    def ju(t, cfg, gen, ti):
        top5 = [fid for fid, _ in evocore.rank_lib(lib, t["q"], cfg)[:5]]
        hit = 1.0 if t["golden"] in top5 else 0.0
        u = hit - evocore.LAMBDA * sum(1 for f in top5 if f != t["golden"])
        if scene == "drift":
            # 真实语料的"吸睛"代理：条目 content 含「超绝/激进/内省」等高宣称词的条目被评判者偏爱
            flashy = sum(1 for f in top5 if any(w in all_entries[f]["content"] for w in ("超绝", "激进", "内省", "前沿")))
            u += beta * 0.2 * flashy
        return u

    def jc(cfg, tasks, gen, tag):
        nonlocal evals
        key = (tuple(sorted(cfg.items())), tag, gen)
        if key not in cache:
            evals += 1
            cache[key] = sum(ju(t, cfg, gen, ti) for ti, t in enumerate(tasks)) / len(tasks)
        return cache[key]

    for gen in range(1, GENS + 1):
        if scene == "drift":
            beta = min(1.0, 0.1 * gen)
        for _ in range(KIDS):
            pool = sorted(archive, key=lambda c: -jc(c, train, gen, "train"))
            parent = rng.choice(pool[:3]) if (rng.random() < 0.7 and len(pool) >= 3) else rng.choice(pool)
            child = evocore.mutate(parent, rng)
            tr_p, tr_c = jc(parent, train, gen, "train"), jc(child, train, gen, "train")
            if group in ("G4", "G5") and tr_c < tr_p - 3 * TOL:
                continue
            if tr_c - tr_p > -TOL:
                archive.append(dict(child))
                if len(archive) > 30:
                    archive.pop(min(range(len(archive)), key=lambda i: jc(archive[i], train, gen, "train")))
            if group == "G2":
                ok = tr_c > jc(mainline, train, gen, "train")
            else:
                d_tr = [ju(t, child, gen, i) - ju(t, mainline, gen, i) for i, t in enumerate(train)]
                if sum(d_tr) / len(d_tr) <= MARGIN:
                    ok = False
                else:
                    d_he = [ju(t, child, gen, i) - ju(t, mainline, gen, i) for i, t in enumerate(held0)]
                    evals += 1
                    ok = (sum(d_he) / len(d_he) > MARGIN
                          and evocore.bootstrap_ci_low(d_he, n_boot=META["bootstrap_n"], seed=seed) > max(0.0, MARGIN))
            if ok:
                mainline = dict(child)
                promote(lib, gen, mainline)
    out = {"group": group, "seed": seed, "scene": scene, "final_mainline": mainline,
           "audit_true": runs(lib, audit, mainline, None),
           "g0_audit": g0[seed], "evals": evals}
    buf = StringIO()
    with redirect_stdout(buf):
        m.cmd_engine("doctor", str(lib))
    out["doctor_ok"] = ("全一致" in buf.getvalue()) or ("无 state" in buf.getvalue())
    (DATA / f"x5_real-{group}-{scene}-s{seed}.json").write_text(
        json.dumps(out, ensure_ascii=False, indent=1), encoding="utf-8")
    return out
