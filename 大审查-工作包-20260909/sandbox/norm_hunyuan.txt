def num(s):
    s = s.replace("**", "").strip()
    s = re.sub(r"[(（].*?[)）]\s*$", "", s).strip()
    m = re.match(r"^[+−\-]?\d+(\.\d+)?%?$", s)
    if not m: return None
    s2 = s.replace("%", "").replace("−", "-").replace("+", "")
    try: return float(s2)
    except ValueError: return None

def first_num(s):
    m = re.search(r"[+−\-]?\d+\.?\d*", s.replace("**", ""))
    if not m: return None
    try: return float(m.group(0).replace("−", "-"))
    except ValueError: return None

def cells(line):
    return [c.strip() for c in line.strip().strip("|").split("|")]

def clean(s):
    return s.replace("**", "").replace("`", "").strip()

# 特例：任务报告 H7 复合单元格表（two-tier ΔU / 回撤 等）
def extract_task_h7():
    fn = "任务报告.md"
    lines = open(os.path.join(SRC, fn), encoding="utf-8").read().splitlines()
    for i, ln in enumerate(lines):
        if ln.startswith("| α | two-tier ΔU"):
            j = i + 2
            while j < len(lines) and lines[j].strip().startswith("|"):
                cs = cells(lines[j])
                if len(cs) >= 6:
                    alpha = clean(cs[0])
                    for col, suf in ((1, "-two-tier"), (2, "-anchor")):
                        gs = first_num(cs[col]) if col < len(cs) else None
                        rid = f"混元任务-H7-α{alpha}{suf}"
                        # 冻结列: "0/12"->drift=False, "12/12"->True
                        fr = cs[5] if len(cs) > 5 else ""
                        df = None
                        m = re.match(r"^(\d+)/(\d+)$", fr.replace("**", "").strip())
                        if m: df = int(m.group(1)) > 0
                        rows.append({"source": "混元", "run_id": rid, "round": None, "gen_score": gs,
                                     "adopt_count": None, "drift_flag": df, "raw_line_no": j + 1,
                                     "raw_file": f"混元/{fn}", "notes_raw": lines[j].rstrip()})
                j += 1
            return

FILES_CACHE = {}
def get_lines(fn):
    if fn not in FILES_CACHE:
        FILES_CACHE[fn] = open(os.path.join(SRC, fn), encoding="utf-8").read().splitlines()
    return FILES_CACHE[fn]

used = set()
for cfg in CFG:
    fn, sig, keycols, fmt, metrics, adopt_col, mode, round_col = cfg
    lines = get_lines(fn)
    hit = None
    for i, ln in enumerate(lines):
        if ln.rstrip() == sig:
            hit = i; break
    if hit is None:
        dirty.append((fn, 0, sig, "表头未找到(配置失效)"))
        continue
    used.add((fn, hit + 1))
    j = hit + 2  # 跳过分隔行
    while j < len(lines) and lines[j].strip().startswith("|"):
        cs = cells(lines[j])
        if len(cs) < 2: break
        keys = [clean(cs[kc]) if kc < len(cs) else "?" for kc in keycols]
        kd = {f"k{i}": k for i, k in enumerate(keys)}
        if len(keys) == 1: kd["k"] = keys[0]
        rid_base = fmt.format(**kd)
        if mode == "detect":
            rv = num(cs[round_col]) if round_col < len(cs) else None
            rv2 = first_num(cs[round_col]) if rv is None and round_col < len(cs) else rv
            if rv2 is not None and ("—" not in cs[round_col]):
                rows.append({"source": "混元", "run_id": rid_base, "round": rv2, "gen_score": None,
                             "adopt_count": None, "drift_flag": True, "raw_line_no": j + 1,
                             "raw_file": f"混元/{fn}", "notes_raw": lines[j].rstrip()})
            else:
                rows.append({"source": "混元", "run_id": rid_base, "round": None, "gen_score": None,
                             "adopt_count": None, "drift_flag": False, "raw_line_no": j + 1,
                             "raw_file": f"混元/{fn}", "notes_raw": lines[j].rstrip()})
        elif mode == "gen":
            rv = num(cs[round_col]) if round_col < len(cs) else None
            gs = None
            for col, suf in metrics:
                if col < len(cs): gs = num(cs[col])
            rows.append({"source": "混元", "run_id": rid_base, "round": rv, "gen_score": gs,
                         "adopt_count": None, "drift_flag": None, "raw_line_no": j + 1,
                         "raw_file": f"混元/{fn}", "notes_raw": lines[j].rstrip()})
        else:
            for col, suf in metrics:
                rid = rid_base + suf
                if col >= len(cs):
                    dirty.append((fn, j + 1, lines[j], f"列数不足 col={col}")); continue
                raw_v = cs[col]
                gs = num(raw_v)
                if gs is None: gs = first_num(raw_v)  # 复合单元格取首值(口径已登记)
                if gs is None and raw_v not in ("—", "-", ""):
                    dirty.append((fn, j + 1, lines[j], f"真值列无法解析: {raw_v}"))
                    continue
                ac = num(cs[adopt_col]) if adopt_col is not None and adopt_col < len(cs) else None
                rows.append({"source": "混元", "run_id": rid, "round": None, "gen_score": gs,
                             "adopt_count": ac, "drift_flag": None, "raw_line_no": j + 1,
                             "raw_file": f"混元/{fn}", "notes_raw": lines[j].rstrip()})
        j += 1

extract_task_h7()
