# 阶段0 盘点脚本：遍历大审查8源全部文件，记录格式/大小/行数/结构特征
# 输出: reports/00-inventory.tsv (全量清单) + stdout 摘要
import os, sys, json, hashlib

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))  # 大审查/
SOURCES = ["deepseek", "glm", "qwen", "复审", "混元", "统合报告", "衔尾蛇", "豆包"]
OUT = os.path.join(ROOT, "reports", "00-inventory.tsv")

def sniff(path):
    """返回 (ext, is_text, n_lines, first_line, sha1_8)"""
    ext = os.path.splitext(path)[1].lower()
    sha1 = ""
    try:
        with open(path, "rb") as f:
            data = f.read()
        sha1 = hashlib.sha1(data).hexdigest()[:8]
        if b"\x00" in data[:4096]:
            return ext, False, 0, "", sha1
        for enc in ("utf-8", "utf-8-sig", "gb18030", "utf-16"):
            try:
                t = data.decode(enc)
                break
            except (UnicodeDecodeError, LookupError):
                t = None
        if t is None:
            return ext, False, 0, "", sha1
        lines = t.splitlines()
        first = lines[0].strip()[:120] if lines else ""
        return ext, True, len(lines), first, sha1
    except OSError as e:
        return ext, False, -1, "ERR:" + str(e)[:60], ""

rows = []
agg = {}
for src in SOURCES:
    base = os.path.join(ROOT, src)
    st = {"files": 0, "bytes": 0, "text": 0, "binary": 0, "err": 0, "ext": {}, "subdirs": set()}
    for dirpath, dirnames, filenames in os.walk(base):
        rel_dir = os.path.relpath(dirpath, base)
        if rel_dir != ".":
            st["subdirs"].add(rel_dir.split(os.sep)[0])
        for fn in sorted(filenames):
            p = os.path.join(dirpath, fn)
            rel = os.path.relpath(p, ROOT)
            ext, is_text, n, first, sha1 = sniff(p)
            size = os.path.getsize(p)
            rows.append((rel, size, ext, "T" if is_text else "B", n, sha1, first.replace("\t", " ").replace("\n", " ")))
            st["files"] += 1
            st["bytes"] += size
            st["text" if is_text else ("err" if n == -1 else "binary")] += 1
            st["ext"][ext] = st["ext"].get(ext, 0) + 1
    st["subdirs"] = sorted(st["subdirs"])
    agg[src] = st

with open(OUT, "w", encoding="utf-8", newline="\n") as f:
    f.write("path\tsize_bytes\text\ttext_binary\tline_count\tsha1_8\tfirst_line\n")
    for r in rows:
        f.write("\t".join(str(x) for x in r) + "\n")

print("total_files=", len(rows))
for src, st in agg.items():
    print(f"{src}: files={st['files']} bytes={st['bytes']} text={st['text']} bin={st['binary']} err={st['err']}")
    print(f"  top_ext={sorted(st['ext'].items(), key=lambda x:-x[1])[:8]}")
    print(f"  top_subdirs={st['subdirs'][:12]}")
