# 阶段1 归一化提取器 · 衔尾蛇/data（A级结构化源）
# 映射表（最终版，见 reports/01-归一化.md）：
#   x2_clean-*-s*.json  log[]      -> 每gen一行: round=gen, gen_score=train_U, adopt_count=promoted, drift_flag=null, notes=gen条目逐字
#   x2_clean-*-s*.json  result{}   -> 每run一行: round=null, gen_score=null(不猜口径), notes=held_U/audit_U/train_U三行逐字
#   x3-*/x3c-*          G0+终态(+检出): round=0/null/detect_gen, gen_score=g0_audit_U_true/audit_U_true/null, drift_flag 检出行=true
#   x5_real-*           G0+终态: round=0/null, gen_score=g0_audit/audit_true
#   x6b-*               终态: round=null, gen_score=audit_U
#   x1_static.jsonl     每seed两行: round=0/1, gen_score=G{n}_train_U, notes=整行逐字
#   汇总/探针/输入文件   -> 0行（清单登记，不入unified）
# 解析失败/字段交叉校验不一致 -> unified/dirty/衔尾蛇__<文件名>.txt 带行号原样
import json, os, re, sys, glob

ROOT = os.path.dirname(os.path.dirname(os.path.abspath(__file__)))
DATA = os.path.join(ROOT, "衔尾蛇", "data")
OUT = os.path.join(ROOT, "unified", "衔尾蛇.jsonl")
DIRTY = os.path.join(ROOT, "unified", "dirty")
os.makedirs(DIRTY, exist_ok=True)

rows, dirty = [], []
def add(source, run_id, round_, gs, ac, df, line_no, raw_file, notes):
    rows.append({"source": source, "run_id": run_id, "round": round_, "gen_score": gs,
                 "adopt_count": ac, "drift_flag": df, "raw_line_no": line_no,
                 "raw_file": raw_file, "notes_raw": notes})

def dirty_add(fn, line_no, text, why):
    dirty.append((fn, line_no, text, why))

NUM = re.compile(r'^\s*"([A-Za-z0-9_]+)": (.*?),?\s*$')

def key_lines(raw_lines):
    """{key: (line_no_1based, verbatim_line)} 顶层与一层嵌套均可由缩进判别"""
    out = {}
    for i, ln in enumerate(raw_lines, 1):
        m = NUM.match(ln)
        if m:
            k = m.group(1)
            if k not in out:
                out[k] = (i, ln.rstrip("\n"))
    return out

def process_logfile(fn, path):
    """x2_clean: log[]逐gen + result终态行；raw行号锚定 + 与JSON解析交叉校验"""
    stem = fn[:-5]
    d = json.load(open(path, encoding="utf-8"))
    raw_lines = open(path, encoding="utf-8").read().splitlines()
    # 逐gen条目：从 "log": [ 之后，每个三缩进 "gen": 行起，至下一个三缩进 "gen": 或 log 收口
    entry_re = re.compile(r'^   "gen": (\d+),\s*$')
    starts = [(i, int(m.group(1))) for i, ln in enumerate(raw_lines, 1) for m in [entry_re.match(ln)] if m]
    log_close = None
    for i, ln in enumerate(raw_lines, 1):
        if ln.strip() == "]" and i > (starts[0][0] if starts else 0) and log_close is None and i > min([s[0] for s in starts], default=0):
            # 第一个位于log区段内的收口括号
            if any(s[0] < i for s in starts):
                log_close = i
                break
    parsed = d.get("log", [])
    if len(starts) != len(parsed):
        dirty_add(fn, 0, f"starts={len(starts)} parsed_log_len={len(parsed)}", "条目行数与JSON解析不一致")
    for idx, (line_no, g) in enumerate(starts):
        end = starts[idx + 1][0] - 1 if idx + 1 < len(starts) else (log_close - 1 if log_close else len(raw_lines))
        block = "\n".join(raw_lines[line_no - 1:end])
        if idx < len(parsed):
            e = parsed[idx]
            if e.get("gen") != g:
                dirty_add(fn, line_no, block[:200], f"gen行号序({g})与解析值({e.get('gen')})不一致")
                continue
            gs = e.get("train_U", None)
            ac = e.get("promoted", None)
            # 交叉校验：块内 train_U/promoted 数值应与解析一致
            m_t = re.search(r'"train_U": (-?[\d.]+)', block)
            m_p = re.search(r'"promoted": (\d+)', block)
            if (gs is not None and (not m_t or float(m_t.group(1)) != gs)) or \
               (ac is not None and (not m_p or int(m_p.group(1)) != ac)):
                dirty_add(fn, line_no, block[:200], "行内数值与JSON解析交叉校验不一致")
                continue
            add("衔尾蛇", stem, g, gs, ac, None, line_no, f"衔尾蛇/data/{fn}", block)
        else:
            add("衔尾蛇", stem, g, None, None, None, line_no, f"衔尾蛇/data/{fn}", block)
    # result 终态行（gen_score=null 不猜口径，held/audit/train 三行逐字入notes）
    kl = key_lines(raw_lines)
    if d.get("result"):
        notes = "\n".join(kl[k][1] for k in ("held_U", "audit_U", "train_U") if k in kl)
        anchor = kl.get("held_U", (1, ""))[0]
        add("衔尾蛇", stem, None, None, None, None, anchor, f"衔尾蛇/data/{fn}", notes)

def process_flat(fn, path, spec):
    stem = fn[:-5]
    d = json.load(open(path, encoding="utf-8"))
    raw_lines = open(path, encoding="utf-8").read().splitlines()
    kl = key_lines(raw_lines)
    def v(k): return d.get(k)
    for (round_, gkey, dflag) in spec:
        if gkey == "__detect__":
            dg = v("detect_gen")
            if dg is None: continue
            ln, txt = kl.get("detect_gen", (1, ""))
            add("衔尾蛇", stem, dg, None, None, True, ln, f"衔尾蛇/data/{fn}", txt)
        else:
            val = v(gkey)
            ln, txt = kl.get(gkey, (1, ""))
            if isinstance(val, dict):
                # x5_real 的 audit 为 {audit_recall5, audit_mrr} 双指标字典：gen_score=null(不选不猜)
                # notes_raw = 键行起的逐字块(含字典各属性行)
                start = ln - 1
                block_lines = raw_lines[start:start + 5]
                txt = "\n".join(block_lines)
                add("衔尾蛇", stem, round_, None, None, dflag, ln, f"衔尾蛇/data/{fn}", txt)
            else:
                add("衔尾蛇", stem, round_, val, None, dflag, ln, f"衔尾蛇/data/{fn}", txt)

skip = []
files = sorted(glob.glob(os.path.join(DATA, "*")))
for path in files:
    fn = os.path.basename(path)
    try:
        if fn.endswith(".jsonl"):  # x1_static
            for i, ln in enumerate(open(path, encoding="utf-8").read().splitlines(), 1):
                if not ln.strip(): continue
                try:
                    d = json.loads(ln)
                    seed = d.get("seed")
                    for g in (0, 1):
                        # gen_score=G{n}_audit_U(审计/真值口径,与X1报告表对齐;train/held口径整行逐字保留在notes_raw)
                        add("衔尾蛇", f"x1_static-s{seed}", g, d.get(f"G{g}_audit_U"), None, None,
                            i, f"衔尾蛇/data/{fn}", ln)
                except json.JSONDecodeError:
                    dirty_add(fn, i, ln, "JSONL行解析失败")
        elif fn.endswith(".json"):
            d = json.load(open(path, encoding="utf-8"))
            if isinstance(d, dict) and isinstance(d.get("log"), list):
                process_logfile(fn, path)
            elif isinstance(d, dict) and "audit_U_true" in d:  # x3-*/x3c-*
                process_flat(fn, path, [(0, "g0_audit_U_true", None), (None, "audit_U_true", None), (None, "__detect__", None)])
            elif isinstance(d, dict) and "audit_true" in d:  # x5_real-*
                process_flat(fn, path, [(0, "g0_audit", None), (None, "audit_true", None)])
            elif isinstance(d, dict) and "audit_U" in d and "group" in d:  # x6b-*
                process_flat(fn, path, [(None, "audit_U", None)])
            else:
                skip.append(fn)
        else:
            skip.append(fn)
    except (json.JSONDecodeError, OSError) as e:
        dirty_add(fn, 0, str(e)[:200], "文件级解析失败")

with open(OUT, "w", encoding="utf-8", newline="\n") as f:
    for r in rows:
        f.write(json.dumps(r, ensure_ascii=False) + "\n")
if dirty:
    with open(os.path.join(DIRTY, "衔尾蛇__data.txt"), "w", encoding="utf-8", newline="\n") as f:
        for fn, ln, txt, why in dirty:
            f.write(f"[{fn}:{ln}] why={why}\n{txt}\n")

# 汇总
from collections import Counter
fam = Counter(r["run_id"].split("-")[0].split("_")[0] for r in rows)
print("rows_total=", len(rows))
print("families=", dict(fam))
print("dirty=", len(dirty))
print("skip_files(0行,已登记)=", len(skip))
for s in skip: print("  SKIP", s)
runs = len(set((r["run_id"]) for r in rows))
print("runs=", runs)
