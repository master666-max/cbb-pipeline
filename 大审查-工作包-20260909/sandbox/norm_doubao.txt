import os, re, json

rows, dirty = [], []

def num(s):
    s = s.replace("**", "").strip()
    s = re.sub(r"[(（].*?[)）]\s*$", "", s).strip()
    m = re.match(r"^[+−\-]?\d+(\.\d+)?$", s)
    if not m: return None
    try: return float(s.replace("−", "-").replace("+", ""))
    except ValueError: return None

def first_num(s):
    m = re.search(r"[+−\-]?\d+\.?\d*", s.replace("**", ""))
    if not m: return None
    try: return float(m.group(0).replace("−", "-"))
    except ValueError: return None

def cells(line):
    return [c.strip() for c in line.strip().strip("|").split("|")]

def clean(s): return s.replace("**", "").replace("`", "").strip()

def val(cs, col, allow_first=True):
    if col >= len(cs): return None
    v = num(cs[col])
    if v is None and allow_first: v = first_num(cs[col])
    return v

def add(run_id, rnd, gs, line_no, fn, note, df=None):
    rows.append({"source": "豆包", "run_id": run_id, "round": rnd, "gen_score": gs,
                 "adopt_count": None, "drift_flag": df, "raw_line_no": line_no,
                 "raw_file": f"豆包/{fn}", "notes_raw": note})

def table(fn, sig, keycols, fmt, metrics, mode="score", round_col=None, gs_col=None, per_round=None):
    """metrics: [(col,suffix)]; per_round: [(col,round)] 逐代表"""
    lines = open(os.path.join(ROOT, "豆包", fn), encoding="utf-8").read().splitlines()
    hit = next((i for i, l in enumerate(lines) if l.rstrip() == sig), None)
    if hit is None:
        dirty.append((fn, 0, sig, "表头未找到(配置失效)")); return
    j = hit + 2
    while j < len(lines) and lines[j].strip().startswith("|"):
        cs = cells(lines[j])
        if len(cs) < 2: break
        kd = {f"k{i}": (clean(cs[kc]) if kc < len(cs) else "?") for i, kc in enumerate(keycols)}
        if len(keycols) == 1: kd["k"] = kd["k0"]
        base = fmt.format(**kd)
        if per_round:
            for col, rnd in per_round:
                gs = val(cs, col)
                if gs is None and col < len(cs) and cs[col] not in ("—", "-", ""):
                    dirty.append((fn, j + 1, lines[j], f"col{col}无法解析: {cs[col]}")); continue
                add(base, rnd, gs, j + 1, fn, lines[j].rstrip())
        else:
            for col, suf in metrics:
                gs = val(cs, col)
                if gs is None and col < len(cs) and cs[col] not in ("—", "-", ""):
                    dirty.append((fn, j + 1, lines[j], f"col{col}无法解析: {cs[col]}")); continue
                add(base + suf, None, gs, j + 1, fn, lines[j].rstrip())
        j += 1
