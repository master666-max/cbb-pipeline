class Ctx:
    """文件上下文：原文行 + 顺序游标锚点定位"""
    def __init__(self, name):
        self.name = name
        self.rel = "衔尾蛇/data/" + name
        with open(os.path.join(DATA, name), encoding="utf-8") as f:
            self.raw = f.read()
        self.lines = self.raw.splitlines()
        # JSONL 不做整档解析；.json 解析失败直接抛出，由外层捕获记 dirty
        self.data = json.loads(self.raw) if name.endswith(".json") else None
        self.cursor = 0
    def find(self, token):
        for i in range(self.cursor, len(self.lines)):
            if token in self.lines[i]:
                self.cursor = i + 1
                return i + 1
        for i in range(0, self.cursor):
            if token in self.lines[i]:
                return i + 1
        return None
    def anchor(self, token, desc):
        ln = self.find(token)
        if ln is None:
            dirty.append("%s:1:%s:锚点未找到[%s]（记录[%s]行号回退为1）" % (
                self.name, self.lines[0][:60] if self.lines else "", token, desc))
            return 1
        return ln
    def excerpt(self, line_no, maxlen=200):
        if line_no is None:
            return self.raw.strip()[:maxlen]
        parts, total = [], 0
        for i in range(line_no - 1, len(self.lines)):
            parts.append(self.lines[i])
            total += len(self.lines[i]) + 1
            if total >= maxlen:
                break
        return "\n".join(parts)[:maxlen]

def emit(run_id, rnd, score, adopt, drift, line_no, notes, raw_file):
    records.append({
        "source": "衔尾蛇",
        "run_id": run_id,
        "round": rnd,
        "gen_score": score,
        "adopt_count": adopt,
        "drift_flag": drift,
        "raw_line_no": line_no,
        "notes_raw": (notes or "")[:200],
        "raw_file": raw_file,
    })

def num(x):
    if isinstance(x, bool):
        return None
    return x if isinstance(x, (int, float)) else None

def adopt_frac(s):
    if isinstance(s, str):
        m = re.match(r"\s*(\d+)\s*/\s*\d+\s*$", s)
        if m:
            return int(m.group(1))
    return None

def G(gstr):
    return int(gstr[1:])

FAM = {}  # run_id 前缀族 -> 计数

def fam(tag):
    FAM[tag] = FAM.get(tag, 0) + 1
