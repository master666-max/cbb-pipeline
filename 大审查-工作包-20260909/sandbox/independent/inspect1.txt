def brief(v, depth=0, maxdepth=3):
    if depth >= maxdepth:
        return type(v).__name__
    if isinstance(v, dict):
        return {k: brief(x, depth+1, maxdepth) for k, x in list(v.items())[:12]}
    if isinstance(v, list):
        return [brief(v[0], depth+1, maxdepth), f"...len={len(v)}"] if v else []
    if isinstance(v, str):
        return v[:60]
    return v

    p = os.path.join(DATA, name)

    print("FILE:", name, "size=", os.path.getsize(p))
    try:
        with open(p, encoding="utf-8") as f:
            d = json.load(f)
    except Exception as e:
        print("  LOAD ERROR:", repr(e))
        continue
    print(json.dumps(brief(d, 0, 2), ensure_ascii=False, indent=1, default=str)[:3000])
