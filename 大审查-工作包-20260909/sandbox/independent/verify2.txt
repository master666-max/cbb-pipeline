# -*- coding: utf-8 -*-
import json, os, sys, io
sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding='utf-8', errors='replace')
BASE = r"D:\zcode专用！！！！危险！！！！！！！！！\大审查"
recs = [json.loads(l) for l in open(os.path.join(BASE, "sandbox", "independent", "xws_independent.jsonl"), encoding="utf-8")]

def dump(r):
    print(json.dumps(r, ensure_ascii=False)[:400])

print("== x2bc_results.json 的 w_imp_scan 记录（第 4-8 条 x2bc 记录） ==")
n = 0
for r in recs:
    if r["raw_file"].endswith("x2bc_results.json"):
        n += 1
        if 4 <= n <= 8:
            dump(r)

print("\n== x2bc x2c 行记录样例（第 9-10 条与最后一条） ==")
n = 0
xs = [r for r in recs if r["raw_file"].endswith("x2bc_results.json")]
for r in xs[8:10] + xs[-6:-4]:
    dump(r)

print("\n== x2d_stage1 全部 5 条 ==")
for r in recs:
    if r["run_id"] == "x2d_stage1":
        dump(r)

print("\n== x6c 行记录前 3 条 ==")
n = 0
for r in recs:
    if r["run_id"].startswith("x6c_e32-"):
        n += 1
        if n <= 3:
            dump(r)

print("\n== dirty 文件内容 ==")
print(open(os.path.join(BASE, "sandbox", "independent", "dirty_independent.txt"), encoding="utf-8").read())
