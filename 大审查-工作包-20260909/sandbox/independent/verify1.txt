# 抽查：随机+定点核验 raw_line_no 与 notes_raw 是否与原文一致
import json, os, sys, io, random
sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding='utf-8', errors='replace')
BASE = r"D:\zcode专用！！！！危险！！！！！！！！！\大审查"
DATA = os.path.join(BASE, "衔尾蛇", "data")
recs = [json.loads(l) for l in open(os.path.join(BASE, "sandbox", "independent", "xws_independent.jsonl"), encoding="utf-8")]
print("总行数:", len(recs))

rawcache = {}
def lines_of(rel):
    if rel not in rawcache:
        rawcache[rel] = open(os.path.join(BASE, rel.replace("/", os.sep)), encoding="utf-8").read().splitlines()
    return rawcache[rel]

# 1) 核验 notes_raw 是原文行前缀（从 raw_line_no 起的原文以 notes_raw 开头，或 notes_raw 是其截断）
bad = 0
for r in recs:
    ls = lines_of(r["raw_file"])
    ln = r["raw_line_no"]
    if ln < 1 or ln > len(ls):
        print("行号越界:", r["raw_file"], ln); bad += 1; continue
    joined = "\n".join(ls[ln - 1:])[:200]
    if joined.find(r["notes_raw"]) != 0:
        print("非逐字前缀:", r["raw_file"], ln, repr(r["notes_raw"][:60]), repr(joined[:60])); bad += 1
print("逐字核验异常数:", bad)
