# 细节值检查
import json, os, sys, io, difflib
sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding='utf-8', errors='replace')
DATA = r"D:\zcode专用！！！！危险！！！！！！！！！\大审查\衔尾蛇\data"
def raw(n):
    with open(os.path.join(DATA, n), encoding="utf-8") as f:
        return f.read()

print("== x4 全文 ==")
print(raw("x4_results.json"))
print("\n== x24 全文 ==")
print(raw("x24_summary.json"))
print("\n== x2bc coef0.25 vs 默认 diff ==")
a = raw("x2bc_results.json").splitlines()
b = raw("x2bc_results_coef0.25.json").splitlines()
for line in difflib.unified_diff(a, b, lineterm="", n=1):
    print(line)
print("\n== x6c summary.audit ==")
print(json.dumps(json.loads(raw("x6c_e32_summary.json"))["summary"]["audit"], ensure_ascii=False))
print("\n== x3b 全部 rows ==")
for r in json.loads(raw("x3b_valley.json"))["rows"]:
    print(json.dumps(r, ensure_ascii=False))
print("\n== x2d_stage1 原文行 ==")
for i, l in enumerate(raw("x2d_stage1.json").splitlines(), 1):
    print(i, l)
print("\n== x3_summary noiseind/noisecorr 组键 ==")
s = json.loads(raw("x3_summary.json"))
for scene, blk in s["summary"].items():
    print(scene, list(blk.keys()))

print(json.dumps(json.loads(raw("x1_probes.json"))["truncation"], ensure_ascii=False)[:400])

print(raw("x6b_summary.json")[:1200])
