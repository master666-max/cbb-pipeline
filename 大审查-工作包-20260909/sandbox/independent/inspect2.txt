def load(name):
    with open(os.path.join(DATA, name), encoding="utf-8") as f:
        return json.load(f)

def p(label, obj, n=2, maxlen=1800):
    print("-" * 70)
    print("##", label)
    if isinstance(obj, list):
        for x in obj[:n]:
            print(json.dumps(x, ensure_ascii=False, default=str)[:maxlen])
    else:
        print(json.dumps(obj, ensure_ascii=False, default=str)[:maxlen])
