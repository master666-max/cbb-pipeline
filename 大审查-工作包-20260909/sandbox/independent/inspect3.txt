import json, os, sys, io, hashlib
sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding='utf-8', errors='replace')

def raw(name):
    with open(os.path.join(DATA, name), encoding="utf-8") as f:
        return f.read()

def sha(name):
    return hashlib.sha256(raw(name).encode("utf-8")).hexdigest()[:16]

print("== x2bc 三个文件哈希 ==")
for n in ["x2bc_results.json", "x2bc_results_coef0.08.json", "x2bc_results_coef0.25.json"]:

print("\n== x5_real_summary.results 是否与独立文件一致 ==")
s = json.loads(raw("x5_real_summary.json"))
for key, arr in s["results"].items():
    parts = key.split("_")
    stem = f"x5_real-{parts[0]}-{parts[1]}"
    mism = 0
    for elem in arr:
        fn = f"{stem}-s{elem['seed']}.json"
        if not os.path.exists(os.path.join(DATA, fn)):
            print(key, "MISSING FILE", fn); continue
        fd = json.loads(raw(fn))
        # 文件里可能有额外字段，比较共有键
        diff = {k: (elem.get(k), fd.get(k)) for k in set(elem) & set(fd) if elem.get(k) != fd.get(k)}
        if diff:
            mism += 1
            print(key, "seed", elem["seed"], "DIFF", diff)
    print(key, "n=", len(arr), "mismatch_files=", mism)
print("g0 keys:", sorted(s["g0"].keys()), {k: sorted(v.keys()) for k, v in list(s["g0"].items())[:1]})
