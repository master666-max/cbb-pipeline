# review_r10.txt · R10-REVIEW 三件套机械操作（泛化版，覆盖 1-10 轮全档案）
# 运行: py -X utf8 sandbox/review_r10.txt
import csv, os, re

os.chdir(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

BUCKETS = {
    'B1评估噪声': ['噪声', 'σ', 'sigma', '评估', '假采纳', '口径', '漂移'],
    'B2世界景观': ['景观', '尖峰', '平坦', '世界', '切换', '目标', '遗忘', '恢复', '峰'],
    'B3门控阈限': ['margin', '门控', '阈', '护栏', '饥饿', '两阶段'],
    'B4采纳粒度': ['采纳', '参数', '粒度', '复合', '单参数', 'k='],
    'B5数据形态': ['run', '族', '序列', '跨', 'x2', 'x3', '豆包', '混元', '平台', '符号', '幅度', '点数', '档案', '收敛', '双峰', '增益'],
    'B6梯度信噪': ['梯度', '信噪', '曲率', '区域', 'U形'],
}

def bigrams(s):
    s = re.sub(r'\s', '', s)
    return set(s[i:i + 2] for i in range(len(s) - 1))

def jaccard(a, b):
    if not a or not b:
        return 0.0
    return len(a & b) / len(a | b)

rows = list(csv.DictReader(open('data-archive.csv', encoding='utf-8-sig')))
clusters = {b: [] for b in BUCKETS}
for r in rows:
    bc = r.get('boundary_conditions', '')
    for b, kws in BUCKETS.items():
        if any(k in bc for k in kws):
            clusters[b].append('%s轮%s' % (r['round'], r['hypothesis_id']))
print('== R1 边界合并 ==')
for b in sorted(clusters):
    if clusters[b]:
        print('%s (%d): %s' % (b, len(clusters[b]), ', '.join(clusters[b])))
n_clusters = sum(1 for b in clusters if clusters[b])
print('CLUSTERS_FOUND: %d' % n_clusters)

refuted = [r for r in rows if r['verdict'] == 'REFUTED']
print()
print('== R2 证伪复习 ==')
thr_fail = dir_hold = 0
for r in refuted:
    bc = r.get('boundary_conditions', '')
    m1 = bool(re.search(r'阈|先验|猜测', bc))
    m2 = bool(re.search(r'方向|成立|单调|复现', bc))
    thr_fail += m1
    dir_hold += m2
    print('| %s轮 %s | %s | 阈值/先验层失败=%s | 机制方向层成立=%s |' % (
        r['round'], r['hypothesis_id'], r['mechanical'], 'Y' if m1 else 'N', 'Y' if m2 else 'N'))
common = 'yes' if refuted and thr_fail == len(refuted) and dir_hold == len(refuted) else 'no'
print('REFUTED_COMMON_CAUSE: %s' % common)

nonempty = {b: v for b, v in clusters.items() if v}
cand = min(nonempty, key=lambda b: (len(nonempty[b]), b))
cand_text = ' '.join(r.get('boundary_conditions', '') for r in rows if any(k in r.get('boundary_conditions', '') for k in BUCKETS[cand]))
cand_bg = bigrams(cand_text)
print()
print('== R3 反例检索 ==')
print('下轮候选簇（覆盖最少）: %s (%d 假设)' % (cand, len(nonempty[cand])))
dists = sorted(((1 - jaccard(cand_bg, bigrams(r.get('boundary_conditions', ''))), r['round'], r['hypothesis_id'])
                for r in rows), reverse=True)
print('差异最大的 3 条历史假设:')
for d, rnd, hid in dists[:3]:
    print('| %.4f | %s轮 %s |' % (d, rnd, hid))
