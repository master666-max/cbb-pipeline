# 自测门运行器：analyze.txt 输出 vs expected.json 字符串级比对
import json, subprocess, sys, os, io

SB = os.path.dirname(os.path.abspath(__file__))
actual_txt = subprocess.run(
    [sys.executable, "-X", "utf8", os.path.join(SB, "analyze.txt"),
     os.path.join(SB, "synth", "s1.jsonl"), os.path.join(SB, "synth", "s2.jsonl"),
     os.path.join(SB, "synth", "s3.jsonl")],
    capture_output=True, text=True, encoding="utf-8").stdout
actual = json.loads(actual_txt)
expected = json.load(open(os.path.join(SB, "synth", "expected.json"), encoding="utf-8"))

fa = json.dumps(actual, ensure_ascii=False, sort_keys=True, indent=1)
fe = json.dumps(expected, ensure_ascii=False, sort_keys=True, indent=1)
if fa == fe:
    print("SELFTEST PASS: analyze.txt 输出与手算 expected.json 完全一致(字符串级)")
    sys.exit(0)

print("SELFTEST FAIL — 差异如下(左=脚本输出 右=手算):")
la, le = fa.splitlines(), fe.splitlines()
import difflib
for line in difflib.unified_diff(le, la, "expected(手算)", "actual(脚本)", lineterm="", n=1):
    print(line)
sys.exit(1)
