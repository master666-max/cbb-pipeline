# engine_r6.txt · Round 6 多峰景观引擎
# 运行: py -X utf8 sandbox/eng/engine_r6.txt
# 世界: U(x)=max_i h_i*exp(-||x-t_i||^2/(2 r_i^2)), 5峰; sigma=0.08; margin=0.05; step=0.35; G=50; 100 run
import random, json, os, math

os.chdir(os.path.dirname(os.path.dirname(os.path.dirname(os.path.abspath(__file__)))))  # 工作根=大审查/

G, STEP, SIGMA, MARGIN, N_RUNS = 50, 0.35, 0.08, 0.05, 100

def make_peaks(rng):
    for _ in range(1000):
        peaks = [(rng.uniform(-1, 1), rng.uniform(-1, 1), rng.uniform(0.5, 1.0), rng.uniform(0.15, 0.35))
                 for _ in range(5)]
        ok = True
        for i in range(5):
            for j in range(i + 1, 5):
                if math.hypot(peaks[i][0] - peaks[j][0], peaks[i][1] - peaks[j][1]) < 0.5:
                    ok = False
        if ok:
            return peaks
    return peaks  # 1000 次后放弃约束（概率极低）

def peak_u(x, p):
    d2 = (x[0] - p[0]) ** 2 + (x[1] - p[1]) ** 2
    return p[2] * math.exp(-d2 / (2 * p[3] ** 2))

def U_all(x, peaks):
    vals = [peak_u(x, p) for p in peaks]
    m = max(vals)
    return m, vals.index(m)

def run_once(seed):
    rng = random.Random(seed)
    peaks = make_peaks(rng)
    gmax = max(p[2] for p in peaks)
    x = (rng.uniform(-1, 1), rng.uniform(-1, 1))
    first_peak = None
    first_gen = None
    cur_peak = None
    switches = 0
    for g in range(1, G + 1):
        cand = (x[0] + rng.gauss(0, STEP), x[1] + rng.gauss(0, STEP))
        u_c, pc = U_all(cand, peaks)
        u_m, pm = U_all(x, peaks)
        if u_c + rng.gauss(0, SIGMA) > u_m + rng.gauss(0, SIGMA) + MARGIN:
            x = cand
            if cur_peak is not None and pc != cur_peak:
                switches += 1
            cur_peak = pc
            if first_peak is None and u_c > 0.4:
                first_peak = pc
                first_gen = g
    u_f, pf = U_all(x, peaks)
    return {'final_U': round(u_f, 10), 'trap_ratio': round(u_f / gmax, 10),
            'first_peak': first_peak, 'final_peak': pf,
            'stay': 1 if first_peak is not None and first_peak == pf else 0,
            'h_first': round(peaks[first_peak][2], 10) if first_peak is not None else None,
            'first_gen': first_gen, 'switches': switches, 'global_max': round(gmax, 10)}

def main():
    runs = [run_once(120000 + ri) for ri in range(N_RUNS)]
    traps = sorted(r['trap_ratio'] for r in runs)
    stays = sum(r['stay'] for r in runs)
    found = sum(1 for r in runs if r['first_peak'] is not None)
    print('median_trap=%.4f stay_rate=%.4f first_peak_found=%d/100' % (
        traps[len(traps) // 2], stays / N_RUNS, found))
    json.dump({'runs': runs}, open('sandbox/out/r6_results.json', 'w', encoding='utf-8'), ensure_ascii=False)
    print('written sandbox/out/r6_results.json')

if __name__ == '__main__':
    main()
