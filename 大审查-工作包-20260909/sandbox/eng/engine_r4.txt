# engine_r4.txt · Round 4 尖峰景观引擎（r2 骨架 + 窄高斯峰）
# 运行: py -X utf8 sandbox/eng/engine_r4.txt
# 世界: U(x)=exp(-||x-t||^2/(2*0.15^2)); sigma=0.08; margin=0.05; step=0.35; G=30; 100 run; seed=90000+run
import random, json, os, math

os.chdir(os.path.dirname(os.path.dirname(os.path.dirname(os.path.abspath(__file__)))))  # 工作根=大审查/

G, STEP, SIGMA, MARGIN, N_RUNS, R_SHARP = 30, 0.35, 0.08, 0.05, 100, 0.15

def U(x, t):
    d2 = (x[0] - t[0]) ** 2 + (x[1] - t[1]) ** 2
    return math.exp(-d2 / (2 * R_SHARP ** 2))

def run_once(seed):
    rng = random.Random(seed)
    t = (rng.uniform(-1, 1), rng.uniform(-1, 1))
    x = (rng.uniform(-1, 1), rng.uniform(-1, 1))
    u_init = U(x, t)
    events = []
    for g in range(1, G + 1):
        cand = (x[0] + rng.gauss(0, STEP), x[1] + rng.gauss(0, STEP))
        obs_c = U(cand, t) + rng.gauss(0, SIGMA)
        obs_m = U(x, t) + rng.gauss(0, SIGMA)
        if obs_c > obs_m + MARGIN:
            events.append({'g': g, 'dtrue': round(U(cand, t) - U(x, t), 10)})
            x = cand
    return {'final_U': round(U(x, t), 10), 'init_U': round(u_init, 10), 'events': events}

def main():
    runs = [run_once(90000 + ri) for ri in range(N_RUNS)]
    n_ad = sum(len(r['events']) for r in runs)
    fr = sum(1 for r in runs for e in r['events'] if e['dtrue'] <= 0)
    finals = sorted(r['final_U'] for r in runs)
    print('adopt=%d false=%d rate=%.4f final_mean=%.4f final_median=%.4f conv(>0.8)=%d' % (
        n_ad, fr, fr / n_ad, sum(finals) / len(finals), finals[len(finals) // 2],
        sum(1 for f in finals if f > 0.8)))
    json.dump({'cells': [{'sigma': SIGMA, 'margin': MARGIN, 'runs': runs}]},
              open('sandbox/out/r4_results.json', 'w', encoding='utf-8'), ensure_ascii=False)
    print('written sandbox/out/r4_results.json')

if __name__ == '__main__':
    main()
