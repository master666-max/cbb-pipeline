# judge_r2_noise.txt · Round 2 判据脚本（噪声×margin 边界）
# 用法: py -X utf8 sandbox/judges/judge_r2_noise.txt --selftest | --real
import json, sys, os

os.chdir(os.path.dirname(os.path.dirname(os.path.dirname(os.path.abspath(__file__)))))  # 工作根=大审查/

def false_rate(events):
    if not events:
        return None
    return sum(1 for e in events if e['dtrue'] <= 0) / len(events)

def cell_metrics(cell):
    runs = cell['runs']
    ev = [e for r in runs for e in r['events']]
    return {'sigma': cell['sigma'], 'margin': cell['margin'],
            'n_adopt': len(ev),
            'false_rate': false_rate(ev),
            'mean_final': sum(r['final_U'] for r in runs) / len(runs),
            'mean_init': sum(r['init_U'] for r in runs) / len(runs)}

def h1_rule(rates):
    """rates: {sigma: rate}（margin=0.05 三档）。返回 verdict, mechanical, monotone。"""
    sigmas = sorted(rates)
    mono = all(rates[sigmas[i]] < rates[sigmas[i + 1]] for i in range(len(sigmas) - 1))
    lo, hi = rates[0.02], rates[0.10]
    if lo < 0.05 and hi > 0.20 and mono:
        v = 'SUPPORTED'
    elif hi <= 0.10 or lo >= 0.05 or not mono:
        v = 'REFUTED'
    else:
        v = 'INCONCLUSIVE'
    return v, hi, mono

def h2_rule(finals):
    """finals: {margin: mean_final}（sigma=0.08 四档）。返回 verdict, mechanical(两两胜率), diff。"""
    interior = [finals[m] for m in (0.05, 0.15)]
    edge = [finals[m] for m in (0.0, 0.30)]
    wins = sum(1 for a in interior for b in edge if a > b)
    frac = wins / (len(interior) * len(edge))
    diff = sum(interior) / len(interior) - sum(edge) / len(edge)
    if min(interior) > max(edge) + 0.02:
        v = 'SUPPORTED'
    elif max(interior) <= max(edge):
        v = 'REFUTED'
    else:
        v = 'INCONCLUSIVE'
    return v, frac, diff

def selftest():
    ok = True
    # 用例1：事件级
    evA = [{'dtrue': 0.1}, {'dtrue': -0.05}]
    evB = [{'dtrue': 0.2}, {'dtrue': 0.0}]
    all_ev = evA + evB
    fr = false_rate(all_ev)
    mf = (0.8 + 0.6) / 2
    print('case1 false_rate got=%.4f exp=0.5000 %s' % (fr, 'PASS' if abs(fr - 0.5) < 5e-5 else 'FAIL'))
    print('case1 mean_final got=%.4f exp=0.7000 %s' % (mf, 'PASS' if abs(mf - 0.7) < 5e-5 else 'FAIL'))
    ok = ok and abs(fr - 0.5) < 5e-5 and abs(mf - 0.7) < 5e-5
    # 用例2：H1 规则
    v1, m1, mono = h1_rule({0.02: 0.03, 0.06: 0.15, 0.10: 0.35})
    print('case2 verdict=%s exp=SUPPORTED %s' % (v1, 'PASS' if v1 == 'SUPPORTED' else 'FAIL'))
    print('case2 mechanical got=%.4f exp=0.3500 %s' % (m1, 'PASS' if abs(m1 - 0.35) < 5e-5 else 'FAIL'))
    ok = ok and v1 == 'SUPPORTED' and abs(m1 - 0.35) < 5e-5
    # 用例3：H2 规则
    v2, m2, d2 = h2_rule({0.0: 0.45, 0.05: 0.80, 0.15: 0.70, 0.30: 0.40})
    print('case3 verdict=%s exp=SUPPORTED %s' % (v2, 'PASS' if v2 == 'SUPPORTED' else 'FAIL'))
    print('case3 mechanical got=%.4f exp=1.0000 %s' % (m2, 'PASS' if abs(m2 - 1.0) < 5e-5 else 'FAIL'))
    print('case3 diff got=%.4f exp=0.3250 %s' % (d2, 'PASS' if abs(d2 - 0.325) < 5e-5 else 'FAIL'))
    ok = ok and v2 == 'SUPPORTED' and abs(m2 - 1.0) < 5e-5 and abs(d2 - 0.325) < 5e-5
    print('SELFTEST', 'PASS' if ok else 'FAIL')
    return ok

def real():
    data = json.load(open('sandbox/out/r2_results.json', encoding='utf-8'))
    cms = [cell_metrics(c) for c in data['cells']]
    print('== cell 指标 ==')
    for cm in cms:
        print('sigma=%.2f margin=%.2f adopt=%d false_rate=%s mean_final=%.4f mean_init=%.4f' % (
            cm['sigma'], cm['margin'], cm['n_adopt'],
            ('%.4f' % cm['false_rate']) if cm['false_rate'] is not None else 'NA', cm['mean_final'], cm['mean_init']))
    rates = {cm['sigma']: cm['false_rate'] for cm in cms if cm['margin'] == 0.05}
    v1, m1, mono = h1_rule(rates)
    finals = {cm['margin']: cm['mean_final'] for cm in cms if abs(cm['sigma'] - 0.08) < 1e-9}
    v2, m2, d2v = h2_rule(finals)
    eff1 = rates[0.10] - rates[0.02]
    print('== H-R2-1 == verdict=%s mechanical=%.4f monotone=%s effect=%.4f' % (v1, m1, mono, eff1))
    print('== H-R2-2 == verdict=%s mechanical=%.4f effect=%.4f' % (v2, m2, d2v))

if __name__ == '__main__':
    if '--selftest' in sys.argv:
        sys.exit(0 if selftest() else 1)
    real()
