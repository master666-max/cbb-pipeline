# judge_r3_switch.txt · Round 3 判据脚本（世界切换×档案结构）
# 用法: py -X utf8 sandbox/judges/judge_r3_switch.txt --selftest | --real
import json, sys, os

os.chdir(os.path.dirname(os.path.dirname(os.path.dirname(os.path.abspath(__file__)))))  # 工作根=大审查/

def aggregate(runs):
    n = len(runs)
    return {'mean_retA': sum(r['retA'] for r in runs) / n,
            'mean_Bg': sum(r['Bg'] for r in runs) / n, 'n': n}

def h1_rule(m):
    """m: aggregate(single). 返回 verdict, mechanical."""
    if m['mean_retA'] < -0.10:
        v = 'SUPPORTED'
    elif m['mean_retA'] > -0.03:
        v = 'REFUTED'
    else:
        v = 'INCONCLUSIVE'
    return v, max(0.0, -m['mean_retA'])

def paired_frac(single, pareto, key, strict=True):
    n = len(single)
    c = 0
    for a, b in zip(single, pareto):
        if strict:
            c += 1 if b[key] > a[key] else 0
        else:
            c += 1 if b[key] >= a[key] else 0
    return c / n

def h2_rule(ms, mp, single_runs, pareto_runs):
    d = mp['mean_retA'] - ms['mean_retA']
    bg_ok = mp['mean_Bg'] >= ms['mean_Bg'] - 0.05
    if d > 0.10 and bg_ok:
        v = 'SUPPORTED'
    elif d <= 0:
        v = 'REFUTED'
    else:
        v = 'INCONCLUSIVE'
    mech = paired_frac(single_runs, pareto_runs, 'retA', strict=True)
    return v, mech, d

def h3_rule(ms, mp, single_runs, pareto_runs):
    d = mp['mean_Bg'] - ms['mean_Bg']
    if d >= -0.03:
        v = 'SUPPORTED'
    elif d < -0.10:
        v = 'REFUTED'
    else:
        v = 'INCONCLUSIVE'
    mech = paired_frac(single_runs, pareto_runs, 'Bg', strict=False)
    return v, mech, d

def selftest():
    ok = True
    single = [{'retA': -0.30, 'Bg': 0.40}, {'retA': -0.10, 'Bg': 0.20}]
    pareto = [{'retA': -0.05, 'Bg': 0.30}, {'retA': 0.05, 'Bg': 0.50}]
    ms, mp = aggregate(single), aggregate(pareto)
    print('mean_retA(single) got=%.4f exp=-0.2000 %s' % (ms['mean_retA'], 'PASS' if abs(ms['mean_retA'] + 0.2) < 5e-5 else 'FAIL'))
    print('mean_Bg(single)   got=%.4f exp=0.3000 %s' % (ms['mean_Bg'], 'PASS' if abs(ms['mean_Bg'] - 0.3) < 5e-5 else 'FAIL'))
    ok = ok and abs(ms['mean_retA'] + 0.2) < 5e-5 and abs(ms['mean_Bg'] - 0.3) < 5e-5
    v1, m1 = h1_rule(ms)
    print('H1 verdict=%s exp=SUPPORTED %s; mechanical got=%.4f exp=0.2000 %s' % (
        v1, 'PASS' if v1 == 'SUPPORTED' else 'FAIL', m1, 'PASS' if abs(m1 - 0.2) < 5e-5 else 'FAIL'))
    ok = ok and v1 == 'SUPPORTED' and abs(m1 - 0.2) < 5e-5
    v2, m2, d2 = h2_rule(ms, mp, single, pareto)
    print('H2 verdict=%s exp=SUPPORTED %s; mech got=%.4f exp=1.0000 %s; diff got=%.4f exp=0.2000 %s' % (
        v2, 'PASS' if v2 == 'SUPPORTED' else 'FAIL', m2, 'PASS' if abs(m2 - 1.0) < 5e-5 else 'FAIL',
        d2, 'PASS' if abs(d2 - 0.2) < 5e-5 else 'FAIL'))
    ok = ok and v2 == 'SUPPORTED' and abs(m2 - 1.0) < 5e-5 and abs(d2 - 0.2) < 5e-5
    v3, m3, d3 = h3_rule(ms, mp, single, pareto)
    print('H3 verdict=%s exp=SUPPORTED %s; mech got=%.4f exp=0.5000 %s; diff got=%.4f exp=0.1000 %s' % (
        v3, 'PASS' if v3 == 'SUPPORTED' else 'FAIL', m3, 'PASS' if abs(m3 - 0.5) < 5e-5 else 'FAIL',
        d3, 'PASS' if abs(d3 - 0.1) < 5e-5 else 'FAIL'))
    ok = ok and v3 == 'SUPPORTED' and abs(m3 - 0.5) < 5e-5 and abs(d3 - 0.1) < 5e-5
    print('SELFTEST', 'PASS' if ok else 'FAIL')
    return ok

def real():
    data = json.load(open('sandbox/out/r3_results.json', encoding='utf-8'))
    ms, mp = aggregate(data['single']), aggregate(data['pareto'])
    print('single: mean_retA=%.4f mean_Bg=%.4f (n=%d)' % (ms['mean_retA'], ms['mean_Bg'], ms['n']))
    print('pareto: mean_retA=%.4f mean_Bg=%.4f (n=%d)' % (mp['mean_retA'], mp['mean_Bg'], mp['n']))
    v1, m1 = h1_rule(ms)
    print('== H-R3-1 == verdict=%s mechanical=%.4f effect=%.4f' % (v1, m1, -ms['mean_retA']))
    v2, m2, d2 = h2_rule(ms, mp, data['single'], data['pareto'])
    print('== H-R3-2 == verdict=%s mechanical=%.4f effect=%.4f' % (v2, m2, d2))
    v3, m3, d3 = h3_rule(ms, mp, data['single'], data['pareto'])
    print('== H-R3-3 == verdict=%s mechanical=%.4f effect=%.4f' % (v3, m3, d3))

if __name__ == '__main__':
    if '--selftest' in sys.argv:
        sys.exit(0 if selftest() else 1)
    real()
