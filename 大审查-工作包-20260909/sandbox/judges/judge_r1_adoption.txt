# judge_r1_adoption.txt · Round 1 判据脚本（采纳微观机制）
# 用法: py -X utf8 sandbox/judges/judge_r1_adoption.txt --selftest | --real
# 锁定后不得修改（元层冻结）。toolbox: 只追加。
import json, re, sys, random, statistics, glob, os

os.chdir(os.path.dirname(os.path.dirname(os.path.dirname(os.path.abspath(__file__)))))  # 工作根=大审查/（judges/ 在 sandbox/ 下，需上溯三级）

MAIN_RE = re.compile(r'"mainline":\s*\{([^}]*)\}')

def parse_mainline(notes_raw):
    if not notes_raw:
        return None
    m = MAIN_RE.search(notes_raw)
    if not m:
        return None
    d = {}
    for part in m.group(1).split(','):
        part = part.strip().rstrip(',')
        if not part:
            continue
        kv = part.split(':')
        if len(kv) != 2:
            continue
        k = kv[0].strip().strip('"')
        try:
            d[k] = float(kv[1])
        except ValueError:
            return None
    return d

def k_changes(m0, m1):
    if m0 is None or m1 is None:
        return None
    keys = set(m0) | set(m1)
    return sum(1 for k in keys if m0.get(k) != m1.get(k))

def build_events(rows):
    """rows: list of dict（同一 run，按 round 升序）。返回事件列表。"""
    runs = {}
    for r in rows:
        rid = r['run_id']
        if r.get('round') is None or r.get('gen_score') is None:
            continue
        runs.setdefault(rid, []).append(r)
    events = []   # 采纳事件
    nomove = []   # 不采纳代 |Δ|
    first_adopt = {}  # run -> first adoption Δ
    gains = {}    # run -> (first_gen_score, last_gen_score)
    for rid, rs in runs.items():
        rs.sort(key=lambda x: x['round'])
        seq = [r for r in rs if isinstance(r['round'], int) or (isinstance(r['round'], float) and r['round'] == int(r['round']))]
        # 只保留连续整数对
        for a, b in zip(seq, seq[1:]):
            if int(b['round']) != int(a['round']) + 1:
                continue
            d = round(b['gen_score'] - a['gen_score'], 10)  # 浮点卫生：防 0.9-0.4=0.5000000000000001 破坏平局判定
            k = k_changes(parse_mainline(a.get('notes_raw')), parse_mainline(b.get('notes_raw')))
            ev = {'run': rid, 't': int(b['round']), 'delta': d, 'k': k}
            if b.get('adopt_count'):
                events.append(ev)
                if rid not in first_adopt:
                    first_adopt[rid] = d
            else:
                nomove.append(abs(d))
        if len(seq) >= 2:
            gains[rid] = (seq[0]['gen_score'], seq[-1]['gen_score'])
    return events, nomove, first_adopt, gains

def metrics(events, nomove, first_adopt, gains):
    adopts = [e for e in events if e['k'] is not None or True]  # 全部采纳事件（k可能None=解析失败，单独计）
    n_adopt = len(events)
    false_n = sum(1 for e in events if e['delta'] <= 0)
    rate = false_n / n_adopt if n_adopt else None
    g2 = [e['delta'] for e in events if e['k'] is not None and e['k'] >= 2]
    g1 = [e['delta'] for e in events if e['k'] == 1]
    g0n = sum(1 for e in events if e['k'] == 0)
    kn = sum(1 for e in events if e['k'] is None)
    m2 = statistics.mean(g2) if g2 else None
    m1 = statistics.mean(g1) if g1 else None
    diff = (m2 - m1) if (m2 is not None and m1 is not None) else None
    if g2 and g1:
        wins = sum((1 if a > b else 0.5 if a == b else 0) for a in g2 for b in g1)
        pairP = wins / (len(g2) * len(g1))
    else:
        pairP = None
    # 置换检验（H-R1-2 判据）
    perm_p = None
    if diff is not None and g2 and g1:
        rng = random.Random(42)
        pool = g2 + g1
        n2 = len(g2)
        cnt = 0
        for _ in range(10000):
            rng.shuffle(pool)
            pm2 = statistics.mean(pool[:n2])
            pm1 = statistics.mean(pool[n2:])
            if pm2 - pm1 >= diff:
                cnt += 1
        perm_p = cnt / 10000
    # H-R1-3 首采纳占比
    shares, excl = [], 0
    for rid, fd in first_adopt.items():
        if rid not in gains:
            continue
        gain = gains[rid][1] - gains[rid][0]
        if gain <= 0:
            excl += 1
            continue
        shares.append(fd / gain)
    med_share = statistics.median(shares) if shares else None
    med_nomove = statistics.median(nomove) if nomove else None
    return {'n_adopt': n_adopt, 'false_n': false_n, 'rate': rate,
            'mean_k2': m2, 'mean_k1': m1, 'diff': diff, 'pairP': pairP, 'perm_p': perm_p,
            'n_k2': len(g2), 'n_k1': len(g1), 'n_k0': g0n, 'k_parse_fail': kn,
            'n_runs_share': len(shares), 'excl_gain_le0': excl, 'median_share': med_share,
            'median_nomove_absd': med_nomove, 'n_nomove': len(nomove)}

def verdicts(mt):
    v1 = 'SUPPORTED' if mt['rate'] is not None and mt['rate'] <= 0.10 else 'REFUTED' if mt['rate'] is not None and mt['rate'] >= 0.25 else 'INCONCLUSIVE'
    if mt['rate'] is None:
        v1 = 'INCONCLUSIVE'
    if mt['diff'] is not None:
        if mt['diff'] > 0 and mt['perm_p'] < 0.05:
            v2 = 'SUPPORTED'
        elif mt['diff'] <= 0:
            v2 = 'REFUTED'
        else:
            v2 = 'INCONCLUSIVE'
    else:
        v2 = 'INCONCLUSIVE'
    if mt['median_share'] is not None:
        v3 = 'SUPPORTED' if mt['median_share'] > 0.5 else 'REFUTED' if mt['median_share'] < 0.3 else 'INCONCLUSIVE'
    else:
        v3 = 'INCONCLUSIVE'
    return {'H-R1-1': v1, 'H-R1-2': v2, 'H-R1-3': v3}

def fmt(x):
    return ('%.4f' % x) if isinstance(x, (int, float)) else str(x)

SYNTH = [

 {'run_id':'x2_clean-SYNTH-A','round':4,'gen_score':0.4,'adopt_count':1,'notes_raw':'"mainline": {"w_kw": 2.5, "fz": 1}'},
 {'run_id':'x2_clean-SYNTH-A','round':5,'gen_score':0.9,'adopt_count':1,'notes_raw':'"mainline": {"w_kw": 1.0, "fz": 1, "w_imp": 2.0}'},

 {'run_id':'x2_clean-SYNTH-B','round':3,'gen_score':0.7,'adopt_count':1,'notes_raw':'"mainline": {"w_kw": 3.0, "fz": 0, "w_imp": 1.0}'},
]

EXPECTED = {'n_adopt': 5, 'rate': 0.4000, 'mean_k2': 0.5000, 'mean_k1': 0.3000, 'diff': 0.2000,
            'pairP': 0.6667, 'median_share': 0.7778, 'median_nomove_absd': 0.0000}

def selftest():
    mt = metrics(*build_events(SYNTH))
    ok = True
    for key, exp in EXPECTED.items():
        got = mt[key]
        match = (abs(got - exp) < 5e-5) if isinstance(exp, float) else got == exp
        print('%-22s got=%-10s exp=%-10s %s' % (key, fmt(got), fmt(exp), 'PASS' if match else 'FAIL'))
        ok = ok and match
    # 附加：两 run 的首采纳占比分项核对
    print('note: median_share over 2 runs; perm_p 不在自测表内（合成样本量过小，判据阈值只作用于真实数据）')
    print('SELFTEST', 'PASS' if ok else 'FAIL')
    return ok

def real():
    rows = [json.loads(l) for l in open('unified/衔尾蛇.jsonl', encoding='utf-8')]
    x2 = [r for r in rows if str(r['run_id']).startswith('x2_clean')]
    mt = metrics(*build_events(x2))
    vs = verdicts(mt)
    print('== Round1 真实数据（unified/衔尾蛇.jsonl · x2_clean-*）==')
    for k in sorted(mt):
        print('%-24s %s' % (k, fmt(mt[k])))
    print('== 判定 ==')
    for h in sorted(vs):
        print(h, vs[h])

if __name__ == '__main__':
    if '--selftest' in sys.argv:
        sys.exit(0 if selftest() else 1)
    else:
        real()
