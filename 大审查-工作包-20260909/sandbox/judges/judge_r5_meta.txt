# judge_r5_meta.txt · Round 5 判据脚本（跨库元分析：早信号与平台期）
# 用法: py -X utf8 sandbox/judges/judge_r5_meta.txt --selftest | --real
import json, sys, os, glob, math

os.chdir(os.path.dirname(os.path.dirname(os.path.dirname(os.path.abspath(__file__)))))  # 工作根=大审查/

def ols_slope(pts):
    """pts: [(round, score)]。返回 OLS 斜率。"""
    n = len(pts)
    mx = sum(p[0] for p in pts) / n
    my = sum(p[1] for p in pts) / n
    den = sum((p[0] - mx) ** 2 for p in pts)
    if den == 0:
        return None
    return sum((p[0] - mx) * (p[1] - my) for p in pts) / den

def pearson(xs, ys):
    n = len(xs)
    mx, my = sum(xs) / n, sum(ys) / n
    sx = math.sqrt(sum((x - mx) ** 2 for x in xs))
    sy = math.sqrt(sum((y - my) ** 2 for y in ys))
    if sx == 0 or sy == 0:
        return None
    return sum((x - mx) * (y - my) for x, y in zip(xs, ys)) / (sx * sy)

def runs_from_rows(rows):
    runs = {}
    for r in rows:
        if r.get('round') is None or r.get('gen_score') is None:
            continue
        rd = r['round']
        if not (isinstance(rd, int) or (isinstance(rd, float) and rd == int(rd))):
            continue
        runs.setdefault(r['run_id'], []).append((int(rd), r['gen_score']))
    return {k: sorted(v) for k, v in runs.items() if len(v) >= 3}

def metrics(runs):
    agree_n = agree_ok = 0
    plate_n = plate_ok = 0
    early_slopes, gains = [], []
    for rid, pts in runs.items():
        if len(pts) >= 5:
            es = ols_slope(pts[:5])
            gain = pts[-1][1] - pts[0][1]
            if es is not None:
                agree_n += 1
                if (es > 0 and gain > 0) or (es < 0 and gain < 0) or (es == 0 and gain == 0):
                    agree_ok += 1
                early_slopes.append(es)
                gains.append(gain)
        if len(pts) >= 8:
            ls = ols_slope(pts[-5:])
            if ls is not None:
                plate_n += 1
                if abs(ls) < 0.005:
                    plate_ok += 1
    return {'n_h1': agree_n, 'agree': agree_ok / agree_n if agree_n else None,
            'n_h2': plate_n, 'plateau_frac': plate_ok / plate_n if plate_n else None,
            'pearson_r': pearson(early_slopes, gains) if len(early_slopes) >= 3 else None}

def h1_rule(agree):
    if agree >= 0.80:
        return 'SUPPORTED'
    if agree < 0.50:
        return 'REFUTED'
    return 'INCONCLUSIVE'

def h2_rule(frac):
    if frac >= 0.50:
        return 'SUPPORTED'
    if frac < 0.30:
        return 'REFUTED'
    return 'INCONCLUSIVE'

SYNTH_ROWS = [
 {'run_id': 'SYN-A', 'round': i + 1, 'gen_score': v} for i, v in enumerate([0.0, 0.1, 0.2, 0.25, 0.25, 0.25, 0.25, 0.25])
] + [
 {'run_id': 'SYN-B', 'round': i + 1, 'gen_score': v} for i, v in enumerate([0.5, 0.45, 0.35, 0.2, 0.15, 0.14, 0.13, 0.12])
] + [
 {'run_id': 'SYN-C', 'round': i + 1, 'gen_score': v} for i, v in enumerate([0.1, 0.3, 0.2, 0.25, 0.25])
]

def selftest():
    ok = True
    ptsA = [(i + 1, v) for i, v in enumerate([0.0, 0.1, 0.2, 0.25, 0.25, 0.25, 0.25, 0.25])]
    sA5 = ols_slope(ptsA[:5])
    sAlast = ols_slope(ptsA[-5:])
    print('A early_slope got=%.4f exp=0.0650 %s' % (sA5, 'PASS' if abs(sA5 - 0.065) < 5e-5 else 'FAIL'))
    print('A last5_slope got=%.4f exp=0.0000 %s' % (sAlast, 'PASS' if abs(sAlast) < 5e-5 else 'FAIL'))
    ok = ok and abs(sA5 - 0.065) < 5e-5 and abs(sAlast) < 5e-5
    sB = ols_slope([(1, 0.5), (2, 0.45), (3, 0.35), (4, 0.2), (5, 0.15)])
    print('B early_slope got=%.4f exp=-0.0950 %s' % (sB, 'PASS' if abs(sB + 0.095) < 5e-5 else 'FAIL'))
    ok = ok and abs(sB + 0.095) < 5e-5
    sC = ols_slope([(1, 0.1), (2, 0.3), (3, 0.2), (4, 0.25), (5, 0.25)])
    print('C early_slope got=%.4f exp=0.0250 %s' % (sC, 'PASS' if abs(sC - 0.025) < 5e-5 else 'FAIL'))
    ok = ok and abs(sC - 0.025) < 5e-5
    mt = metrics(runs_from_rows(SYNTH_ROWS))
    print('agree got=%.4f exp=1.0000 %s' % (mt['agree'], 'PASS' if abs(mt['agree'] - 1.0) < 5e-5 else 'FAIL'))
    print('n_h1 got=%d exp=3 %s' % (mt['n_h1'], 'PASS' if mt['n_h1'] == 3 else 'FAIL'))
    print('plateau_frac got=%.4f exp=0.5000 %s' % (mt['plateau_frac'], 'PASS' if abs(mt['plateau_frac'] - 0.5) < 5e-5 else 'FAIL'))
    print('n_h2 got=%d exp=2 %s' % (mt['n_h2'], 'PASS' if mt['n_h2'] == 2 else 'FAIL'))
    ok = ok and abs(mt['agree'] - 1.0) < 5e-5 and mt['n_h1'] == 3 and abs(mt['plateau_frac'] - 0.5) < 5e-5 and mt['n_h2'] == 2
    print('SELFTEST', 'PASS' if ok else 'FAIL')
    return ok

def real():
    rows = []
    for f in glob.glob('unified/*.jsonl'):
        rows += [json.loads(l) for l in open(f, encoding='utf-8')]
    runs = runs_from_rows(rows)
    fams = {}
    for rid in runs:
        fams[rid.split('-')[0]] = fams.get(rid.split('-')[0], 0) + 1
    mt = metrics(runs)
    print('eligible runs=%d families=%s' % (len(runs), fams))
    for k in ('n_h1', 'agree', 'n_h2', 'plateau_frac', 'pearson_r'):
        v = mt[k]
        print('%-14s %s' % (k, ('%.4f' % v) if isinstance(v, float) else v))
    print('== H-R5-1 == verdict=%s mechanical=%s effect=%s' % (
        h1_rule(mt['agree']), '%.4f' % mt['agree'], '%.4f' % (mt['agree'] - 0.5)))
    print('== H-R5-2 == verdict=%s mechanical=%s effect=%s' % (
        h2_rule(mt['plateau_frac']), '%.4f' % mt['plateau_frac'], '%.4f' % (mt['plateau_frac'] - 0.5)))

if __name__ == '__main__':
    if '--selftest' in sys.argv:
        sys.exit(0 if selftest() else 1)
    real()
