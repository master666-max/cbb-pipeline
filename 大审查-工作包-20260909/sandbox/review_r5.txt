# review_r5.txt · R5-REVIEW 三件套机械操作（边界合并/证伪复习/反例检索）
# 运行: py -X utf8 sandbox/review_r5.txt
# 输入: data-archive.csv（唯一档案源）+ reports/R5-REVIEW.md 由本脚本 stdout 组装
import csv, os, re

os.chdir(os.path.dirname(os.path.dirname(os.path.abspath(__file__))))  # 工作根=大审查/

BUCKETS = {
    'B1评估噪声': ['噪声', 'σ', 'sigma', '评估', '假采纳', '口径', '漂移'],
    'B2世界景观': ['景观', '尖峰', '平坦', '世界', '切换', '目标', '遗忘', '恢复'],
    'B3门控阈限': ['margin', '门控', '阈', '护栏', '饥饿'],
    'B4采纳粒度': ['采纳', '参数', '粒度', '复合', '单参数', 'k='],
    'B5数据形态': ['run', '族', '序列', '跨', 'x2', 'x3', '豆包', '混元', '平台', '符号', '幅度', '点数', '档案', '收敛', '双峰'],
}

def bigrams(s):
    s = re.sub(r'\s', '', s)
    return set(s[i:i + 2] for i in range(len(s) - 1))

def jaccard(a, b):
    if not a or not b:
        return 0.0
    return len(a & b) / len(a | b)

rows = list(csv.DictReader(open('data-archive.csv', encoding='utf-8-sig')))

# R1 边界合并
clusters = {b: [] for b in BUCKETS}
for r in rows:
    bc = r.get('boundary_conditions', '')
    for b, kws in BUCKETS.items():
        if any(k in bc for k in kws):
            clusters[b].append('%s轮%s' % (r['round'], r['hypothesis_id']))
print('== R1 边界合并（关键词聚类，多标签） ==')
for b in sorted(clusters):
    if clusters[b]:
        print('%s (%d): %s' % (b, len(clusters[b]), ', '.join(clusters[b])))
n_clusters = sum(1 for b in clusters if clusters[b])
print('CLUSTERS_FOUND: %d' % n_clusters)

# R2 证伪复习
refuted = [r for r in rows if r['verdict'] == 'REFUTED']
print()
print('== R2 证伪复习（verdict=REFUTED 全列） ==')
thr_fail = dir_hold = 0
for r in refuted:
    bc = r.get('boundary_conditions', '')
    m1 = bool(re.search(r'阈|先验|猜测', bc))
    m2 = bool(re.search(r'方向|成立|单调|复现', bc))
    thr_fail += m1
    dir_hold += m2
    print('| %s轮 %s | %s | 阈值/先验层失败标记=%s | 机制方向层成立标记=%s |' % (
        r['round'], r['hypothesis_id'], r['mechanical'], 'Y' if m1 else 'N', 'Y' if m2 else 'N'))
common = 'yes' if refuted and thr_fail == len(refuted) and dir_hold == len(refuted) else 'no'
print('REFUTED_COMMON_CAUSE: %s（全部证伪条目同时携带[阈值/先验层失败]与[机制方向层成立]标记=%s）' % (common, common == 'yes'))

# R3 反例检索：候选=覆盖最少的非空桶；反例=与该桶合并文本 bigram-Jaccard 距离最大的 3 条假设
nonempty = {b: v for b, v in clusters.items() if v}
cand = min(nonempty, key=lambda b: len(nonempty[b]))
cand_text = ' '.join(r.get('boundary_conditions', '') for r in rows
                     if cand.split(maxsplit=1)[-1][:2] in r.get('boundary_conditions', '') or any(
                         k in r.get('boundary_conditions', '') for k in BUCKETS[cand]))
cand_bg = bigrams(cand_text)
print()
print('== R3 反例检索 ==')
print('下轮候选簇（覆盖最少）: %s' % cand)
dists = sorted(((1 - jaccard(cand_bg, bigrams(r.get('boundary_conditions', ''))), r['round'], r['hypothesis_id'])
                for r in rows), reverse=True)
print('差异最大的 3 条历史假设（bigram-Jaccard 距离降序）:')
for d, rnd, hid in dists[:3]:
    print('| %.4f | %s轮 %s |' % (d, rnd, hid))
