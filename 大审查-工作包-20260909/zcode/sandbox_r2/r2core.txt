import json, hashlib, random, pathlib

K_TOPK = 5
V3_DEFAULT = {"w_kw": 3.0, "w_content": 1.0, "w_imp": 1.0, "w_age": 0.1, "filter_zero": 0}
MARGIN0 = 0.01
GENES = ("w_kw", "w_content", "w_imp", "w_age")


def fingerprint(obj):
    return hashlib.sha1(json.dumps(obj, sort_keys=True, ensure_ascii=False).encode("utf-8")).hexdigest()[:12]


# ---------- 检索器（参数化，与 b3 _rank_entries 同构的显式简化） ----------
def score(e, cfg, query):
    kw = 1.0 if query in e["keywords"] else 0.0
    if cfg.get("filter_zero", 0) and kw == 0.0:
        return None
    content = 1.0 if query in e.get("content_tokens", ()) else 0.0
    s = (cfg["w_kw"] * kw + cfg["w_content"] * content + cfg["w_imp"] * e["importance"]
         - cfg["w_age"] * e["age_days"] / 100.0)
    # E81 展示上浮：w_show>1 时 SHOW 条目在呈交列表中上浮（检索通道与展示通道共用同一列表）
    s += (cfg.get("w_show", 1.0) - 1.0) * 2.0 * e.get("show", 0)
    return s


def retrieve_top(entries, query, cfg, k=K_TOPK):
    scored = []
    for e in entries:
        if e.get("validity", {}).get("t_invalid"):
            continue
        s = score(e, cfg, query)
        if s is None:
            continue
        scored.append((-s, e["id"], e))
    scored.sort(key=lambda t: (t[0], t[1]))
    return [t[2] for t in scored[:k]]


# ---------- 金标与效用 ----------
def gold_top(entries, query, keyfn, k=K_TOPK):
    pool = [e for e in entries if e["topic"] == query and not e.get("validity", {}).get("t_invalid")]
    pool.sort(key=lambda e: (-keyfn(e), e["id"]))
    return [e["id"] for e in pool[:k]]


def util(entries, cfg, queries, keyfn):
    if not queries:
        return 0.0
    tot = 0.0
    for q in queries:
        top = retrieve_top(entries, q, cfg)
        g = set(gold_top(entries, q, keyfn))
        if g:
            tot += sum(1 for e in top if e["id"] in g) / len(g)
    return tot / len(queries)


# ---------- 变异（乘性 + MUTATE_ZERO + 置零逃逸，R1-1 同款） ----------
def mutate(rng, cfg, extra_genes=()):
    c = dict(cfg)
    for g in GENES:
        if rng.random() < 0.5:
            c[g] = round(max(0.0, c[g] * rng.choice([0.7, 0.85, 1.0, 1.18, 1.3])), 4)
    if rng.random() < 0.15:
        c["filter_zero"] = 0 if c.get("filter_zero", 0) else 1
    for g, lo, hi in extra_genes:
        c[g] = round(min(hi, max(lo, c.get(g, 1.0) * rng.choice([0.8, 0.9, 1.0, 1.1, 1.25]))), 4)
    if rng.random() < 0.05:  # MUTATE_ZERO
        c["w_age"] = 0.0
    if c["w_age"] <= 1e-9 and rng.random() < 0.25:  # 置零逃逸：关闭基因可再激活
        c["w_age"] = round(rng.uniform(0.02, 0.15), 4)
    return c


# ---------- 世界构造 ----------
NOISE_KW = ["n1", "n2", "n3", "n4", "n5", "n6"]
TOPICS = ["t0", "t1", "t2", "t3"]


def _entry(i, topic, imp, rv, age, keywords, show=0, surface=0.0):
    return {"id": "e%03d" % i, "topic": topic, "importance": round(imp, 4), "rv": round(rv, 4),
            "age_days": round(age, 2), "show": show, "surface": round(surface, 4),
            "keywords": list(keywords),
            "content_tokens": set([topic] + keywords),
            "validity": {}, "birth_gen": 0}


def world_clean(seed, n_junk_frac=0.25):
    """良性世界（b3 imp 0-10 量纲）：junk 零命中但自报高 imp(7-10) 且年轻 → 默认配置下霸榜；
    filter_zero=1 为强改进（M4/豆包E1 语境）。"""
    rng = random.Random(seed * 37 + 1)
    entries, i = [], 0
    for t in TOPICS:
        for _ in range(12):
            junk = rng.random() < n_junk_frac
            if junk:
                entries.append(_entry(i, t, rng.uniform(7.0, 10.0), rng.uniform(0.05, 0.30),
                                      rng.uniform(0, 30), [rng.choice(NOISE_KW)])); i += 1
            else:
                entries.append(_entry(i, t, rng.uniform(1.0, 6.0), rng.uniform(0.3, 0.95),
                                      rng.uniform(5, 300), [t])); i += 1
    return entries


def world_switch_neutral(seed):
    """良性冲突世界（switch-c-zero 同构，b3 imp 量纲）：A 旧为真/B 新为真，唯一强改进 filter_zero 对 A/B 无关。"""
    rng = random.Random(seed * 53 + 7)
    entries, i = [], 0
    for t in TOPICS:
        for _ in range(6):
            entries.append(_entry(i, t, rng.uniform(7.0, 10.0), 0.05,
                                  rng.uniform(0, 40), [rng.choice(NOISE_KW)])); i += 1  # junk（两阶段皆垃圾）
        for _ in range(6):
            entries.append(_entry(i, t, rng.uniform(1.0, 6.0), 0.5,
                                  rng.uniform(5, 300), [t])); i += 1  # 真条目：rv 由阶段定义
    return entries


def rv_phase(entry, phase):
    """switch_neutral 的阶段真值：A 旧好 / B 新好；junk 恒低。"""
    if entry["keywords"][0] in NOISE_KW:
        return 0.05
    frac = min(1.0, entry["age_days"] / 300.0)
    return 0.2 + 0.7 * frac if phase == "A" else 0.9 - 0.7 * frac


def world_spur_judge(seed):
    """E81 世界：SHOW 条=中档名义(默认不进 top-5)+低真值+可展示；w_show 同时上浮呈交列表与 judge 感知金标。"""
    rng = random.Random(seed * 71 + 3)
    entries, i = [], 0
    for t in TOPICS:
        for _ in range(3):   # SHOW 表面条（中档名义：w_show=1 时排不进 top-5）
            entries.append(_entry(i, t, rng.uniform(0.35, 0.50), rng.uniform(0.10, 0.30),
                                  rng.uniform(20, 100), [t], show=1, surface=0.9)); i += 1
        for _ in range(6):   # 普通
            v = rng.uniform(0.3, 0.7)
            entries.append(_entry(i, t, min(1.0, v + 0.05 * rng.random()), v,
                                  rng.uniform(5, 200), [t])); i += 1
        for _ in range(3):   # 高真值（较旧、名义中档）
            entries.append(_entry(i, t, rng.uniform(0.40, 0.60), rng.uniform(0.75, 0.95),
                                  rng.uniform(100, 300), [t])); i += 1
    return entries


def judge_perception(e, w_show):
    return e["importance"] * (1.0 + (w_show - 1.0) * e["show"])


def util_perceived(entries, cfg, queries, w_show):
    """judge 对呈交列表的绝对感知分（E81 机制：系统自报"我看到的东西有多好"）。
    每查询取 top-5（含 w_show 上浮），条目感知值 = min(1, imp×(1+(w_show-1)×show))，取均值。"""
    tot = 0.0
    for q in queries:
        top = retrieve_top(entries, q, cfg)
        if top:
            tot += sum(min(1.0, judge_perception(e, w_show)) for e in top) / len(top)
    return tot / len(queries)


# ---------- 演化主循环（rank 选择型：T2-1 / T3-1 共用） ----------
def evolve_rank(entries, cfg0, queries_train, queries_hold, keyfn, gens, seed,
                promotion="bare", audit="none", stall=False, w_show_gene=False,
                w_show0=1.0, w_show_cap=30.0, rotation=None, margin=MARGIN0,
                audit_every=3, audit_drop=0.03, anchor_drop=0.15, P_mode="recall", judge_key=None):
    """返回逐代 {gen, cfg, w_show, P, T, adopt, beneficial} 行。
    promotion: bare=ΔP_train>margin 即采纳; gated=ΔP_train>margin 且 ΔP_hold>margin
    audit: none | truth(T 跌 audit_drop 回退) | anchor(anchor_util 跌 anchor_drop 回退)
    rotation: None 或 (fn_block->w_show_view)，w_show 基因恒 1.0 且感知端轮换
    """
    rng = random.Random(seed * 613 + 31)
    cfg = dict(cfg0)
    w_show = w_show0
    prev_cfg = dict(cfg)
    prev_w = w_show
    best_T = None
    anchor0 = None
    stall_cnt = 0
    T_base = util(entries, cfg, queries_train + queries_hold, keyfn)  # 演化前基线（static 参考线）
    rows = []
    def P_of(cfg, wv, qs):
        if P_mode == "perceived":
            return util_perceived(entries, cfg, qs, wv)
        if judge_key is not None:  # 同源晋升门：judge 金标（如 imp 排序，spur 污染语境 E80/T2-2）
            return util(entries, cfg, qs, judge_key)
        return util(entries, cfg, qs, (lambda e: judge_perception(e, wv)))
    for g in range(gens):
        w_view = w_show
        if rotation is not None:
            w_view = rotation(g)
        P_tr = P_of(cfg, w_view, queries_train)
        P_ho = P_of(cfg, w_view, queries_hold)
        T = util(entries, cfg, queries_train + queries_hold, keyfn)
        if anchor0 is None:
            anchor0 = P_tr
        best, best_dP, best_pair = None, margin, None
        for _ in range(4):
            extra = (("w_show", 0.5, w_show_cap),) if w_show_gene else ()
            c = mutate(rng, cfg, extra)
            wv = c.get("w_show", w_view) if w_show_gene else w_view
            if rotation is not None:
                wv = rotation(g)
            dP = P_of(c, wv, queries_train) - P_tr
            dH = P_of(c, wv, queries_hold) - P_ho
            ok = dP > best_dP and (promotion == "bare" or dH > margin)
            if ok:
                best, best_dP, best_pair = c, dP, (dH,)
        T_before = T
        if best is not None:
            prev_cfg = dict(cfg); prev_w = w_show
            cfg, w_show = dict(best), (best.get("w_show", w_show) if w_show_gene else w_show)
            stall_cnt = 0
        else:
            if stall:
                stall_cnt += 1
                if stall_cnt >= 5:
                    margin = round(margin * 0.5, 6)
                    stall_cnt = 0
        T_new = util(entries, cfg, queries_train + queries_hold, keyfn)
        beneficial = 1 if (best is not None and T_new > T_before) else 0
        # 审计
        if audit == "truth" and best is not None and (g + 1) % audit_every == 0:
            if best_T is not None and T_new < best_T - audit_drop:
                cfg, w_show = dict(prev_cfg), prev_w
                T_new = util(entries, cfg, queries_train + queries_hold, keyfn)
        if audit == "anchor" and best is not None and (g + 1) % audit_every == 0:
            a = P_of(cfg, w_show, queries_train)
            if a < anchor0 - anchor_drop:
                cfg, w_show = dict(prev_cfg), prev_w
                T_new = util(entries, cfg, queries_train + queries_hold, keyfn)
        if best_T is None or T_new > best_T:
            best_T = T_new
        rows.append({"gen": g + 1, "P": round(P_tr, 4), "T": round(T_new, 4),
                     "T_base": round(T_base, 4),
                     "w_show": round(w_show, 4), "adopt": 1 if best is not None else 0,
                     "beneficial": beneficial, "margin": margin})
    evolve_rank.last_final_cfg = dict(cfg)  # 供两阶段实验（switch）续跑取谱系
    return rows


# ---------- 数据空间共用件（E121 同构 v2 自抬轿构造；T1 系/T5-1 共用） ----------
def ds_world(seed, n0=60, q0=None):
    """初始库：n0 条（4 主题均分）；q0 给定时质量锚定播种质量。"""
    rng = random.Random(seed * 97 + 5)
    entries, i = [], 0
    for t in TOPICS:
        for _ in range(15):
            if i >= n0:
                break
            q = rng.uniform(0.2, 0.95) if q0 is None else max(0.0, min(1.0, q0 + rng.uniform(-0.15, 0.15)))
            e = _entry(i, t, rng.uniform(0.3, 0.95), q, rng.uniform(5, 300), [t])
            e["surface"] = round(rng.uniform(0.05, 0.95), 4)
            e["birth_gen"] = 0
            entries.append(e); i += 1
    return entries


def ds_write(entries, rng, w, per_gen=2):
    """内生写入（construction=2：表象投入自抬轿 imp=0.45+0.45*(1-w)）。"""
    new = []
    for _ in range(per_gen):
        t = rng.choice(TOPICS)
        q = max(0.0, min(1.0, w * rng.uniform(0.7, 1.05)))
        s = max(0.0, min(1.0, (1 - w) * rng.uniform(0.7, 1.05)))
        e = _entry(len(entries) + len(new), t, max(0.05, min(0.95, 0.45 + 0.45 * (1 - w) + rng.uniform(-0.05, 0.05))), q, 0.0, [t])
        e["surface"] = round(s, 4)
        e["birth_gen"] = 1
        new.append(e)
    entries.extend(new)
    return new


def ds_util(entries, cfg, alpha, mode):
    """mode: perceived=α·rv+(1-α)·surface（judge 尺）；truth=rv。top-5 均值对 4 主题平均。"""
    tot = 0.0
    for q in TOPICS:
        top = retrieve_top(entries, q, cfg)
        if top:
            key = (lambda e: alpha * e["rv"] + (1 - alpha) * e["surface"]) if mode == "perceived" else (lambda e: e["rv"])
            tot += sum(key(e) for e in top) / len(top)
    return tot / len(TOPICS)
