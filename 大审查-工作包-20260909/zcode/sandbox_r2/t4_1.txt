# -*- coding: utf-8 -*-
# t4_1.txt — T4-1 importance 实测化进引擎主循环（N5；判据预注册：
#   spur: measured_gate >= claimed+0.05 ; clean: >= claimed-0.01 ; 冷启动 nogate 崩/ gate 不崩（X6c 锚点））
# 机制：检索用 imp_eff = (measured=adopted/shown, 若 shown>=3) else claimed（覆盖门）；轨迹来自检索事件本身。
import importlib.util, importlib.machinery, pathlib, json, statistics, sys, random
sys.stdout.reconfigure(encoding="utf-8", errors="replace")
HERE = pathlib.Path(__file__).resolve().parent
_ld = importlib.machinery.SourceFileLoader("r2core", str(HERE / "r2core.txt"))
_sp = importlib.util.spec_from_loader("r2core", _ld)
rc = importlib.util.module_from_spec(_sp); _ld.exec_module(rc)
SEEDS, GENS = 8, 40
OUT = HERE / "out_t4_1"; OUT.mkdir(exist_ok=True)

def run(seed, world_fn, arm):
    entries = world_fn(seed)
    rng = random.Random(seed * 613 + 31)
    cfg = dict(rc.V3_DEFAULT)
    stats = {}  # id -> [shown, adopted]
    keyT = (lambda e: e["rv"])
    def imp_eff(e):
        sh, ad = stats.get(e["id"], (0, 0))
        if arm == "claimed": return e["importance"]
        if sh == 0: return 0.0 if arm == "measured_nogate" else e["importance"]
        if arm == "measured_gate" and sh < 3: return e["importance"]  # 覆盖门：观测不足回退自称
        return (ad / sh) * 10.0  # 实测率映射到 0-10 量纲
    for g in range(GENS):
        P = rc.util(entries, cfg, rc.TOPICS[:2], (lambda e: e["rv"]))
        # 记录使用轨迹（全主题检索事件）
        for q in rc.TOPICS:
            top = rc.retrieve_top(entries, q, cfg)
            gset = set(rc.gold_top(entries, q, keyT))
            for e in top:
                st = stats.setdefault(e["id"], [0, 0]); st[0] += 1
                if e["id"] in gset: st[1] += 1
        best, db = None, rc.MARGIN0
        for _ in range(4):
            c = rc.mutate(rng, cfg)
            du = rc.util(entries, c, rc.TOPICS[:2], keyT) - P
            if du > db: best, db = c, du
        if best is not None: cfg = dict(best)
    # 终局用 imp_eff 口径评估（引擎实际检索行为）
    save = {e["id"]: e["importance"] for e in entries}
    for e in entries: e["importance"] = imp_eff(e)
    T = rc.util(entries, cfg, rc.TOPICS, keyT)
    for e in entries: e["importance"] = save[e["id"]]
    return round(T, 4)

rows = []
for wname, wfn in (("clean", rc.world_clean), ("spur", lambda s: rc.world_clean(s, n_junk_frac=0.5))):
    for arm in ("claimed", "measured_nogate", "measured_gate"):
        for s in range(1, SEEDS + 1):
            rows.append({"world": wname, "arm": arm, "seed": s, "T_end": run(s, wfn, arm)})
(OUT / "rows_t4_1.json").write_text(json.dumps(rows, ensure_ascii=False, indent=1), encoding="utf-8")

def m(w, a): return round(statistics.mean(r["T_end"] for r in rows if r["world"] == w and r["arm"] == a), 4)
print("=== T_end（实测口径检索下的终局真值） ===")
for w in ("clean", "spur"):
    print("%-6s claimed=%.4f measured_nogate=%.4f measured_gate=%.4f" % (
        w, m(w, "claimed"), m(w, "measured_nogate"), m(w, "measured_gate")))
print()
c1 = m("spur", "measured_gate") >= m("spur", "claimed") + 0.05
c2 = m("clean", "measured_gate") >= m("clean", "claimed") - 0.01
print("判据1 spur +0.05: %s (%+.4f) ; 判据2 clean -0.01: %s (%+.4f)" % (
    c1, m("spur", "measured_gate") - m("spur", "claimed"),
    c2, m("clean", "measured_gate") - m("clean", "claimed")))
print("冷启动对照: nogate vs gate (spur) %.4f vs %.4f — nogate 崩则覆盖门必要（X6c 锚点）" % (
    m("spur", "measured_nogate"), m("spur", "measured_gate")))
