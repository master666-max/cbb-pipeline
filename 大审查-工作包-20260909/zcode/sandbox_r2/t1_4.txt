# -*- coding: utf-8 -*-
# t1_4.txt — T1-4 写入×整理×审计 三元 2×2×2（判据预注册：三开优于任何二开(差>=+0.02)→正交互；
#   否则复合失配台账+1。锚点校验格（E123）：有审计无整理应为最差格）
import importlib.util, importlib.machinery, pathlib, json, random, statistics, sys
sys.stdout.reconfigure(encoding="utf-8", errors="replace")
HERE = pathlib.Path(__file__).resolve().parent
_ld = importlib.machinery.SourceFileLoader("r2core", str(HERE / "r2core.txt"))
_sp = importlib.util.spec_from_loader("r2core", _ld)
rc = importlib.util.module_from_spec(_sp); _ld.exec_module(rc)
SEEDS, GENS, ALPHA = 8, 40, 0.7
OUT = HERE / "out_t1_4"; OUT.mkdir(exist_ok=True)

def run(seed, audit_on, curate_on, w_evolvable):
    rng = random.Random(seed * 1013 + 17)
    entries = rc.ds_world(seed)
    cfg = dict(rc.V3_DEFAULT); w = 0.5
    base = rc.ds_util(entries, cfg, ALPHA, "truth")
    best_T, rollbacks = base, 0
    prev_cfg = dict(cfg)
    for g in range(GENS):
        rc.ds_write(entries, rng, w)
        P = rc.ds_util(entries, cfg, ALPHA, "perceived")
        T = rc.ds_util(entries, cfg, ALPHA, "truth")
        best, db = None, rc.MARGIN0
        for _ in range(4):
            c = rc.mutate(rng, cfg)
            du = rc.ds_util(entries, c, ALPHA, "perceived") - P
            if du > db: best, db = c, du
        if best is not None: cfg = dict(best)
        w2 = max(0.0, min(1.0, w + rng.uniform(-0.08, 0.08)))
        trial = [dict(e) for e in entries]
        probe = rc.ds_write(trial, rng, w2)
        du = rc.ds_util(trial, cfg, ALPHA, "perceived") - P
        if w_evolvable and du > rc.MARGIN0:
            w = w2; entries.extend(probe)
        elif not w_evolvable:
            entries.extend(probe)  # 冻结 0.5：照常写，只是不允许 w 被选择
        if curate_on and (g + 1) % 10 == 0:
            entries.sort(key=lambda e: e["rv"]); del entries[:5]
        if audit_on and (g + 1) % 3 == 0:
            Tn = rc.ds_util(entries, cfg, ALPHA, "truth")
            if Tn < best_T - 0.03:
                rollbacks += 1
                cfg = dict(prev_cfg)  # 审计回退（E81c 同构：检出即回退）
            if Tn > best_T: best_T = Tn
        prev_cfg = dict(cfg)
    T_end = rc.ds_util(entries, cfg, ALPHA, "truth")
    return {"seed": seed, "audit": audit_on, "curate": curate_on, "w_evo": w_evolvable,
            "gain": round(T_end - base, 4), "T_end": round(T_end, 4), "w_end": round(w, 4),
            "rollbacks": rollbacks}

rows = []
for audit_on in (False, True):
    for curate_on in (False, True):
        for w_evo in (False, True):
            rows += [run(s, audit_on, curate_on, w_evo) for s in range(1, SEEDS + 1)]
(OUT / "rows_t1_4.json").write_text(json.dumps(rows, ensure_ascii=False, indent=1), encoding="utf-8")

def m(a, c, w, f="gain"):
    return round(statistics.mean(r[f] for r in rows if r["audit"] == a and r["curate"] == c and r["w_evo"] == w), 4)

print("=== 2×2×2 真值增益（audit × curate × w_evo） ===")
print("%-10s %-8s %-8s" % ("", "w冻结0.5", "w可演化"))
for a in (False, True):
    for c in (False, True):
        print("a=%d c=%d  %+8.4f  %+8.4f" % (a, c, m(a, c, False), m(a, c, True)))
cells = {(a, c, w): m(a, c, w) for a in (0, 1) for c in (0, 1) for w in (0, 1)}
triple = cells[(1, 1, 1)]
best_two = max(v for k, v in cells.items() if sum(k) == 2)
print()
print("三开(1,1,1)=%+.4f ; 最优二开=%+.4f ; 差=%+.4f" % (triple, best_two, triple - best_two))
print("判据: 三开优于最优二开(差>=+0.02) -> %s ; 否则复合失配台账+1" % ("正交互成立" if triple - best_two >= 0.02 else "复合失配+1"))
print("E123 锚点格校验: 有审计无整理(1,0,·) 是否最差区: mean(1,0,·)=%.4f vs (0,0,·)=%.4f" % (
    (m(1, 0, 0) + m(1, 0, 1)) / 2, (m(0, 0, 0) + m(0, 0, 1)) / 2))
