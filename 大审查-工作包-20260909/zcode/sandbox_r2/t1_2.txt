# -*- coding: utf-8 -*-
# t1_2.txt — T1-2 整理经济学（判据预注册于 规划§4-T1-2；α=0.3 最污染档；8 seeds × 40 代）
#   a 分层: 只自写 收益 >= 全库的 80% -> 分层可行
#   b 混合裁判: judge 选池(16) + 真值终判 N 条; N=4 收益 >= 纯真值整里的 80% -> 成立（成本 16/周期 vs 全库约百条/周期）
#   c 定价: 周期{5,10,20} × 条数{4,8} 前沿 + 默认值
import importlib.util, importlib.machinery, pathlib, json, random, statistics, sys
sys.stdout.reconfigure(encoding="utf-8", errors="replace")
HERE = pathlib.Path(__file__).resolve().parent
_ld = importlib.machinery.SourceFileLoader("r2core", str(HERE / "r2core.txt"))
_sp = importlib.util.spec_from_loader("r2core", _ld)
rc = importlib.util.module_from_spec(_sp); _ld.exec_module(rc)
SEEDS, GENS, ALPHA = 8, 40, 0.3
OUT = HERE / "out_t1_2"; OUT.mkdir(exist_ok=True)

def run(seed, curate_mode, period=10, n_rm=5):
    rng = random.Random(seed * 1013 + 17)
    entries = rc.ds_world(seed)
    cfg = dict(rc.V3_DEFAULT); w = 0.5
    base = rc.ds_util(entries, cfg, ALPHA, "truth")
    cost = 0
    for g in range(GENS):
        rc.ds_write(entries, rng, w)
        P = rc.ds_util(entries, cfg, ALPHA, "perceived")
        best, db = None, rc.MARGIN0
        for _ in range(4):
            c = rc.mutate(rng, cfg)
            du = rc.ds_util(entries, c, ALPHA, "perceived") - P
            if du > db: best, db = c, du
        if best is not None: cfg = dict(best)
        w2 = max(0.0, min(1.0, w + rng.uniform(-0.08, 0.08)))
        trial = [dict(e) for e in entries]
        probe = rc.ds_write(trial, rng, w2)
        du = rc.ds_util(trial, cfg, ALPHA, "perceived") - P
        if du > rc.MARGIN0:
            w = w2; entries.extend(probe)
        if curate_mode != "none" and (g + 1) % period == 0:
            if curate_mode.startswith("mixed"):
                n = int(curate_mode.split("N")[1])
                pool = sorted(entries, key=lambda e: perceived(e))[:16]
                pool = sorted(pool, key=lambda e: e["rv"])[:n]
                ids = {id(e) for e in pool}; cost += 16
                entries[:] = [e for e in entries if id(e) not in ids]
            elif curate_mode in ("all", "self", "hot"):
                if curate_mode == "all":
                    cands = entries; cost += len(entries)
                elif curate_mode == "self":
                    cands = [e for e in entries if e.get("birth_gen", 0) > 0]; cost += len(cands)
                else:
                    hot_ids = set()
                    for q in rc.TOPICS:
                        hot_ids.update(id(e) for e in rc.retrieve_top(entries, q, cfg))
                    cands = [e for e in entries if id(e) in hot_ids]; cost += len(cands)
                if len(cands) > n_rm:
                    rm = sorted(cands, key=lambda e: e["rv"])[:n_rm]
                    ids = {id(e) for e in rm}
                    entries[:] = [e for e in entries if id(e) not in ids]
    return {"seed": seed, "mode": curate_mode, "period": period, "n": n_rm,
            "gain": round(rc.ds_util(entries, cfg, ALPHA, "truth") - base, 4),
            "T_end": round(rc.ds_util(entries, cfg, ALPHA, "truth"), 4), "judge_cost": cost}

def perceived(e): return ALPHA * e["rv"] + (1 - ALPHA) * e["surface"]

rows = []
MODES = ["none", "all", "self", "hot", "mixedN2", "mixedN4", "mixedN8"]
for seed in range(1, SEEDS + 1):
    for mode in MODES:
        rows.append(run(seed, mode))
for period in (5, 10, 20):
    for n in (4, 8):
        for seed in range(1, SEEDS + 1):
            rows.append(run(seed, "all", period=period, n_rm=n))
(OUT / "rows_t1_2.json").write_text(json.dumps(rows, ensure_ascii=False, indent=1), encoding="utf-8")

def m(f, **kw):
    v = [r[f] for r in rows if all(r[k] == v for k, v in kw.items())]
    return round(statistics.mean(v), 4)

print("=== a/b 分层与混合裁判（period10 n5 对齐比较） ===")
g_all, g_none = m("gain", mode="all"), m("gain", mode="none")
for mode in MODES:
    print("%-9s gain=%+.4f  judge_cost=%g" % (mode, m("gain", mode=mode), m("judge_cost", mode=mode)))
print("判据a: 只自写 %.4f >= 全库%.4f×0.8=%.4f -> %s" % (
    m("gain", mode="self"), g_all, g_all * 0.8, "成立" if m("gain", mode="self") >= g_all * 0.8 else "不成立"))
g_n4 = m("gain", mode="mixedN4")
print("判据b: mixedN4 %.4f >= 纯真值%.4f×0.8=%.4f -> %s (真值判定成本 16/周期 vs 全库)" % (
    g_n4, g_all, g_all * 0.8, "成立" if g_n4 >= g_all * 0.8 else "不成立"))
print()
print("=== c 整理定价（纯真值，全部=按 gain；cost=真值判定条数） ===")
for period in (5, 10, 20):
    for n in (4, 8):
        print("period=%2d n=%d: gain=%+.4f cost=%g" % (
            period, n, m("gain", mode="all", period=period, n=n), m("judge_cost", mode="all", period=period, n=n)))
