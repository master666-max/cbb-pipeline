# -*- coding: utf-8 -*-
# t1_1.txt — T1-1 污染传导路径解剖（预注册判据见 实验树规划 §4-T1-1）
# 问题：混元26轮残留①——α 从 0.3→0.95 增益从 −0.067→−0.004，但 writalloc 不动，传导路径未识别。
# 构造：E121 同构数据空间（60 条初始库，每代写 2 条；新条目 imp U(0.3,0.7) 只能零星进 top-5 → 间接稀释路径）。
#   中介封堵法：候选通道逐个钉死，看 α 梯度 Δ(α)=gain(0.95)−gain(0.3) 是否塌缩（|Δ|<=0.01）。
# 臂（5）：natural / freeze_weights(检索权重冻结) / gate_truth(采纳判据换真值) /
#          curate_truth(每10代删最低q×5) / freeze_write(不写入)。
# 预注册预测（开跑前写下）：freeze_write 塌缩（载体=写入累积）；gate_truth 塌缩；curate 收缩；
#                           freeze_weights 部分残留（预测记录，供事后对照）。
import importlib.util, importlib.machinery, pathlib, json, random, statistics, sys
sys.stdout.reconfigure(encoding="utf-8", errors="replace")
HERE = pathlib.Path(__file__).resolve().parent
_ld = importlib.machinery.SourceFileLoader("r2core", str(HERE / "r2core.txt"))
_sp = importlib.util.spec_from_loader("r2core", _ld)
rc = importlib.util.module_from_spec(_sp); _ld.exec_module(rc)

SEEDS, GENS, WRITE_PER_GEN = 8, 40, 2
ALPHAS = [0.3, 0.7, 0.95]
ARMS = ["natural", "freeze_weights", "gate_truth", "curate_truth", "freeze_write"]
OUT = HERE / "out_t1_1"; OUT.mkdir(exist_ok=True)
# v1 构造失败记录（2026-09-09）：新条目 imp 与 w 无关(0.3-0.7 随机) → natural 无 α 梯度(Δ=0.005)，无梯度可解剖。
# v2 构造变更（预注册机制假设）：表象投入自抬轿——新条目 imp = 0.45 + 0.45*(1-w) + U(-0.05,0.05)。
#   即系统越投表象，写出的条目自报重要度越高（垃圾自推广），低 α 下 judge 奖励这种写入 → 梯度载体候选。
CONSTRUCTION = 2


def new_entry_imp(w, rng):
    if CONSTRUCTION == 1:
        return rng.uniform(0.3, 0.7)
    return max(0.05, min(0.95, 0.45 + 0.45 * (1 - w) + rng.uniform(-0.05, 0.05)))


def world_data(seed):
    rng = random.Random(seed * 97 + 5)
    entries, i = [], 0
    for t in rc.TOPICS:
        for _ in range(15):
            e = rc._entry(i, t, rng.uniform(0.3, 0.95), rng.uniform(0.2, 0.95),
                          rng.uniform(5, 300), [t])
            e["surface"] = round(rng.uniform(0.05, 0.95), 4)
            entries.append(e); i += 1
    return entries


def perceived(e, alpha):
    return alpha * e["rv"] + (1 - alpha) * e["surface"]


def util_of(entries, cfg, alpha, mode):
    tot = 0.0
    for q in rc.TOPICS:
        top = rc.retrieve_top(entries, q, cfg)
        if top:
            key = (lambda e: perceived(e, alpha)) if mode == "perceived" else (lambda e: e["rv"])
            tot += sum(key(e) for e in top) / len(top)
    return tot / len(rc.TOPICS)


def write_entries(entries, rng, w, n=WRITE_PER_GEN):
    new = []
    for _ in range(n):
        t = rng.choice(rc.TOPICS)
        q = max(0.0, min(1.0, w * rng.uniform(0.7, 1.05)))
        s = max(0.0, min(1.0, (1 - w) * rng.uniform(0.7, 1.05)))
        e = rc._entry(len(entries) + len(new), t, new_entry_imp(w, rng), q, 0.0, [t])
        e["surface"] = round(s, 4)
        new.append(e)
    entries.extend(new)
    return new


def run(seed, alpha, arm):
    rng = random.Random(seed * 1013 + 17)
    entries = world_data(seed)
    cfg = dict(rc.V3_DEFAULT)
    w = 0.5
    base_T = util_of(entries, cfg, alpha, "truth")
    w_traj, gains = [w], []
    for g in range(GENS):
        if arm != "freeze_write":
            write_entries(entries, rng, w)
        if arm == "curate_truth" and (g + 1) % 10 == 0:
            entries.sort(key=lambda e: e["rv"])
            del entries[:5]
        P_tr = util_of(entries, cfg, alpha, "perceived")
        T_tr = util_of(entries, cfg, alpha, "truth")
        key = ("truth" if arm == "gate_truth" else "perceived")
        # cfg 变异（freeze_weights 冻结）
        if arm != "freeze_weights":
            best, d_best = None, rc.MARGIN0
            for _ in range(4):
                c = rc.mutate(rng, cfg)
                du = (util_of(entries, c, alpha, key) - P_tr) if key == "perceived" else (util_of(entries, c, alpha, key) - T_tr)
                if du > d_best:
                    best, d_best = c, du
            if best is not None:
                cfg = dict(best)
        # writalloc 变异（假设式写入：采纳才保留新条目）
        if arm not in ("freeze_write",):
            w2 = max(0.0, min(1.0, w + rng.uniform(-0.08, 0.08)))
            n0 = len(entries)
            probe = []
            for _ in range(WRITE_PER_GEN):
                t = rng.choice(rc.TOPICS)
                q = max(0.0, min(1.0, w2 * rng.uniform(0.7, 1.05)))
                s = max(0.0, min(1.0, (1 - w2) * rng.uniform(0.7, 1.05)))
                e = rc._entry(n0 + len(probe), t, new_entry_imp(w2, rng), q, 0.0, [t])
                e["surface"] = round(s, 4)
                probe.append(e)
            trial = entries + probe
            du = (util_of(trial, cfg, alpha, "perceived") if key == "perceived" else util_of(trial, cfg, alpha, "truth"))
            du -= (P_tr if key == "perceived" else T_tr)
            if du > rc.MARGIN0:
                w = w2
                entries.extend(probe)
        w_traj.append(w)
        gains.append(util_of(entries, cfg, alpha, "truth"))
    return {"seed": seed, "alpha": alpha, "arm": arm, "T_base": round(base_T, 4),
            "T_end": round(gains[-1], 4), "gain": round(gains[-1] - base_T, 4),
            "w_end": round(w, 4), "w_min": round(min(w_traj), 4), "w_max": round(max(w_traj), 4),
            "n_entries": len(entries),
            "world_fp": rc.fingerprint({"construction": CONSTRUCTION, "gen": GENS, "write": WRITE_PER_GEN,
                                        "alpha": alpha, "arm": arm,
                                        "default": rc.V3_DEFAULT, "margin": rc.MARGIN0})}


rows = []
for alpha in ALPHAS:
    for arm in ARMS:
        for seed in range(1, SEEDS + 1):
            rows.append(run(seed, alpha, arm))
(OUT / "rows_t1_1.json").write_text(json.dumps(rows, ensure_ascii=False, indent=1), encoding="utf-8")

def m(alpha, arm, f):
    v = [r[f] for r in rows if r["alpha"] == alpha and r["arm"] == arm]
    return round(statistics.mean(v), 4)

print("=== gain(真值增益) 按 α×臂 ===")
print("%-15s %9s %9s %9s %9s %8s" % ("arm", "a=0.3", "a=0.7", "a=0.95", "D(a)", "|D|<=0.01?"))
for arm in ARMS:
    g3, g7, g95 = m(0.3, arm, "gain"), m(0.7, arm, "gain"), m(0.95, arm, "gain")
    D = round(g95 - g3, 4)
    print("%-15s %9.4f %9.4f %9.4f %9.4f %8s" % (arm, g3, g7, g95, D, "塌缩" if abs(D) <= 0.01 else "残留"))
print()
print("=== writalloc 终值（对照 E121 的 0.52-0.59 不漂移） ===")
for alpha in ALPHAS:
    print("a=%.2f: " % alpha + "  ".join("%s=%.3f" % (arm, m(alpha, arm, "w_end")) for arm in ARMS))
print()
print("=== T_base / T_end（natural 参照） ===")
for alpha in ALPHAS:
    print("a=%.2f: base=%.4f end=%.4f" % (alpha, m(alpha, "natural", "T_base"), m(alpha, "natural", "T_end")))
