# -*- coding: utf-8 -*-
# t5_2.txt — T5-2 注意力时序分布（人类疲劳维度的第一刀：总量相同，时机不同）
# 世界：spur_judge（w_show 可演化，真值漂移内生）；审计只在"预算活跃窗"内可用。
# profile: front(第1-20代活跃)/uniform(全程)/back(第21-40代活跃)；总审计周期数相同(~7)。
# 判据：三分布 T_end 极差 >= 0.03 → 时机重要（总量之外新增自由度）。
import importlib.util, importlib.machinery, pathlib, json, statistics, sys, random
sys.stdout.reconfigure(encoding="utf-8", errors="replace")
HERE = pathlib.Path(__file__).resolve().parent
_ld = importlib.machinery.SourceFileLoader("r2core", str(HERE / "r2core.txt"))
_sp = importlib.util.spec_from_loader("r2core", _ld)
rc = importlib.util.module_from_spec(_sp); _ld.exec_module(rc)
SEEDS, GENS = 8, 40
OUT = HERE / "out_t5_2"; OUT.mkdir(exist_ok=True)
PROFILES = {"front": lambda g: g < 20, "uniform": lambda g: True, "back": lambda g: g >= 20}

def run(seed, profile):
    entries = rc.world_spur_judge(seed)
    rng = random.Random(seed * 613 + 31)
    cfg = dict(rc.V3_DEFAULT); w_show = 1.0
    keyT = (lambda e: e["rv"]); prev = (dict(cfg), w_show); best_T = None
    active = PROFILES[profile]
    for g in range(GENS):
        P = rc.util_perceived(entries, cfg, rc.TOPICS[:2], w_show)
        best, db = None, rc.MARGIN0
        for _ in range(4):
            c = rc.mutate(rng, cfg, extra_genes=(("w_show", 0.5, 30.0),))
            wv = c.get("w_show", w_show)
            du = rc.util_perceived(entries, c, rc.TOPICS[:2], wv) - P
            if du > db: best, db = c, du
        if best is not None:
            prev = (dict(cfg), w_show); cfg, w_show = dict(best), best.get("w_show", w_show)
        Tn = rc.util(entries, cfg, rc.TOPICS, keyT)
        if active(g) and (g + 1) % 3 == 0 and best is not None:
            if best_T is not None and Tn < best_T - 0.03:
                cfg, w_show = dict(prev[0]), prev[1]
                Tn = rc.util(entries, cfg, rc.TOPICS, keyT)
            if best_T is None or Tn > best_T: best_T = Tn
    return {"seed": seed, "profile": profile, "T_end": round(Tn, 4), "w_show_end": round(w_show, 3)}

rows = [run(s, p) for p in PROFILES for s in range(1, SEEDS + 1)]
(OUT / "rows_t5_2.json").write_text(json.dumps(rows, ensure_ascii=False, indent=1), encoding="utf-8")
def m(p): return round(statistics.mean(r["T_end"] for r in rows if r["profile"] == p), 4)
vals = {p: m(p) for p in PROFILES}
print("=== 注意力时序（T_end） ===")
for p, v in vals.items(): print("%-8s %.4f" % (p, v))
print("极差=%.4f -> %s (阈值0.03)" % (max(vals.values()) - min(vals.values()),
      "时机重要" if max(vals.values()) - min(vals.values()) >= 0.03 else "只有总量重要"))
