# -*- coding: utf-8 -*-
# t5_1.txt — T5-1 双引擎共享世界（单智能体假设破冰；探索性叶：无 pass/fail，只登记现象）
# 模式：independent（两个独立世界池）/ shared（共享单池+双写者串行写入）；观测：各自真值、世界均质、交叉污染占比。
import importlib.util, importlib.machinery, pathlib, json, statistics, sys, random
sys.stdout.reconfigure(encoding="utf-8", errors="replace")
HERE = pathlib.Path(__file__).resolve().parent
_ld = importlib.machinery.SourceFileLoader("r2core", str(HERE / "r2core.txt"))
_sp = importlib.util.spec_from_loader("r2core", _ld)
rc = importlib.util.module_from_spec(_sp); _ld.exec_module(rc)
SEEDS, GENS, ALPHA = 6, 40, 0.7
OUT = HERE / "out_t5_1"; OUT.mkdir(exist_ok=True)

def engine_step(entries, cfg, w, rng, tag):
    rc.ds_write(entries, rng, w, ) if False else None
    # 写 1 条（本引擎署名）
    t = rng.choice(rc.TOPICS)
    q = max(0.0, min(1.0, w * rng.uniform(0.7, 1.05)))
    s = max(0.0, min(1.0, (1 - w) * rng.uniform(0.7, 1.05)))
    e = rc._entry(len(entries), t, max(0.05, min(0.95, 0.45 + 0.45 * (1 - w) + rng.uniform(-0.05, 0.05))), q, 0.0, [t])
    e["surface"] = round(s, 4); e["birth_gen"] = 1; e["writer"] = tag
    entries.append(e)
    P = rc.ds_util(entries, cfg, ALPHA, "perceived")
    best, db = None, rc.MARGIN0
    for _ in range(4):
        c = rc.mutate(rng, cfg)
        du = rc.ds_util(entries, c, ALPHA, "perceived") - P
        if du > db: best, db = c, du
    if best is not None: cfg = dict(best)
    w2 = max(0.0, min(1.0, w + rng.uniform(-0.08, 0.08)))
    trial = [dict(x) for x in entries]
    probe = rc.ds_write(trial, rng, w2, ) if False else None
    e2 = dict(e); e2.pop("writer", None)
    # 以 w2 假设写入 1 条
    t = rng.choice(rc.TOPICS)
    q = max(0.0, min(1.0, w2 * rng.uniform(0.7, 1.05)))
    s = max(0.0, min(1.0, (1 - w2) * rng.uniform(0.7, 1.05)))
    e3 = rc._entry(len(trial), t, max(0.05, min(0.95, 0.45 + 0.45 * (1 - w2) + rng.uniform(-0.05, 0.05))), q, 0.0, [t])
    e3["surface"] = round(s, 4); e3["birth_gen"] = 1
    trial.append(e3)
    if rc.ds_util(trial, cfg, ALPHA, "perceived") - P > rc.MARGIN0:
        w = w2; entries.append(e3)
    return cfg, w

def run(seed, mode):
    rA = random.Random(seed * 613 + 31); rB = random.Random(seed * 911 + 57)
    if mode == "independent":
        poolA = rc.ds_world(seed); poolB = rc.ds_world(seed, n0=60)
        poolB = [dict(e, id="b" + e["id"][1:]) for e in rc.ds_world(seed)]
    else:
        poolA = rc.ds_world(seed); poolB = poolA  # 共享同一列表对象
    cfgA = dict(rc.V3_DEFAULT); cfgB = dict(rc.V3_DEFAULT); wA = 0.5; wB = 0.5
    for g in range(GENS):
        if mode == "independent":
            cfgA, wA = engine_step(poolA, cfgA, wA, rA, "A")
            cfgB, wB = engine_step(poolB, cfgB, wB, rB, "B")
        else:
            cfgA, wA = engine_step(poolB, cfgA, wA, rA, "A")   # 共享池，串行写
            cfgB, wB = engine_step(poolB, cfgB, wB, rB, "B")
    TA = rc.ds_util(poolA if mode == "independent" else poolB, cfgA, ALPHA, "truth")
    TB = rc.ds_util(poolB if mode == "independent" else poolB, cfgB, ALPHA, "truth")
    mq = statistics.mean(e["rv"] for e in poolB)
    if mode == "shared":
        a_written = [e for e in poolB if e.get("writer") == "A"]
        topB_ids = {e["id"] for q in rc.TOPICS for e in rc.retrieve_top(poolB, q, cfgB)}
        cross = statistics.mean([1.0 if e["id"] in topB_ids else 0.0 for e in a_written]) if a_written else 0.0
    else:
        cross = 0.0
    return {"seed": seed, "mode": mode, "T_A": round(TA, 4), "T_B": round(TB, 4),
            "world_mean_q": round(mq, 4), "n_world": len(poolB), "A_entries_in_B_top5": round(cross, 3)}

rows = [run(s, md) for md in ("independent", "shared") for s in range(1, SEEDS + 1)]
(OUT / "rows_t5_1.json").write_text(json.dumps(rows, ensure_ascii=False, indent=1), encoding="utf-8")
def m(md, f): return round(statistics.mean(r[f] for r in rows if r["mode"] == md), 4)
print("=== 双引擎（探索性观测，6 seeds × 40 代） ===")
for md in ("independent", "shared"):
    print("%-12s T_A=%.4f T_B=%.4f 世界均质=%.4f 世界大小=%d 交叉污染A→B top5=%.3f" % (
        md, m(md, "T_A"), m(md, "T_B"), m(md, "world_mean_q"), m(md, "n_world"),
        m(md, "A_entries_in_B_top5")))
