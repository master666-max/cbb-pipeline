# -*- coding: utf-8 -*-
# t2_2.txt — T2-2 E80 负交互第三次构造（N2；预注册判据：任一护栏臂 T < static-0.02 → 复现；
#   全组合仍 >= static-0.02 → 三振收线，"修正架构下结构性不可达"升格定理候选（与 X6a 同待遇））
# 触发假设组装（R1-2 精化：同源门+伪特征放大+采纳密集，而非加危险）：
#   同源晋升门(judge imp 金标) × margin{0.01, 0.002放松} × 变异幅度{常,大(×2)} × spur密度0.5 × 60代长窗
#   护栏栈=v4h：双门+真值审计(3/0.03)+停摆监控
import importlib.util, importlib.machinery, pathlib, json, random, statistics, sys
sys.stdout.reconfigure(encoding="utf-8", errors="replace")
HERE = pathlib.Path(__file__).resolve().parent
_ld = importlib.machinery.SourceFileLoader("r2core", str(HERE / "r2core.txt"))
_sp = importlib.util.spec_from_loader("r2core", _ld)
rc = importlib.util.module_from_spec(_sp); _ld.exec_module(rc)
SEEDS, GENS = 8, 60
OUT = HERE / "out_t2_2"; OUT.mkdir(exist_ok=True)

def run(seed, margin, big_mut):
    entries = rc.world_clean(seed, n_junk_frac=0.5)
    rng_scale = (0.5, 0.7, 1.0, 1.4, 2.0) if big_mut else (0.7, 0.85, 1.0, 1.18, 1.3)
    rng = random.Random(seed * 613 + 31)
    cfg = dict(rc.V3_DEFAULT)
    margin_cur = margin
    prev_cfg = dict(cfg); best_T = None; stall = 0
    q_tr, q_ho = rc.TOPICS[:2], rc.TOPICS[2:]
    keyT = (lambda e: e["rv"]); keyJ = (lambda e: e["importance"])
    for g in range(GENS):
        P_tr = rc.util(entries, cfg, q_tr, keyJ)
        P_ho = rc.util(entries, cfg, q_ho, keyJ)
        T = rc.util(entries, cfg, q_tr + q_ho, keyT)
        best, db = None, margin_cur
        for _ in range(4):
            c = dict(cfg)
            for gg in rc.GENES:
                if rng.random() < 0.5:
                    c[gg] = round(max(0.0, c[gg] * rng.choice(rng_scale)), 4)
            if rng.random() < 0.15: c["filter_zero"] = 0 if c["filter_zero"] else 1
            if rng.random() < 0.05: c["w_age"] = 0.0
            dP = rc.util(entries, c, q_tr, keyJ) - P_tr
            dH = rc.util(entries, c, q_ho, keyJ) - P_ho
            if dP > db and dH > rc.MARGIN0:
                best, db = c, dP
        if best is not None:
            prev_cfg = dict(cfg); cfg = dict(best); stall = 0
        else:
            stall += 1
            if stall >= 5: margin_cur = round(margin_cur * 0.5, 6); stall = 0
        Tn = rc.util(entries, cfg, q_tr + q_ho, keyT)
        if (g + 1) % 3 == 0 and best is not None:
            if best_T is not None and Tn < best_T - 0.03:
                cfg = dict(prev_cfg)
                Tn = rc.util(entries, cfg, q_tr + q_ho, keyT)
        if best_T is None or Tn > best_T: best_T = Tn
    return Tn

# static 参考
def static_T(seed):
    entries = rc.world_clean(seed, n_junk_frac=0.5)
    return rc.util(entries, dict(rc.V3_DEFAULT), rc.TOPICS, (lambda e: e["rv"]))

ARMS = {"v4(m0.01)": (0.01, False), "v4+m002": (0.002, False),
        "v4+big": (0.01, True), "v4+both": (0.002, True)}
rows = []
for seed in range(1, SEEDS + 1):
    st = round(static_T(seed), 4)
    for arm, (mg, bm) in ARMS.items():
        t = round(run(seed, mg, bm), 4)
        rows.append({"seed": seed, "arm": arm, "static_T": st, "T_end": t,
                     "vs_static": round(t - st, 4),
                     "world_fp": rc.fingerprint({"junk": 0.5, "gens": GENS, "margin": mg, "big": bm})})
(OUT / "rows_t2_2.json").write_text(json.dumps(rows, ensure_ascii=False, indent=1), encoding="utf-8")

print("=== E80 第三造（护栏栈 vs static，负值=护栏臂低于 static） ===")
print("%-12s %9s %9s %10s" % ("arm", "T_end", "static", "vs_static"))
worst = 0
for arm in ARMS:
    t = round(statistics.mean(r["T_end"] for r in rows if r["arm"] == arm), 4)
    st = round(statistics.mean(r["static_T"] for r in rows if r["arm"] == arm), 4)
    d = round(t - st, 4); worst = min(worst, d)
    print("%-12s %9.4f %9.4f %+10.4f" % (arm, t, st, d))
print()
print("判据: 最差护栏臂 vs static = %+.4f -> %s" % (
    worst, "E80 复现" if worst < -0.02 else "三振收线：修正架构下结构性不可达 → 定理候选"))
