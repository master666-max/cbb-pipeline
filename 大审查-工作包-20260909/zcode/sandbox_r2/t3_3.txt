# -*- coding: utf-8 -*-
# t3_3.txt — T3-3 自适应三条件的对抗验证（C3；预注册预测：
#   V1 双向+相对反馈 → 震荡；V2 单向+绝对反馈 → 单向收缩(饿死)；V3 双向+绝对+无界 → 越界发散。
#   ≥2/3 预测命中 → 修正85解释升格设计规则）
# 场景：E44 型停摆（clean 世界，初始 margin=0.15 致命档），控制器窗口 5 代。base=双向+绝对+有界(0.001-0.05)。
import importlib.util, importlib.machinery, pathlib, json, statistics, sys, random
sys.stdout.reconfigure(encoding="utf-8", errors="replace")
HERE = pathlib.Path(__file__).resolve().parent
_ld = importlib.machinery.SourceFileLoader("r2core", str(HERE / "r2core.txt"))
_sp = importlib.util.spec_from_loader("r2core", _ld)
rc = importlib.util.module_from_spec(_sp); _ld.exec_module(rc)
SEEDS, GENS, WIN = 6, 30, 5
OUT = HERE / "out_t3_3"; OUT.mkdir(exist_ok=True)

def run(seed, controller):
    entries = rc.world_clean(seed)
    rng = random.Random(seed * 613 + 31)
    cfg = dict(rc.V3_DEFAULT); margin = 0.15
    margins, adopts_hist = [margin], []
    prev_m = margin
    for g in range(GENS):
        n_adopt = 0
        for _ in range(4):
            c = rc.mutate(rng, cfg)
            du = rc.util(entries, c, rc.TOPICS[:2], (lambda e: e["rv"])) - \
                 rc.util(entries, cfg, rc.TOPICS[:2], (lambda e: e["rv"]))
            if du > margin:
                cfg = dict(c); n_adopt += 1
        adopts_hist.append(n_adopt)
        if (g + 1) % WIN == 0:
            a = sum(adopts_hist[-WIN:])
            if controller == "base":       # 双向+绝对+有界
                margin = margin / 2 if a == 0 else min(0.05, margin * 1.5)
            elif controller == "V1":       # 双向+相对反馈（目标 1 采纳/窗口）
                target = 1.0
                margin = max(0.001, margin * (1 + 0.8 * (target - a) / target))
            elif controller == "V2":       # 单向+绝对反馈（只能降）
                margin = margin / 2 if a == 0 else margin
                margin = max(0.0005, margin)
            elif controller == "V3":       # 双向+绝对+无界
                margin = margin / 2 if a == 0 else margin * 1.8
            margins.append(margin)
    osc = sum(1 for i in range(1, len(margins)) if (margins[i] - margins[i-1]) * (margins[i-1] - margins[max(0,i-1)]) < 0)
    T_end = rc.util(entries, cfg, rc.TOPICS, (lambda e: e["rv"]))
    return {"seed": seed, "controller": controller, "T_end": round(T_end, 4),
            "adopts": sum(adopts_hist), "margin_end": round(margins[-1], 6),
            "margin_max": round(max(margins), 6), "sign_changes": osc}

rows = [run(s, c) for c in ("base", "V1", "V2", "V3") for s in range(1, SEEDS + 1)]
(OUT / "rows_t3_3.json").write_text(json.dumps(rows, ensure_ascii=False, indent=1), encoding="utf-8")

def m(c, f): return round(statistics.mean(r[f] for r in rows if r["controller"] == c), 4)
print("=== 控制器对比（E44 停摆场景，margin0=0.15） ===")
print("%-5s %9s %8s %10s %10s %8s" % ("ctrl", "T_end", "adopts", "margin_end", "margin_max", "震荡次"))
for c in ("base", "V1", "V2", "V3"):
    print("%-5s %9.4f %8.2f %10.5f %10.5f %8.1f" % (c, m(c, "T_end"), m(c, "adopts"),
          m(c, "margin_end"), m(c, "margin_max"), m(c, "sign_changes")))
base_T = m("base", "T_end")
hits = []
if m("V1", "T_end") < base_T - 0.005 and m("V1", "sign_changes") >= 2: hits.append("V1震荡")
if m("V2", "adopts") <= m("base", "adopts") * 0.5 or m("V2", "margin_end") <= 0.002: hits.append("V2收缩")
if m("V3", "margin_max") > 0.15 and m("V3", "T_end") < base_T - 0.005: hits.append("V3越界")
print()
print("预测命中: %s -> %d/3 ; 修正85解释 %s" % (
    "+".join(hits) if hits else "无", len(hits), "升格设计规则" if len(hits) >= 2 else "不完整，登记缺口"))
